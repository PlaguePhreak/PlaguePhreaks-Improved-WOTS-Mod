	private void Awake()
	{
		instance = this;
		interval1Min = 60f;
		interval5Min = 300f;
		intervalHour = 3600f;

	private void Start()
	{
		base.gameObject.SetActive(value: false);

	private void FixedUpdate()
	{
		if (CampaignInterface.instance.map.timeSpeed != 0f)
		{
			campaignData.timerWeather += Time.deltaTime * CampaignInterface.instance.map.timeSpeed;
			int num = ModClock.Due(campaignData);
			if (((uint)num & 0x10u) != 0)
			{
				CheckWeekly();
			}
			if (((uint)num & 8u) != 0)
			{
				CheckDaily();
			}
			if (((uint)num & 4u) != 0)
			{
				CheckHourly();
			}
			if (((uint)num & 2u) != 0)
			{
				Check5Min();
			}
			if (((uint)num & (true ? 1u : 0u)) != 0)
			{
				Check1Min();
			}
		}
	}

	private void CheckCargoRetreatTimers(List<GameObjectAnimator> goa)
	{
		for (int i = 0; i < goa.Count; i++)
		{
			if (goa[i].mobileMapObject.postEngagementTimer > 0f)
			{
				goa[i].mobileMapObject.postEngagementTimer -= interval5Min;
				if (goa[i].mobileMapObject.postEngagementTimer <= 0f)
				{
					goa[i].mobileMapObject.postEngagementTimer = 0f;
				}
			}
		}
	}

	public void CheckAISeaIntercept()
	{
		for (int i = 0; i < otherMobile.Count; i++)
		{
			if (otherMobile[i].type != 0)
			{
				continue;
			}
			if (CampaignAI.instance.seaMissions[otherMobile[i].mobileMapObject.missionIndex].aggressive)
			{
				CampaignAI.instance.GetNewSpottedWaypoint(otherMobile[i], ModEnemyCampaign.ChaseRange(200000f));
			}
			if (CampaignAI.instance.seaMissions[otherMobile[i].mobileMapObject.missionIndex].airCoverPriority > 0 && campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour])
			{
				CampaignAI.instance.OnRequestAirCover(otherMobile[i].mobileMapObject);
			}
			if (otherMobile[i].mobileMapObject.patrolForHours <= 0 || !otherMobile[i].mobileMapObject.patrolZones.Contains(otherMobile[i].mobileMapObject.currentWaypoint))
			{
				continue;
			}
			otherMobile[i].mobileMapObject.patrolForHours--;
			if (otherMobile[i].mobileMapObject.patrolForHours <= 0)
			{
				otherMobile[i].mobileMapObject.patrolForHours = -1;
				CampaignAI.instance.ApplyNavDataToMobile(otherMobile[i], CampaignAI.instance.seaMissions[CampaignAI.instance.GetMissionByType(MissionType.Home)], string.Empty);
				GetNextNavPosition(otherMobile[i]);
				if (CampaignAI.instance.seaMissions[otherMobile[i].mobileMapObject.originalMissionIndex].autoRenew)
				{
					Debug.Log("Respawn Mission " + i);
					campaignData.recreateMissions.Add(otherMobile[i].mobileMapObject.originalMissionIndex);
				}
			}
		}
	}

	private bool IsOutsideAirOps()
	{
		return !campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour];
	}

	private void CheckAirEndurance(List<GameObjectAnimator> goa, bool updateWaypoint = false)
	{
		for (int num = goa.Count - 1; num > -1; num--)
		{
			if (goa[num].mobileMapObject.type == 1 && goa[num].mobileMapObject.endurance > 0f)
			{
				goa[num].mobileMapObject.endurance -= interval5Min;
				if (goa[num].mobileMapObject.endurance <= 0f || IsOutsideAirOps())
				{
					GameObjectAnimator gameObjectAnimator = goa[num];
					gameObjectAnimator.mobileMapObject.endurance = 0f;
					if (!AirAttemptLand(gameObjectAnimator))
					{
						AirReturnToBase(gameObjectAnimator);
						if (goa.Contains(gameObjectAnimator) && goa[num].mobileMapObject.currentFaction == 0)
						{
							if (goa[num].mobileMapObject.currentCampaignAction != 1)
							{
								ErrorPanel.instance.SetErrorMessage("BingoFuel", playerMobile[num].mobileMapObject.mobileName, playerMobile[num].mobileMapObject.createdByDisplayName, string.Empty);
								AutoPlayerManager.instance.ClearEscortData(goa[num]);
							}
							if (CampaignInterface.instance.enablePlotCourse)
							{
								CampaignInterface.instance.SetEnablePlotCourse(isOn: false);
								CampaignInterface.instance.SetCampaignCursor();
							}
							CampaignInterface.instance.PlotMobileCourse();
						}
					}
				}
				else if (updateWaypoint)
				{
					if (goa[num].mobileMapObject.currentCampaignAction == 1)
					{
						AirReturnToBase(goa[num]);
					}
					else
					{
						CampaignAI.instance.GetNewSpottedWaypoint(goa[num], goa[num].mobileMapObject.endurance * goa[num].mobileMapObject.speed);
					}
				}
				else if (goa[num].mobileMapObject.escortingThisGOA != null)
				{
					AutoPlayerManager.instance.MoveGoaToEscorted(goa[num]);
				}
			}
		}
	}

	private void Check1Min()
	{
		ModConvoy.Tick();
		ModAllied.Tick();
		ModEnemyScouts.Tick();
		playerEncounters = !playerEncounters;
		if (playerEncounters)
		{
			CheckMobileEncounters(playerMobile, otherMobile);
		}
		else
		{
			CheckMobileEncounters(otherMobile, playerMobile, scoutAir: true);
		}
		if (!campaignData.airOpsFirstHourPaused && campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour])
		{
			campaignData.airOpsFirstHourPaused = true;
			ErrorPanel.instance.SetErrorMessage("AirOpsStart", string.Empty, string.Empty, string.Empty);
			if (Config.instance.campaignPauseAtAirOpsStartHour)
			{
				CampaignInterface.instance.PauseCampaign();
			}
			ReplenishLostAircraft();
			AutoPlayerManager.instance.LaunchAutoScouts();
		}
		else
		{
			campaignData.airOpsFirstHourPaused = campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour];
		}
	}

	private void ReplenishLostAircraft()
	{
		List<string> list = new List<string>();
		for (int i = 0; i < landMapLocations.Count; i++)
		{
			if (Utilities.GetListSum(landMapLocations[i].destroyedAir) <= 0)
			{
				continue;
			}
			int[] airReplenishAtLocation = GetAirReplenishAtLocation(landMapLocations[i]);
			int num = airReplenishAtLocation.Length;
			if (!campaignData.usePortResourcesToResupplyAirfields)
			{
				if (!campaignData.replenishLimitAircraft)
				{
					landMapLocations[i].destroyedAir.Clear();
					for (int j = 0; j < num; j++)
					{
						landMapLocations[i].destroyedAir.Add(0);
					}
				}
				else
				{
					for (int k = 0; k < landMapLocations[i].destroyedAir.Count; k++)
					{
						landMapLocations[i].destroyedAir[k] -= airReplenishAtLocation[k];
						if (landMapLocations[i].destroyedAir[k] < 0)
						{
							landMapLocations[i].destroyedAir[k] = 0;
						}
					}
				}
			}
			else
			{
				List<string> airTypesAtLocation = GetAirTypesAtLocation(landMapLocations[i]);
				int currentFaction = landMapLocations[i].currentFaction;
				for (int l = 0; l < airTypesAtLocation.Count; l++)
				{
					int[] resourceCost = EngagementManager.instance.unitAirDictionary[airTypesAtLocation[l]].resourceCost;
					int num2 = 0;
					int num3 = landMapLocations[i].destroyedAir[l];
					for (int m = 0; m < num3; m++)
					{
						if (landMapLocations[i].destroyedAir[l] <= 0)
						{
							continue;
						}
						if (landMapLocations[i].troops[currentFaction] >= resourceCost[0] && landMapLocations[i].supplies[currentFaction] >= resourceCost[1] && landMapLocations[i].engineering[currentFaction] >= resourceCost[2] && landMapLocations[i].fuel[currentFaction] >= resourceCost[3])
						{
							landMapLocations[i].troops[currentFaction] -= resourceCost[0];
							landMapLocations[i].supplies[currentFaction] -= resourceCost[1];
							landMapLocations[i].engineering[currentFaction] -= resourceCost[2];
							landMapLocations[i].fuel[currentFaction] -= resourceCost[3];
							landMapLocations[i].destroyedAir[l]--;
							num2++;
							if (campaignData.replenishLimitAircraft && num2 >= airReplenishAtLocation[l])
							{
								break;
							}
						}
						else if (landMapLocations[i].currentFaction == 0 && !list.Contains(landMapLocations[i].locationName))
						{
							list.Add(landMapLocations[i].locationName);
						}
					}
				}
			}
			if (landMapLocations[i].currentFaction == 0 && CampaignInterface.instance.currentMapObject == landMapLocations[i].GetComponent<GameObjectAnimator>() && CampaignInterface.instance.panelTabs[1].activeSelf && !CampaignInterface.instance.dataPanelText.transform.parent.gameObject.activeSelf)
			{
				CampaignInterface.instance.OpenCreateAirStrike();
			}
		}
		if (list.Count <= 0)
		{
			return;
		}
		string text = string.Empty;
		int num4 = 0;
		for (int n = 0; n < list.Count; n++)
		{
			text = text + list[n] + "   ";
			num4++;
			if (num4 > 3)
			{
				text += "...";
				break;
			}
		}
		ErrorPanel.instance.SetErrorMessage("InsufficientAirbase", text, string.Empty, string.Empty);
	}

		return initialDestroyed;

	private void Check5Min()
	{
		for (int i = 0; i < playerMobile.Count; i++)
		{
			GameObjectAnimator gameObjectAnimator = playerMobile[i];
			if (gameObjectAnimator == null || gameObjectAnimator.mobileMapObject == null || gameObjectAnimator.mobileMapObject.type != 0 || gameObjectAnimator.mobileMapObject.unitPrefabs.Count == 0)
			{
				continue;
			}
			MobileMapObject mobileMapObject = gameObjectAnimator.mobileMapObject;
			float num = 10f;
			float fuelUsedDays = mobileMapObject.fuelUsedDays;
			if (gameObjectAnimator.isMoving)
			{
				mobileMapObject.fuelUsedDays += 0.0034722222f;
			}
			if (mobileMapObject.fuelUsedDays >= num)
			{
				mobileMapObject.fuelUsedDays = num;
				float num2 = 2.057776f;
				if (mobileMapObject.speed > num2 + 0.01f)
				{
					mobileMapObject.fuelSavedSpeed = mobileMapObject.speed;
					mobileMapObject.speed = num2;
					if (gameObjectAnimator.isMoving && mobileMapObject.mapWaypoints.Count > 0)
					{
						gameObjectAnimator.MoveTo(mobileMapObject.mapWaypoints[0], CampaignInterface.instance.GetMoveDuration(gameObjectAnimator.currentMap2DLocation, mobileMapObject.mapWaypoints[0], mobileMapObject.speed), DURATION_TYPE.Route);
					}
				}
				if (fuelUsedDays < num)
				{
					ErrorPanel.instance.SetErrorMessage("OutOfFuel", mobileMapObject.mobileName, string.Empty, string.Empty);
				}
			}
			else
			{
				if (mobileMapObject.fuelSavedSpeed > 0f)
				{
					mobileMapObject.speed = mobileMapObject.fuelSavedSpeed;
					mobileMapObject.fuelSavedSpeed = 0f;
					if (gameObjectAnimator.isMoving && mobileMapObject.mapWaypoints.Count > 0)
					{
						gameObjectAnimator.MoveTo(mobileMapObject.mapWaypoints[0], CampaignInterface.instance.GetMoveDuration(gameObjectAnimator.currentMap2DLocation, mobileMapObject.mapWaypoints[0], mobileMapObject.speed), DURATION_TYPE.Route);
					}
				}
				if (fuelUsedDays < num - 3f && mobileMapObject.fuelUsedDays >= num - 3f)
				{
					ErrorPanel.instance.SetErrorMessage("LowFuel", mobileMapObject.mobileName, Mathf.CeilToInt(num - mobileMapObject.fuelUsedDays).ToString(), string.Empty);
				}
			}
			if (CampaignInterface.instance.currentMapObject == gameObjectAnimator)
			{
				CampaignInterface.instance.mobileMapText.text = mobileMapObject.mobileName + "   (FUEL " + Mathf.Max(0f, num - mobileMapObject.fuelUsedDays).ToString("F1") + " DAYS)";
			}
		}
		if (campaignData.revealAllContacts)
		{
			for (int j = 0; j < otherMobile.Count; j++)
			{
				if (otherMobile[j] != null && otherMobile[j].mobileMapObject != null && otherMobile[j].mobileMapObject.unitPrefabs.Count > 0)
				{
					CampaignInterface.instance.SetContactColor(otherMobile[j].mobileMapObject);
				}
			}
		}
		for (int num3 = spottedContacts.Count - 1; num3 > -1; num3--)
		{
			if (spottedContacts[num3] != null)
			{
				spottedContacts[num3].spottedTimer -= interval5Min;
				CampaignInterface.instance.SetContactColor(spottedContacts[num3]);
				if (spottedContacts[num3].spottedTimer <= 0f)
				{
					spottedContacts[num3].spottedTimer = 0f;
					spottedContacts[num3].intelData.Clear();
					spottedContacts.RemoveAt(num3);
				}
			}
		}
		for (int k = 0; k < otherMobile.Count; k++)
		{
			if (otherMobile[k].mobileMapObject.type == 0 && !otherMobile[k].isMoving)
			{
				GetNextNavPosition(otherMobile[k]);
			}
		}
		for (int l = 0; l < playerMobile.Count; l++)
		{
			if (!playerMobile[l].isMoving && playerMobile[l].mobileMapObject.mapWaypoints.Count > 0)
			{
				playerMobile[l].MoveTo(playerMobile[l].mobileMapObject.mapWaypoints[0], CampaignInterface.instance.GetMoveDuration(playerMobile[l].currentMap2DLocation, playerMobile[l].mobileMapObject.mapWaypoints[0], playerMobile[l].mobileMapObject.speed), DURATION_TYPE.Route);
			}
		}
		List<MapLocation> list = new List<MapLocation>();
		List<MapLocation> list2 = new List<MapLocation>();
		for (int m = 0; m < landMapLocations.Count; m++)
		{
			if (landMapLocations[m].currentFaction == 0)
			{
				list.Add(landMapLocations[m]);
			}
			else
			{
				list2.Add(landMapLocations[m]);
			}
		}
		CheckLandSpotMobileObject(list, otherMobile);
		CheckLandSpotMobileObject(list2, playerMobile);
		CheckCargoRetreatTimers(playerMobile);
		CheckCargoRetreatTimers(otherMobile);
		CheckAirEndurance(playerMobile);
		CheckAirEndurance(otherMobile, updateWaypoint: true);
		CampaignAI.instance.CheckRecreateMissions();
		if (CampaignInterface.instance.currentMapObject != null && CampaignInterface.instance.currentMapObject.GetComponent<MobileMapObject>() != null && CampaignInterface.instance.currentMapObject.mobileMapObject.type == 1 && CampaignInterface.instance.currentMapObject.mobileMapObject.currentFaction == 0)
		{
			RefreshAirData(CampaignInterface.instance.currentMapObject.mobileMapObject);
		}
		for (int n = 0; n < landMapLocations.Count; n++)
		{
			if (!(landMapLocations[n].aircraftCooldown > 0f))
			{
				continue;
			}
			landMapLocations[n].aircraftCooldown -= interval5Min;
			if (!(landMapLocations[n].aircraftCooldown <= 0f))
			{
				continue;
			}
			landMapLocations[n].aircraftCooldown = 0f;
			if (landMapLocations[n].currentFaction == 0 && campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour])
			{
				ErrorPanel.instance.SetErrorMessage("AirReady", landMapLocations[n].locationName, string.Empty, string.Empty);
				if (Config.instance.campaignPauseAirfieldReady)
				{
					CampaignInterface.instance.PauseCampaign();
				}
			}
		}
		for (int num4 = 0; num4 < playerMobile.Count; num4++)
		{
			for (int num5 = 0; num5 < playerMobile[num4].mobileMapObject.aircraftCooldown.Count; num5++)
			{
				if (!(playerMobile[num4].mobileMapObject.aircraftCooldown[num5] != string.Empty))
				{
					continue;
				}
				float num6 = float.Parse(playerMobile[num4].mobileMapObject.aircraftCooldown[num5].Split('|')[0]);
				num6 -= interval5Min;
				if (num6 <= 0f)
				{
					playerMobile[num4].mobileMapObject.aircraftCooldown[num5] = string.Empty;
					if (campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour])
					{
						ErrorPanel.instance.SetErrorMessage("AirReady", playerMobile[num4].mobileMapObject.unitDisplayNames[num5], string.Empty, string.Empty);
						if (Config.instance.campaignPauseCarrierAirReady && Utilities.IsCarrier(playerMobile[num4].mobileMapObject.unitPrefabs[num5]))
						{
							CampaignInterface.instance.PauseCampaign();
						}
					}
				}
				else
				{
					playerMobile[num4].mobileMapObject.aircraftCooldown[num5] = num6.ToString();
				}
			}
		}
		for (int num7 = 0; num7 < otherMobile.Count; num7++)
		{
			for (int num8 = 0; num8 < otherMobile[num7].mobileMapObject.aircraftCooldown.Count; num8++)
			{
				if (otherMobile[num7].mobileMapObject.aircraftCooldown[num8] != string.Empty)
				{
					float num9 = float.Parse(otherMobile[num7].mobileMapObject.aircraftCooldown[num8].Split('|')[0]);
					num9 -= interval5Min;
					if (num9 <= 0f)
					{
						otherMobile[num7].mobileMapObject.aircraftCooldown[num8] = string.Empty;
					}
					else
					{
						otherMobile[num7].mobileMapObject.aircraftCooldown[num8] = num9.ToString();
					}
				}
			}
		}
	}

	private void CheckDaily()
	{
		WeatherManager.instance.UpdateWeather();
		CampaignAI.instance.MakeStrategicDecision();
		CheckTroopEngagements();
		ModSupply.Daily();
		ModConvoy.Daily();
		ModHistory.Daily();
		ModNews.Daily();
		ModAllied.Daily();
		CheckRepairedShips();
		string text = string.Empty;
		int num = 0;
		for (int num2 = campaignData.orderDaysRemaining.Count - 1; num2 > -1; num2--)
		{
			campaignData.orderDaysRemaining[num2] = campaignData.orderDaysRemaining[num2] - 1;
			if (campaignData.orderDaysRemaining[num2] <= 0)
			{
				string text2 = campaignData.orderShipClasses[num2];
				int num3 = campaignData.orderShipInstances[num2];
				int indexInDoubleList = GetIndexInDoubleList(text2, num3, campaignData.unavailableShipClasses, campaignData.unavailableShipInstances);
				if (indexInDoubleList != -1)
				{
					bool num4 = campaignData.unavailableShipInstanceNames.Count == campaignData.unavailableShipClasses.Count;
					campaignData.unavailableShipClasses.RemoveAt(indexInDoubleList);
					campaignData.unavailableShipInstances.RemoveAt(indexInDoubleList);
					if (num4)
					{
						campaignData.unavailableShipInstanceNames.RemoveAt(indexInDoubleList);
					}
				}
				string text3 = string.Empty;
				if (campaignData.orderPortNames != null && num2 < campaignData.orderPortNames.Count)
				{
					text3 = campaignData.orderPortNames[num2];
					campaignData.orderPortNames.RemoveAt(num2);
				}
				if (text3 != string.Empty)
				{
					MobileMapObject.SetHomePort(text2, num3, text3);
				}
				string shipDisplayName = Utilities.GetShipDisplayName(text2, string.Empty, num3);
				if (num > 0)
				{
					text += ", ";
				}
				text += shipDisplayName;
				num++;
				campaignData.orderShipClasses.RemoveAt(num2);
				campaignData.orderShipInstances.RemoveAt(num2);
				campaignData.orderDaysRemaining.RemoveAt(num2);
				if (UnitReference.instance.unitData != null && UnitReference.instance.unitData.unitType == UnitType.Sea)
				{
					UnitReference.instance.SetNamesInClassList(UnitReference.instance.namesInClassDropdown.value);
				}
			}
		}
		if (num == 1)
		{
			ErrorPanel.instance.SetErrorMessage("ShipDelivered", text, string.Empty, string.Empty);
		}
		else if (num > 1)
		{
			ErrorPanel.instance.SetErrorMessage("ShipsDelivered", text, string.Empty, string.Empty);
		}
	}

	public List<int> GetPlayerReplenish()
	{
		List<int> list = new List<int>(campaignData.replenish0);
		if (campaignData.campaignDominated)
		{
			for (int i = 0; i < list.Count; i++)
			{
				list[i] = Mathf.RoundToInt((float)list[i] * campaignData.winEnemyShipSunkSupplyIncrease);
			}
		}
		return list;
	}

	private void CheckWeekly()
	{
		List<int> list = new List<int>();
		List<int> list2 = new List<int>();
		int[] array = new int[6];
		for (int i = 0; i < landMapLocations.Count; i++)
		{
			if (IsHomePort(landMapLocations[i], 0) || IsHomePort(landMapLocations[i], 1))
			{
				int num = 0;
				if (landMapLocations[i].currentFaction == 0)
				{
					list = GetPlayerReplenish();
					list2 = campaignData.replenish0Max;
				}
				else
				{
					list = campaignData.replenish1;
					list2 = campaignData.replenish1Max;
					num = 1;
					campaignData.aiTotalTroops += list[0];
				}
				landMapLocations[i].troops[num] = Mathf.Clamp(landMapLocations[i].troops[num] + list[0], 0, list2[0]);

				if (!flag3 || !(mobileMapObject.postEngagementTimer <= 0f) || !(mobileMapObject2.postEngagementTimer <= 0f))
				{
					continue;
				}
				campaignData.subAmbush = false;
				campaignData.engagementWeatherIndex = WeatherManager.instance.GetMapObjectWeatherCellIndex(mobileMapObject.gameObjectAnimator.currentMap2DLocation);
				if (mobileMapObject.currentFaction == 0)
				{
					campaignData.playerObjectEngaged = playerMobile.IndexOf(mobileMapObject.gameObjectAnimator);
					campaignData.otherObjectEngaged = otherMobile.IndexOf(mobileMapObject2.gameObjectAnimator);
				}
				else
				{
					campaignData.playerObjectEngaged = playerMobile.IndexOf(mobileMapObject2.gameObjectAnimator);
					campaignData.otherObjectEngaged = otherMobile.IndexOf(mobileMapObject.gameObjectAnimator);
				}
				if (campaignData.playerObjectEngaged >= playerMobile.Count || campaignData.otherObjectEngaged >= otherMobile.Count || campaignData.playerObjectEngaged == -1 || campaignData.otherObjectEngaged == -1)
				{
					continue;
				}
				bool scoutOnly = IsMobileAircraftScoutOnly(otherMobile[campaignData.otherObjectEngaged].mobileMapObject);
				SpotContact(mobileMapObject2, mobileMapObject, null, scoutOnly);
				SpotContact(mobileMapObject, mobileMapObject2, null, scoutOnly);
				if (mobileMapObject.currentFaction == 0)
				{
					campaignData.playerObjectEngaged = playerMobile.IndexOf(mobileMapObject.gameObjectAnimator);
					campaignData.otherObjectEngaged = otherMobile.IndexOf(mobileMapObject2.gameObjectAnimator);
				}
				else
				{
					campaignData.playerObjectEngaged = playerMobile.IndexOf(mobileMapObject2.gameObjectAnimator);
					campaignData.otherObjectEngaged = otherMobile.IndexOf(mobileMapObject.gameObjectAnimator);
				}
				if (campaignData.playerObjectEngaged >= playerMobile.Count || campaignData.otherObjectEngaged >= otherMobile.Count || campaignData.playerObjectEngaged == -1 || campaignData.otherObjectEngaged == -1)
				{
					continue;
				}
				if (Config.instance.subAutoAmbushPosition[0] && IsMobileSubOnly(playerMobile[campaignData.playerObjectEngaged].mobileMapObject))
				{
					campaignData.subAmbush = true;
				}
				else if (Config.instance.subAutoAmbushPosition[1] && IsMobileSubOnly(otherMobile[campaignData.otherObjectEngaged].mobileMapObject))
				{
					campaignData.subAmbush = true;
				}
				if (IsMobileSubOnly(playerMobile[campaignData.playerObjectEngaged].mobileMapObject) && otherMobile[campaignData.otherObjectEngaged].mobileMapObject.type == 1 && otherMobile[campaignData.otherObjectEngaged].mobileMapObject.unitPrefabs.Count != 1)
				{
					continue;
				}
				if (otherMobile[campaignData.otherObjectEngaged].mobileMapObject.ignoreByPlayer)
				{
					if (EngagementManager.instance.unitDataDictionary[otherMobile[campaignData.otherObjectEngaged].mobileMapObject.unitPrefabs[0]].unitSubtype == UnitSubType.Scout || (otherMobile[campaignData.otherObjectEngaged].mobileMapObject.unitPrefabs.Count == 1 && EngagementManager.instance.unitDataDictionary[otherMobile[campaignData.otherObjectEngaged].mobileMapObject.unitPrefabs[0]].unitType == UnitType.Air))
					{
						EngagementPanel.instance.ClearEncounterData();
						continue;
					}
					if (IsMobileSubOnly(playerMobile[campaignData.playerObjectEngaged].mobileMapObject))
					{
						EngagementPanel.instance.ClearEncounterData();
						continue;
					}
					if (playerMobile[campaignData.playerObjectEngaged].mobileMapObject.type == 1 && otherMobile[campaignData.otherObjectEngaged].mobileMapObject.type == 0)
					{
						EngagementPanel.instance.ClearEncounterData();
						continue;
					}
				}
				campaignData.canIgnore = false;
				if (playerMobile[campaignData.playerObjectEngaged].mobileMapObject.type == 1 && otherMobile[campaignData.otherObjectEngaged].mobileMapObject.type == 0)
				{
					campaignData.canIgnore = true;
				}
				if (EngagementManager.instance.unitDataDictionary[mobileMapObject.unitPrefabs[0]].unitSubtype == UnitSubType.Scout && EngagementManager.instance.unitDataDictionary[mobileMapObject2.unitPrefabs[0]].unitSubtype == UnitSubType.Scout)
				{
					campaignData.canIgnore = true;
				}
				campaignData.canAutoResolve = ModAirResolve.CanResolve(mobileMapObject, mobileMapObject2) | ModAllied.CanResolve(mobileMapObject, mobileMapObject2);
				if ((EngagementManager.instance.unitDataDictionary[mobileMapObject.unitPrefabs[0]].unitSubtype == UnitSubType.Scout && EngagementManager.instance.unitDataDictionary[mobileMapObject2.unitPrefabs[0]].unitSubtype == UnitSubType.Fighter) || (EngagementManager.instance.unitDataDictionary[mobileMapObject.unitPrefabs[0]].unitSubtype == UnitSubType.Fighter && EngagementManager.instance.unitDataDictionary[mobileMapObject2.unitPrefabs[0]].unitSubtype == UnitSubType.Scout))
				{
					campaignData.canAutoResolve = true;
				}
				if (IsMobileSubOnly(playerMobile[campaignData.playerObjectEngaged].mobileMapObject))
				{
					campaignData.canIgnore = true;
				}
				if ((playerMobile[campaignData.playerObjectEngaged].mobileMapObject.type == 0 && EngagementManager.instance.unitDataDictionary[otherMobile[campaignData.otherObjectEngaged].mobileMapObject.unitPrefabs[0]].unitSubtype == UnitSubType.Scout) || (EngagementManager.instance.unitDataDictionary[otherMobile[campaignData.otherObjectEngaged].mobileMapObject.unitPrefabs[0]].unitType == UnitType.Air && otherMobile[campaignData.otherObjectEngaged].mobileMapObject.unitPrefabs.Count == 1))
				{
					if (Config.instance.alwaysIgnoreScouts)
					{
						EngagementPanel.instance.ClearEncounterData();
						continue;
					}
					campaignData.canIgnore = true;
				}
				if (campaignData.playerObjectEngaged != -1)
				{
					EngagementPanel.instance.OpenEngagementPanel(campaignData.playerObjectEngaged);
				}
				return true;
			}
		}
		return false;
	}

	public bool IsMobileAircraftScoutOnly(MobileMapObject mmo)
	{
		if (Config.instance.alwaysIgnoreScouts && mmo.unitPrefabs.Count == 1)
		{
			return mmo.type == 1;
		}
		return false;
	}

	public bool IsMobileSubOnly(MobileMapObject mmo)
	{
		for (int i = 0; i < mmo.unitPrefabs.Count; i++)
		{
			if (EngagementManager.instance.unitDataDictionary[mmo.unitPrefabs[i]].unitSubtype != 0)
			{
				return false;
			}
		}
		return true;
	}

	public void LoadCampaign(string filepath)
	{
		if (filepath == string.Empty)
		{
			return;
		}
		string[] array = File.ReadAllLines(filepath);
		_ = campaignData.gameVersion != MenuSystem.instance.versionString;
		CampaignInterface.instance.dateTime = new DateTime(campaignData.currentDateTime[5], campaignData.currentDateTime[4], campaignData.currentDateTime[3], campaignData.currentDateTime[2], campaignData.currentDateTime[1], campaignData.currentDateTime[0]);
		for (int i = 0; i < landMapLocations.Count; i++)
		{
			for (int j = 0; j < campaignData.mapLocationSaveData.Count; j++)
			{
				if (campaignData.mapLocationSaveData[j].Split('"')[3] == landMapLocations[i].locationID)
				{
					JsonUtility.FromJsonOverwrite(campaignData.mapLocationSaveData[j], landMapLocations[i]);
				}
			}
		}
		for (int k = 0; k < campaignData.playerMobileObjectSaveData.Count; k++)
		{
			CampaignInterface.instance.ClearNewGroupData(CampaignInterface.instance.newTaskForce);
			JsonUtility.FromJsonOverwrite(campaignData.playerMobileObjectSaveData[k], CampaignInterface.instance.newTaskForce);
			GameObjectAnimator gameObjectAnimator = CampaignInterface.instance.CreateNewMobileObject((CampaignObjectType)Enum.Parse(typeof(CampaignObjectType), CampaignInterface.instance.newTaskForce.type.ToString()), CampaignInterface.instance.newTaskForce, CampaignInterface.instance.newTaskForce.currentMap2DLocation, isLoading: true, null, string.Empty);
			JsonUtility.FromJsonOverwrite(campaignData.playerMobileObjectSaveData[k], gameObjectAnimator.mobileMapObject);
			gameObjectAnimator.mobileMapObject.type = SetMMOType(gameObjectAnimator.mobileMapObject);
			if (gameObjectAnimator.mobileMapObject.type == 0 && CampaignAI.instance.seaMissions[gameObjectAnimator.mobileMapObject.missionIndex].patrolForHours > 0 && gameObjectAnimator.mobileMapObject.patrolForHours <= 0)
			{
				gameObjectAnimator.mobileMapObject.patrolForHours = CampaignAI.instance.seaMissions[gameObjectAnimator.mobileMapObject.missionIndex].patrolForHours;
				gameObjectAnimator.mobileMapObject.patrolZones = GetValidNeighbours(gameObjectAnimator.mobileMapObject.lastWaypoint, includeStart: true, gameObjectAnimator.mobileMapObject.prohibitedWaypoints);
			}
			if (gameObjectAnimator.isMoving)
			{
				List<Vector2> list = new List<Vector2>();
				list.Add(gameObjectAnimator.mobileMapObject.endingMap2DLocation);
				for (int l = 0; l < gameObjectAnimator.mobileMapObject.mapWaypoints.Count; l++)
				{
					list.Add(gameObjectAnimator.mobileMapObject.mapWaypoints[l]);
				}
				CampaignInterface.instance.MoveMobileToDestination(gameObjectAnimator, gameObjectAnimator.mobileMapObject.destination, goDirect: true, addWay: false);
				gameObjectAnimator.mobileMapObject.mapWaypoints.Clear();
				gameObjectAnimator.mobileMapObject.mapWaypoints = list;
			}
		}
		for (int m = 0; m < campaignData.enemyMobileObjectSaveData.Count; m++)
		{
			CampaignInterface.instance.ClearNewGroupData(CampaignInterface.instance.newTaskForce);
			JsonUtility.FromJsonOverwrite(campaignData.enemyMobileObjectSaveData[m], CampaignInterface.instance.newTaskForce);
			GameObjectAnimator gameObjectAnimator2 = CampaignInterface.instance.CreateNewMobileObject((CampaignObjectType)Enum.Parse(typeof(CampaignObjectType), CampaignInterface.instance.newTaskForce.type.ToString()), CampaignInterface.instance.newTaskForce, CampaignInterface.instance.newTaskForce.currentMap2DLocation, isLoading: true, null, string.Empty);
			JsonUtility.FromJsonOverwrite(campaignData.enemyMobileObjectSaveData[m], gameObjectAnimator2.mobileMapObject);
			gameObjectAnimator2.mobileMapObject.type = SetMMOType(gameObjectAnimator2.mobileMapObject);
			if (gameObjectAnimator2.mobileMapObject.currentFaction != 0 && CampaignAI.instance.seaMissions[gameObjectAnimator2.mobileMapObject.missionIndex].missionType == MissionType.Deprecated)
			{
				CampaignAI.instance.ApplyNavDataToMobile(gameObjectAnimator2, CampaignAI.instance.seaMissions[CampaignAI.instance.GetMissionByType(MissionType.Home)], string.Empty);
				GetNextNavPosition(gameObjectAnimator2);
			}
		}
		for (int n = 0; n < campaignData.weatherCellData.Count; n++)
		{
			JsonUtility.FromJsonOverwrite(campaignData.weatherCellData[n], WeatherManager.instance.weatherCells[n]);
		}
		WeatherManager.instance.SetWeatherIcons(display: true);
		for (int num = 0; num < otherMobile.Count; num++)
		{
			if (otherMobile[num].mobileMapObject.spottedTimer > 0f)
			{
				spottedContacts.Add(otherMobile[num].mobileMapObject);
				CampaignInterface.instance.SetContactColor(otherMobile[num].mobileMapObject);
				if (Config.instance.vagueCampaignIntel && otherMobile[num].mobileMapObject.type == 0)
				{
					if (otherMobile[num].mobileMapObject.intelData == null || otherMobile[num].mobileMapObject.intelData.Count < 1)
					{
						GetIntelOnSeaGroup(otherMobile[num].mobileMapObject);
					}
					else
					{
						SetSeaGroupSpriteFromIntel(otherMobile[num].mobileMapObject);
					}
				}
			}
			else
			{
				otherMobile[num].GetComponentInChildren<SpriteRenderer>().color = CampaignInterface.instance.noContactColor;
			}
		}
		for (int num2 = 0; num2 < playerMobile.Count; num2++)
		{
			if (playerMobile[num2].mobileMapObject.spottedTimer > 0f)
			{
				spottedContacts.Add(playerMobile[num2].mobileMapObject);
			}
			if (playerMobile[num2].mobileMapObject.escortingGOAIndex != -1 && playerMobile[num2].mobileMapObject.type == 1)
			{
				playerMobile[num2].mobileMapObject.escortingThisGOA = playerMobile[playerMobile[num2].mobileMapObject.escortingGOAIndex];
			}
			else
			{
				playerMobile[num2].mobileMapObject.escortingGOAIndex = -1;
			}
		}
		if (array.Length > 1 && campaignData.currentlyInCombatIndex != -1)
		{
			CombatInterface.instance.tabs[4].transform.SetParent(CombatInterface.instance.transform);
			wasUnitRefOpen = CombatInterface.instance.tabs[4].activeSelf;
			CombatInterface.instance.tabs[4].gameObject.SetActive(value: false);
			TerrainTiles.instance.xyPixelCoord = campaignData.lastEncounterPixelCoords;
			GenerateCampaignEngagement(playerMobile[campaignData.currentlyInCombatIndex], null, filepath);
		}
		for (int num3 = 0; num3 < landMapLocations.Count; num3++)
		{
			if (landMapLocations[num3].troops[0] > 0 && landMapLocations[num3].troops[1] > 0)
			{
				CampaignInterface.instance.SetFireToLocation(landMapLocations[num3], isOn: true);
			}
		}
		if (OptionsManager.instance.campaignDifficultyMultipliers == null || OptionsManager.instance.campaignDifficultyMultipliers.Length < 30)
		{
			string[] array2 = AssetLoader.instance.ImportText("campaign/" + campaignData.campaignID + "/setup.txt");
			CampDifficultyArrays campDifficultyArrays = new CampDifficultyArrays();
			JsonUtility.FromJsonOverwrite(array2[0], campDifficultyArrays);
			campaignData.campDifficultyPreset0 = campDifficultyArrays.campDifficultyPreset0;
			campaignData.campDifficultyPreset1 = campDifficultyArrays.campDifficultyPreset1;
			campaignData.campDifficultyPreset2 = campDifficultyArrays.campDifficultyPreset2;
			campaignData.campDifficultyPreset3 = campDifficultyArrays.campDifficultyPreset3;
			campaignData.campDifficultyPreset4 = campDifficultyArrays.campDifficultyPreset4;
			campaignData.SetCampaignDifficulty();
		}
		if (campaignData.campaignEnded)
		{
			SummaryStats.instance.OpenCampaignSummary();
		}
	}

	public void SetMapTexture(bool clear = false)
	{
		if (clear)
		{
			mapMaterial.SetTexture("_TexTL", null);
			mapMaterial.SetTexture("_TexTR", null);
			mapMaterial.SetTexture("_TexBL", null);
			mapMaterial.SetTexture("_TexBR", null);
			return;
		}
		for (int i = 0; i < mapSharderQuads.Length; i++)
		{
			Texture2D texture2D = AssetLoader.instance.ImportTexture2D(campaignData.mapImagePaths[i], resourcesOnly: false, mipMap: true);
			texture2D.wrapMode = TextureWrapMode.Clamp;
			texture2D.filterMode = FilterMode.Bilinear;
			mapMaterial.SetTexture(mapSharderQuads[i], texture2D);
		}
	}

	public void StartCampaign(string filepath = "")
	{
		base.gameObject.SetActive(value: true);
		if (base.gameObject.GetComponent<ModTFOverlay>() == null)
		{
			base.gameObject.AddComponent<ModTFOverlay>();
		}
		ModGlobe.Ensure(this);
		StartCoroutine(SetupCampaign(filepath));
	}

	private IEnumerator SetupCampaign(string filepath = "")
	{
		LoadScreen.instance.SetLoadProgress(0.5f);
		yield return null;
		campaignData.commandPointSpent = new int[2];
		campaignData.campaignID = MenuSystem.instance.selectionFileNames[MenuSystem.instance.currentSelectedIndex];
		if (filepath != string.Empty)
		{
			string[] array = File.ReadAllLines(filepath);
			JsonUtility.FromJsonOverwrite(array[0], campaignData);
			if (campaignData.gameVersion != MenuSystem.instance.versionString && !Tutorial.instance.gameObject.activeSelf)
			{
				string text = "_" + campaignData.gameVersion.Replace(".", string.Empty);
				string text2 = Path.Combine(Application.persistentDataPath, Path.Combine(Library.instance.backupPath, MenuSystem.instance.selectionFileNames[MenuSystem.instance.currentSelectedIndex]));
				text2 = text2.Replace(".txt", text + ".txt");
				if (!File.Exists(text2))
				{
					File.WriteAllLines(text2, array);
				}
			}
		}
		base.gameObject.SetActive(value: true);
		CombatInterface.instance.transform.parent.gameObject.SetActive(value: false);
		mapRenderer.sharedMaterial = mapMaterial;
		landMapLocations = new List<MapLocation>();
		campaignData.InitialiseCampaignData(filepath);
		if (campaignData.customWaterMask)
		{
			Texture2D waterMask = AssetLoader.instance.ImportTexture2D(campaignData.mapWaterMask);
			CampaignInterface.instance.map.waterMask = waterMask;
		}
		CampaignInterface.instance.InitialiseCampaignInterface();
		CampaignInterface.instance.map.wrapHorizontally = false;
		CampaignInterface.instance.map.zoomMinDistance = campaignData.mapZoomLevels.x;
		CampaignInterface.instance.map.zoomMaxDistance = campaignData.mapZoomLevels.y;
		CampaignInterface.instance.CentreMap();
		SetMapTexture();
		CreateSeaWaypoints("campaign/" + campaignData.campaignID + "/seaWaypoints.txt");
		CreateLandLocations("campaign/" + campaignData.campaignID + "/mapLandLocations.txt");
		CreateObjectives("language/campaign/" + campaignData.campaignID + "/objectives.txt");
		UnitReference.instance.SwitchUnitReferenceTransform(UnitReference.instance.unitRefPositions[1], isOn: false);
		CampaignInterface.instance.TogglePauseCampaign();
		if (filepath == string.Empty)
		{
			campaignData.enemyUsesSubs = CampaignSettings.instance.settingsToggles[0].isOn;
			campaignData.enemyUsesCarriers = CampaignSettings.instance.settingsToggles[1].isOn;
			campaignData.refundPostRepair = CampaignSettings.instance.settingsToggles[2].isOn;
			campaignData.carrierWingsRequireRelease = CampaignSettings.instance.settingsToggles[3].isOn;
			campaignData.usePortResourcesToResupplyAmmo = CampaignSettings.instance.settingsToggles[4].isOn;
			campaignData.usePortResourcesToResupplyShipAircraft = CampaignSettings.instance.settingsToggles[5].isOn;
			campaignData.usePortResourcesToResupplyAirfields = CampaignSettings.instance.settingsToggles[6].isOn;
			campaignData.replenishLimitAircraft = CampaignSettings.instance.settingsToggles[7].isOn;
			campaignData.supplyShipsAllowed = CampaignSettings.instance.settingsToggles[8].isOn;
			campaignData.enableStrategicBombing = CampaignSettings.instance.settingsToggles[9].isOn;
		}
		CampaignAI.instance.InitialiseAI();
		WeatherManager.instance.CreateWeatherCells();
		campaignData.playerObjectEngaged = -1;
		if (filepath != string.Empty)
		{
			LoadCampaign(filepath);
			for (int i = 0; i < landMapLocations.Count; i++)
			{
				landMapLocations[i].GetComponentInChildren<SpriteRenderer>().color = CampaignInterface.instance.factionColours[landMapLocations[i].currentFaction];
				if (landMapLocations[i].troops[0] > 0 && landMapLocations[i].troops[1] > 0)
				{
					CampaignInterface.instance.SetFireToLocation(landMapLocations[i], isOn: true);
				}
			}
			for (int j = 0; j < campaignData.mapMarkersSaveData.Count; j++)
			{
				MapMarkerData mapMarkerData = new MapMarkerData();
				JsonUtility.FromJsonOverwrite(campaignData.mapMarkersSaveData[j], mapMarkerData);
				CampaignInterface.instance.CreateMarker(mapMarkerData.locationPosition, mapMarkerData.locationName, mapMarkerData.locationDescription, mapMarkerData.portLevel);
			}
			CampaignInterface.instance.mapDrawField.transform.parent.parent.gameObject.SetActive(value: false);
		}
		else
		{
			campaignData.commandPoints[0] = Mathf.FloorToInt((float)campaignData.commandPoints[0] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.StartPoints, Faction.Player));
			campaignData.releaseRefund *= Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.ReleaseRefund, Faction.Player);
			campaignData.enemySunkYield *= Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.SunkPoints, Faction.Player);
			campaignData.baseRepairTime[0] = campaignData.baseRepairTime[0] + (int)Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.RepairDays, Faction.Player);
			campaignData.baseRepairTime[1] = campaignData.baseRepairTime[1] + (int)Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.RepairDays, Faction.Enemy);
			campaignData.repairPerComp[0] = campaignData.repairPerComp[0] + (int)Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.AddRepair, Faction.Player);
			campaignData.repairPerComp[1] = campaignData.repairPerComp[1] + (int)Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.AddRepair, Faction.Enemy);
			for (int k = 0; k < campaignData.replenish0.Count; k++)
			{
				campaignData.replenish0[k] = Mathf.FloorToInt((float)campaignData.replenish0[k] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.ReplenishRate, Faction.Player));
				campaignData.replenish1[k] = Mathf.FloorToInt((float)campaignData.replenish0[k] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.ReplenishRate, Faction.Enemy));
			}
			for (int l = 0; l < campaignData.suppliesPerUpgrade0.Count; l++)
			{
				campaignData.suppliesPerUpgrade0[l] = Mathf.FloorToInt((float)campaignData.suppliesPerUpgrade0[l] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.UpgradeCost, Faction.Player));
				campaignData.suppliesPerUpgrade1[l] = Mathf.FloorToInt((float)campaignData.suppliesPerUpgrade1[l] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.UpgradeCost, Faction.Enemy));
				campaignData.engineeringPerUpgrade0[l] = Mathf.FloorToInt((float)campaignData.engineeringPerUpgrade0[l] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.UpgradeCost, Faction.Player));
				campaignData.engineeringPerUpgrade1[l] = Mathf.FloorToInt((float)campaignData.engineeringPerUpgrade1[l] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.UpgradeCost, Faction.Enemy));
				campaignData.fuelPerUpgrade0[l] = Mathf.FloorToInt((float)campaignData.fuelPerUpgrade0[l] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.UpgradeCost, Faction.Player));
				campaignData.fuelPerUpgrade1[l] = Mathf.FloorToInt((float)campaignData.fuelPerUpgrade1[l] * Utilities.GetCampaignDifficultyModifier(CampaignDifficulty.UpgradeCost, Faction.Enemy));
			}
			campaignData.unavailableShipInstances = new List<int>();
			for (int m = 0; m < campaignData.unavailableShipClasses.Count; m++)
			{
				if (EngagementManager.instance.unitDataDictionary.ContainsKey(campaignData.unavailableShipClasses[m]))
				{
					int num = EngagementManager.instance.unitDataDictionary[campaignData.unavailableShipClasses[m]].instancesInClass.IndexOf(campaignData.unavailableShipInstanceNames[m]);
					if (num != -1)
					{
						campaignData.unavailableShipInstances.Add(num);
						continue;
					}
					campaignData.unavailableShipInstances.Add(-1);
					Debug.LogError("Unavailable Ship Not Found: " + campaignData.unavailableShipClasses[m] + " " + campaignData.unavailableShipInstanceNames[m]);
				}
				else
				{
					campaignData.unavailableShipInstances.Add(-1);
				}
			}
			CampaignInterface.instance.dateTime = new DateTime(campaignData.currentDateTime[5], campaignData.currentDateTime[4], campaignData.currentDateTime[3], campaignData.currentDateTime[2], campaignData.currentDateTime[1], campaignData.currentDateTime[0]);
			campaignData.airOpsFirstHourPaused = campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour];
		}
		CampaignInterface.instance.startDate = new DateTime(campaignData.startDate[5], campaignData.startDate[4], campaignData.startDate[3], campaignData.startDate[2], campaignData.startDate[1], campaignData.startDate[0]);
		CampaignInterface.instance.SetDateTimeText();
		CampaignInterface.instance.SetCommandTotal();
		CampaignInterface.instance.SetLightMeter();
		CampaignInterface.instance.strategicBombButton.SetActive(campaignData.enableStrategicBombing);
		if (Config.instance.showCampaignLocationRadii)
		{
			MapRadii.instance.CreateMapRadii();
			MapRadii.instance.SetMapRadii();
		}
		UnitReference.instance.PopulateUnitReferencePulldowns();
		UnitReference.instance.DisplayUnitReference(EngagementManager.instance.unitSeaOrderList[0]);
		if (filepath == string.Empty)
		{
			CampaignAI.instance.CreateStandingMissions();
			CampaignAI.instance.MakeStrategicDecision();
			CampaignEvent.instance.CreateEvent("start");
		}
		if (campaignData.playerObjectEngaged != -1 && campaignData.playerObjectEngaged < playerMobile.Count)
		{
			EngagementPanel.instance.OpenEngagementPanel(campaignData.playerObjectEngaged);
		}
		MenuSystem.instance.CloseMenuSystem();
		LoadScreen.instance.SetLoadProgress(1f);
	}

	private Vector2 GetPointInRadius(Vector3 posRadius)
	{

	}

	public void GetNextNavPosition(GameObjectAnimator obj)
	{
		obj.Stop();
		if (obj.mobileMapObject.currentWaypoint == obj.mobileMapObject.lastWaypoint)
		{
			if (CampaignAI.instance.seaMissions[obj.mobileMapObject.missionIndex].destIsLocation)
			{
				CampaignAI.instance.OnDestinationReached(obj);
			}
			else if (CampaignAI.instance.seaMissions[obj.mobileMapObject.missionIndex].patrolDestZone)
			{
				Vector2 zero = Vector2.zero;
				if (CampaignAI.instance.seaMissions[obj.mobileMapObject.missionIndex].patrolNeighbours && obj.mobileMapObject.currentWaypoint == obj.mobileMapObject.lastWaypoint)
				{
					obj.mobileMapObject.previousWaypoint = obj.mobileMapObject.currentWaypoint;
					obj.mobileMapObject.currentWaypoint = GetValidNeighbour(obj.mobileMapObject.currentWaypoint, includeStart: true, obj.mobileMapObject.prohibitedWaypoints);
					zero = GetPointInNavZone(obj.mobileMapObject.currentWaypoint);
				}
				else
				{
					zero = GetPointInNavZone(obj.mobileMapObject.currentWaypoint);
				}
				obj.MoveTo(zero, CampaignInterface.instance.GetMoveDuration(obj.currentMap2DLocation, zero, obj.mobileMapObject.speed), DURATION_TYPE.Route);
				return;
			}
		}
		if (obj.mobileMapObject.currentWaypoint == obj.mobileMapObject.waitAtWaypoint && CampaignInterface.instance.dateTime.Hour < obj.mobileMapObject.untilHour)
		{
			Vector2 zero2 = Vector2.zero;
			zero2 = GetPointInNavZone(obj.mobileMapObject.currentWaypoint);
			obj.MoveTo(zero2, CampaignInterface.instance.GetMoveDuration(obj.currentMap2DLocation, zero2, obj.mobileMapObject.speed), DURATION_TYPE.Route);
			return;
		}
		int previousWaypoint = obj.mobileMapObject.previousWaypoint;
		obj.mobileMapObject.previousWaypoint = obj.mobileMapObject.currentWaypoint;
		_ = Vector2.zero;
		if (obj.mobileMapObject.mustUseWaypoints.Count != 0 && obj.mobileMapObject.type == 0)
		{
			obj.mobileMapObject.currentWaypoint = GetNextWaypoint(obj, previousWaypoint);
			obj.MoveTo(seaNavigationPoints[obj.mobileMapObject.currentWaypoint].positionRadius, CampaignInterface.instance.GetMoveDuration(obj.currentMap2DLocation, seaNavigationPoints[obj.mobileMapObject.currentWaypoint].positionRadius, obj.mobileMapObject.speed), DURATION_TYPE.Route);
		}
	}

	private Vector2 GetPointInNavZone(int zoneIndex)
	{
		Vector3 positionRadius = seaNavigationPoints[zoneIndex].positionRadius;
		Vector2 vector = UnityEngine.Random.insideUnitCircle * seaNavigationPoints[zoneIndex].positionRadius.z;
		vector.x /= 2f;
		return new Vector2(positionRadius.x, positionRadius.y) + vector;
	}

	private int GetNextWaypoint(GameObjectAnimator obj, int previousWaypoint = -1)
	{
		MobileMapObject mobileMapObject = obj.mobileMapObject;
		mobileMapObject.aiWaypointIndexesToCheck.Clear();
		int num = CheckMustUse(obj);
		if (num != -1)
		{
			if (mobileMapObject.mustUseWaypoints.Count > 1)
			{
				mobileMapObject.mustUseWaypoints.RemoveAt(0);
			}
			return num;
		}
		float num2 = mobileMapObject.destinationPosition.x - obj.currentMap2DLocation.x;
		float num3 = mobileMapObject.destinationPosition.y - obj.currentMap2DLocation.y;
		bool[] array = new bool[4];
		if (num2 >= 0f)
		{
			GetAdditionalPossibleWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].eastWP, previousWaypoint);
			array[1] = true;
		}
		else
		{
			GetAdditionalPossibleWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].westWP, previousWaypoint);
			array[3] = true;
		}
		if (num3 >= 0f)
		{
			GetAdditionalPossibleWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].northWP, previousWaypoint);
			array[0] = true;
		}
		else
		{
			GetAdditionalPossibleWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].southWP, previousWaypoint);
			array[2] = true;
		}
		if (mobileMapObject.aiWaypointIndexesToCheck.Count == 0)
		{
			if (!array[0])
			{
				GetAdditionalPossibleWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].northWP);
			}
			if (!array[1])
			{
				GetAdditionalPossibleWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].eastWP);
			}
			if (!array[2])
			{
				GetAdditionalPossibleWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].southWP);
			}
			if (!array[3])
			{
				GetAdditionalPossibleWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].westWP);
			}
			if (mobileMapObject.aiWaypointIndexesToCheck.Count > 1 && mobileMapObject.aiWaypointIndexesToCheck.Contains(mobileMapObject.previousWaypoint))
			{
				mobileMapObject.aiWaypointIndexesToCheck.Remove(mobileMapObject.previousWaypoint);
			}
		}
		string text = string.Empty;
		for (int i = 0; i < mobileMapObject.aiWaypointIndexesToCheck.Count; i++)
		{
			text = text + seaNavigationPoints[mobileMapObject.aiWaypointIndexesToCheck[i]].waypointName + ", ";
		}
		if (mobileMapObject.aiWaypointIndexesToCheck.Count == 0)
		{
			GetAllWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].northWP);
			GetAllWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].southWP);
			GetAllWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].eastWP);
			GetAllWaypoints(mobileMapObject, seaNavigationPoints[mobileMapObject.currentWaypoint].westWP);
			return mobileMapObject.aiWaypointIndexesToCheck[UnityEngine.Random.Range(0, mobileMapObject.aiWaypointIndexesToCheck.Count)];
		}
		if (mobileMapObject.mustUseWaypoints.Count > 0 && mobileMapObject.aiWaypointIndexesToCheck.Contains(mobileMapObject.mustUseWaypoints[0]))
		{
			int result = mobileMapObject.mustUseWaypoints[0];
			if (mobileMapObject.mustUseWaypoints.Count > 1)
			{
				mobileMapObject.mustUseWaypoints.RemoveAt(0);
			}
			return result;
		}
		if (CampaignAI.instance.seaMissions[mobileMapObject.missionIndex].directPath && mobileMapObject.aiWaypointIndexesToCheck.Count > 1)
		{
			float num4 = float.PositiveInfinity;
			int index = 0;
			for (int j = 0; j < mobileMapObject.aiWaypointIndexesToCheck.Count; j++)
			{
				float num5 = Vector3.Distance(seaNavigationPoints[mobileMapObject.aiWaypointIndexesToCheck[j]].positionRadius, mobileMapObject.destinationPosition);
				if (num5 < num4)
				{
					num4 = num5;
					index = j;
				}
			}
			return mobileMapObject.aiWaypointIndexesToCheck[index];
		}
		return mobileMapObject.aiWaypointIndexesToCheck[UnityEngine.Random.Range(0, mobileMapObject.aiWaypointIndexesToCheck.Count)];
	}

	private void GetAdditionalPossibleWaypoints(MobileMapObject mmo, List<int> waypointsToCheck, int disallow = -1)
	{
		for (int i = 0; i < waypointsToCheck.Count; i++)
		{
			if (waypointsToCheck[i] != mmo.previousWaypoint && !mmo.aiWaypointIndexesToCheck.Contains(waypointsToCheck[i]) && !mmo.prohibitedWaypoints.Contains(waypointsToCheck[i]) && mmo.previousWaypoint != waypointsToCheck[i] && (!seaNavigationPoints[waypointsToCheck[i]].isLocation || mmo.mustUseWaypoints.Contains(waypointsToCheck[i])) && disallow != -1)
			{
				mmo.aiWaypointIndexesToCheck.Add(waypointsToCheck[i]);
			}
		}
	}

	private void GetAllWaypoints(MobileMapObject mmo, List<int> waypointsToCheck)
	{
		for (int i = 0; i < waypointsToCheck.Count; i++)
		{
			mmo.aiWaypointIndexesToCheck.Add(waypointsToCheck[i]);
		}
	}

	public void ApplyUpgradeLocation(MapLocation ml, int baseType, bool player)
	{
		int index = ((baseType != 0) ? ml.airfieldLevel : ml.portLevel);
		if (player)
		{
			ml.supplies[0] -= campaignData.suppliesPerUpgrade0[index];
			ml.fuel[0] -= campaignData.fuelPerUpgrade0[index];
			ml.engineering[0] -= campaignData.engineeringPerUpgrade0[index];
			if (baseType == 0)
			{
				ml.portLevel++;
			}
			else
			{
				ml.airfieldLevel++;
				CampaignAI.instance.CheckUnsafeLocations(ml);
			}
		}
		else
		{
			ml.supplies[1] -= campaignData.suppliesPerUpgrade1[index];
			ml.fuel[1] -= campaignData.fuelPerUpgrade1[index];
			ml.engineering[1] -= campaignData.engineeringPerUpgrade1[index];
			if (baseType == 0)
			{
				ml.portLevel++;
			}
			else
			{
				ml.airfieldLevel++;
			}
		}
		if (Config.instance.showCampaignLocationRadii)
		{
			MapRadii.instance.SetMapRadii();
		}
		if (CampaignInterface.instance.currentMapObject == ml.GetComponent<GameObjectAnimator>())
		{
			CampaignInterface.instance.SelectMapObject(CampaignInterface.instance.currentMapObject);
		}
	}

	private void CreateSeaWaypoints(string path)
	{
		seaNavigationPoints = new List<SeaNavigationWaypoint>();
		seaNavigationIndexes = new Dictionary<string, int>();
		string[] array = AssetLoader.instance.ImportText(path);
		for (int i = 0; i < array.Length; i++)
		{
			SeaNavigationWaypoint seaNavigationWaypoint = new SeaNavigationWaypoint();
			JsonUtility.FromJsonOverwrite(array[i], seaNavigationWaypoint);
			seaNavigationWaypoint.navWaypointID = i;
			seaNavigationWaypoint.positionRadius = new Vector3(float.Parse(seaNavigationWaypoint.pos[0]), float.Parse(seaNavigationWaypoint.pos[1]), float.Parse(seaNavigationWaypoint.pos[2]));
			seaNavigationIndexes.Add(seaNavigationWaypoint.waypointName, i);
			seaNavigationPoints.Add(seaNavigationWaypoint);
		}
		for (int j = 0; j < seaNavigationPoints.Count; j++)
		{
			seaNavigationPoints[j].northWP = GetListIndexes(seaNavigationPoints[j].sNorthWP);
			seaNavigationPoints[j].southWP = GetListIndexes(seaNavigationPoints[j].sSouthWP);
			seaNavigationPoints[j].eastWP = GetListIndexes(seaNavigationPoints[j].sEastWP);
			seaNavigationPoints[j].westWP = GetListIndexes(seaNavigationPoints[j].sWestWP);
		}
	}

	private List<int> GetListIndexes(List<string> inputList)
	{
		List<int> list = new List<int>();
		for (int i = 0; i < inputList.Count; i++)
		{
			list.Add(seaNavigationIndexes[inputList[i]]);
		}
		return list;
	}

			if (listString[i] == prefabName && listInt[i] == instance)
			{
				return i;
			}
		}
		return -1;
	}

	private List<GameObjectAnimator> SortMobilesByDistance(List<GameObjectAnimator> goas, Vector2 pos)
	{
		return new List<GameObjectAnimator>(goas).OrderBy((GameObjectAnimator x) => (pos - x.currentMap2DLocation).sqrMagnitude).ToList();
	}

	private void GetValidNeighbouringMobiles(List<GameObjectAnimator> nearbyMobiles, GameObjectAnimator battleMobile)
	{
		bool flag = nearbyMobiles == otherMobile;
		_ = "[TFJOIN] ---- pass, list size " + nearbyMobiles.Count + ", secondPass " + flag.ToString() + ", in battle so far " + goasInBattle.Count;
		for (int i = 0; i < nearbyMobiles.Count; i++)
		{
			GameObjectAnimator gameObjectAnimator = nearbyMobiles[i];
			if (gameObjectAnimator == null || gameObjectAnimator.mobileMapObject == null)
			{
				continue;
			}
			MobileMapObject mobileMapObject = gameObjectAnimator.mobileMapObject;
			float distanceInMetresBetweenMapObjects = GetDistanceInMetresBetweenMapObjects(battleMobile.currentMap2DLocation, gameObjectAnimator.currentMap2DLocation);
			string text = "[TFJOIN] " + mobileMapObject.mobileName + " type " + mobileMapObject.type + " side " + mobileMapObject.currentFaction + " ships " + mobileMapObject.unitPrefabs.Count + " dist " + distanceInMetresBetweenMapObjects.ToString("F0") + " timer " + mobileMapObject.postEngagementTimer.ToString("F0");
			if (mobileMapObject.type == 0 && flag)
			{
				_ = text + " -> skip (second pass)";
				continue;
			}
			if (distanceInMetresBetweenMapObjects >= ModGeo.JoinRadiusMetres)
			{
				_ = text + " -> skip (too far)";
				continue;
			}
			if (goasInBattle.Contains(gameObjectAnimator))
			{
				_ = text + " -> skip (already in battle)";
				continue;
			}
			int currentFaction = mobileMapObject.currentFaction;
			if (mobileMapObject.type == 0)
			{
				int num = 0;
				for (int j = 0; j < goasInBattle.Count; j++)
				{
					GameObjectAnimator gameObjectAnimator2 = goasInBattle[j];
					if (gameObjectAnimator2 != null && gameObjectAnimator2.mobileMapObject != null && gameObjectAnimator2.mobileMapObject.type == 0 && gameObjectAnimator2.mobileMapObject.currentFaction == currentFaction)
					{
						num += gameObjectAnimator2.mobileMapObject.unitPrefabs.Count;
					}
				}
				if (num + mobileMapObject.unitPrefabs.Count > 40)
				{
					_ = text + " -> skip (ship limit)";
					continue;
				}
			}
			else if (airNumbers[currentFaction] + mobileMapObject.unitPrefabs.Count > Config.instance.maxAirPerSide)
			{
				_ = text + " -> skip (air limit)";
				continue;
			}
			Vector2 vector = ModGeo.Battle((gameObjectAnimator.currentMap2DLocation - battleMobile.currentMap2DLocation) * campaignData.mapWidthMetres * 0.1f, battleMobile);
			EngagementManager.instance.engagementData.unitGroups.Add(GenerateUnitGroupData(gameObjectAnimator, new Vector3(vector.x, 0f, vector.y / 2f), GetHeadingFromObj(gameObjectAnimator), mobileMapObject.speed / 0.514444f, mobileMapObject.currentFaction));
			if (mobileMapObject.type == 0)
			{
				seaPresent[currentFaction] = true;
			}
			else
			{
				airNumbers[currentFaction] += mobileMapObject.unitPrefabs.Count;
			}
			goasInBattle.Add(gameObjectAnimator);
			_ = text + " -> ADDED";
		}
	}

	private int[] GetXYPixelCoord(GameObjectAnimator obj)
	{
		int[] obj2 = new int[2]
		{
			campaignData.mapOriginXY[0],
			campaignData.mapOriginXY[1]
		};
		float num = campaignData.mapPixelWidth;
		int num2 = Mathf.FloorToInt((obj.currentMap2DLocation.x + 0.5f) * num);
		int num3 = Mathf.FloorToInt((obj.currentMap2DLocation.y + 0.5f) * (num * 0.5f));
		obj2[0] += num2;
		obj2[1] += num3;
		return ModGeo.WrapPixel(obj2);
	}

	private string GenerateUnitGroupData(GameObjectAnimator obj, Vector3 position, float heading, float speed, int faction)
	{
		if (faction == 0)
		{
			EngagementManager.instance.engagementData.campaignMobileObjectIndexes0.Add(playerMobile.IndexOf(obj));
		}
		else
		{
			EngagementManager.instance.engagementData.campaignMobileObjectIndexes1.Add(otherMobile.IndexOf(obj));
		}
		UnitGroupData unitGroupData = new UnitGroupData();
		unitGroupData.startPosition = position;
		if (obj.type == 0)
		{
			unitGroupData.unitType = UnitType.Sea;
			unitGroupData.formationSea = (FormationSea)Enum.Parse(typeof(FormationSea), obj.mobileMapObject.formation.ToString());
			unitGroupData.spacing = (float)Config.instance.formationSeaSpacings[obj.mobileMapObject.formationSpacing] * 0.1f;
		}
		else
		{
			unitGroupData.startPosition.y = 100f;
			unitGroupData.unitType = UnitType.Air;
			unitGroupData.formationAir = FormationAir.Vic;
			unitGroupData.spacing = 5f;
			unitGroupData.payloads = obj.mobileMapObject.payload;
		}
		unitGroupData.heading = heading;
		unitGroupData.speed = speed;
		unitGroupData.faction = (Faction)faction;
		unitGroupData.unitPrefabs = obj.mobileMapObject.unitPrefabs;
		unitGroupData.unitNames = obj.mobileMapObject.unitDisplayNames;
		unitGroupData.cellVariables = obj.mobileMapObject.cellVariables;
		unitGroupData.remainingAmmunition = obj.mobileMapObject.ammunitionRemaining;
		unitGroupData.previousBombards = obj.mobileMapObject.bombardments;
		unitGroupData.paint = obj.mobileMapObject.paint;
		unitGroupData.speedModifier = obj.mobileMapObject.speedModifier;
		unitGroupData.vulnerable = GetVulnerableShips(obj.mobileMapObject);
		return JsonUtility.ToJson(unitGroupData);
	}

	private List<int> GetVulnerableShips(MobileMapObject mmo)
	{
		List<int> list = new List<int>();
		for (int i = 0; i < mmo.cargo.Count; i++)
		{
			List<int> cargoOnBoard = CargoInterface.instance.GetCargoOnBoard(mmo.cargo[i]);
			list.Add(CargoInterface.instance.GetCurrentCargoIndex(cargoOnBoard, string.Empty));
			if (Utilities.IsCarrier(mmo.unitPrefabs[i]) && HoursShipLaunchAir(mmo, i) > 0f)
			{
				list[i] = 4;
			}
		}
		return list;
	}

	private float GetHeadingFromObj(GameObjectAnimator obj)
	{
		float result = UnityEngine.Random.value * 360f;
		if (obj.isMoving)
		{
			Vector2 vector = obj.destination - obj.currentMap2DLocation;
			result = Mathf.Atan2(vector.x, vector.y) * 57.29578f + UnityEngine.Random.Range(-25f, 25f);
		}
		return result;
	}

	private float SetSeaCooldown(int faction)
	{
		List<Unit> list = EngagementManager.instance.playerUnits;
		if (faction == 0)
		{
			list = EngagementManager.instance.otherUnits;
		}
		bool flag = false;
		for (int i = 0; i < list.Count; i++)
		{
			if (list[i].unitData.unitType == UnitType.Sea && (!list[i].isDestroyed || list[i].isParked))
			{
				flag = true;
				break;
			}
		}
		if (flag)
		{
			return campaignData.postEngageSeaCooldown;
		}
		return campaignData.postEngageAirCooldown;
	}

	private void ProcessMobileObjectPostCombat(MobileMapObject mmo)
	{
		string item = string.Empty;
		if (mmo.type == 1)
		{
			int num = 0;
			mmo.postEngagementTimer = campaignData.postEngageAirCooldown;
			for (int num2 = mmo.unitPrefabs.Count - 1; num2 > -1; num2--)
			{
				if (mmo.cellVariables[num2] != string.Empty)
				{
					item = mmo.unitPrefabs[0];
					mmo.unitPrefabs.RemoveAt(num2);
					mmo.ammunitionRemaining.RemoveAt(num2);
					if (mmo.currentFaction == 0)
					{
						campaignData.playerAircraftDowned++;
					}
					else
					{
						campaignData.otherAircraftDowned++;
					}
					num++;
				}
			}
			mmo.cellVariables.Clear();
			if (mmo.currentFaction != 0)
			{
				int num3 = spottedContacts.IndexOf(mmo);
				if (num3 != -1)
				{
					spottedContacts[num3].intelData.Clear();
					spottedContacts.RemoveAt(num3);
				}
				mmo.spottedTimer = 0f;
				CampaignInterface.instance.SetContactColor(mmo, forceZero: true);
			}
			bool flag = false;
			if (mmo.unitPrefabs.Count == 0)
			{
				mmo.unitPrefabs.Add(item);
				flag = true;
			}
			if (num > 0)
			{
				SetAircraftReturnedToLocation(mmo, num, destroyed: true);
			}
			if (flag)
			{
				if (CampaignInterface.instance.currentMapObject == mmo.gameObjectAnimator)
				{
					CampaignInterface.instance.currentMapObject = null;
				}
				CampaignInterface.instance.DestroyMapMobileObject(mmo.gameObjectAnimator);
			}
			return;
		}
		mmo.postEngagementTimer = SetSeaCooldown(mmo.currentFaction);
		bool flag2 = IsMobileSubOnly(mmo);
		if (flag2)
		{
			mmo.postEngagementTimer = campaignData.postEngageSubCooldown;
		}
		if (mmo.cellVariables.Count > 0)
		{
			MobileMapObject.AwardVeterancy(mmo);
			if (mmo.currentFaction != 0)
			{
				int num4 = spottedContacts.IndexOf(mmo);
				if (num4 != -1)
				{
					spottedContacts.RemoveAt(num4);
					mmo.intelData.Clear();
				}
				mmo.spottedTimer = 0f;
				CampaignInterface.instance.SetContactColor(mmo, forceZero: true);
			}
			for (int num5 = mmo.unitPrefabs.Count - 1; num5 > -1; num5--)
			{
				mmo.bombardments[num5] = 0;
				if (mmo.cellVariables[num5] == "DESTROYED")
				{
					if (mmo.currentFaction == 0)
					{
						int indexInDoubleList = GetIndexInDoubleList(mmo.unitPrefabs[num5], mmo.unitInstances[num5], campaignData.playerOwnedShipClasses, campaignData.playerOwnedShipInstances);
						if (indexInDoubleList != -1)
						{
							campaignData.playerOwnedShipClasses.RemoveAt(indexInDoubleList);
							campaignData.playerOwnedShipInstances.RemoveAt(indexInDoubleList);
							campaignData.playerOwnedRepairDays.RemoveAt(indexInDoubleList);
						}
					}
					else
					{
						int indexInDoubleList2 = GetIndexInDoubleList(mmo.unitPrefabs[num5], mmo.unitInstances[num5], campaignData.enemyOwnedShipClasses, campaignData.enemyOwnedShipInstances);
						if (indexInDoubleList2 != -1)
						{
							campaignData.enemyOwnedShipClasses.RemoveAt(indexInDoubleList2);
							campaignData.enemyOwnedShipInstances.RemoveAt(indexInDoubleList2);
							campaignData.enemyOwnedRepairDays.RemoveAt(indexInDoubleList2);
						}
					}
					campaignData.sunkShipClasses.Add(mmo.unitPrefabs[num5]);
					campaignData.sunkShipInstances.Add(mmo.unitInstances[num5]);
					campaignData.daySunk.Add(EngagementData.instance.date[0]);

	public float GetMobileMapObjectSpeed(MobileMapObject mmo)
	{
		SetMobileMapObjectSensors(mmo);
		SpriteRenderer componentInChildren = mmo.GetComponentInChildren<SpriteRenderer>();
		if (componentInChildren != null)
		{
			componentInChildren.sprite = CampaignInterface.instance.mobileMapSprites[GetMobileSpriteIndex(mmo)];
		}
		if (mmo.type != 0)
		{
			return EngagementManager.instance.unitDataDictionary[mmo.unitPrefabs[0]].maxKnots * campaignData.airGroupCruiseSpeed * 0.514444f;
		}
		List<float> list = new List<float>();
		for (int i = 0; i < mmo.unitPrefabs.Count; i++)
		{
			list.Add(EngagementManager.instance.unitDataDictionary[mmo.unitPrefabs[i]].maxKnots * MobileMapObject.GetDamageSpeedFactor(mmo, i));
		}
		float lowestFloat = Utilities.GetLowestFloat(list);
		if (mmo.currentFaction != 0 && CampaignAI.instance.seaMissions[mmo.missionIndex].missionType == MissionType.Home)
		{
			return lowestFloat * campaignData.taskForceHomeSpeed * 0.514444f;
		}
		return lowestFloat * campaignData.taskForceCruiseSpeed * 0.514444f;
	}

	public void SetMobileMapObjectSensors(MobileMapObject mmo)
	{
		List<float> list = new List<float>();
		List<float> list2 = new List<float>();
		List<float> list3 = new List<float>();
		List<float> list4 = new List<float>();

				UnitData unitData = EngagementManager.instance.unitDataDictionary[mmo.unitPrefabs[i]];

				list4.Add(unitData.passiveSonarStrength);
			}
			mmo.surfaceDetectionBaseRange = Mathf.Max(list.ToArray());
			mmo.airDetectionBaseRange = Mathf.Max(list2.ToArray());
			mmo.passiveSonarBaseRange = Mathf.Max(list4.ToArray());
			mmo.visualBaseRange = Mathf.Max(list3.ToArray());
		}

		{

			mmo.surfaceDetectionBaseRange = unitData2.surfaceRadarStrength;
			mmo.airDetectionBaseRange = unitData2.airRadarStrength;
			mmo.visualBaseRange = unitData2.unitVisualSizeModifier;

	}

	public float GetMobileEndurance(MobileMapObject mmo)
	{
		if (mmo.type != 1)
		{
			return 0f;
		}
		float num = EngagementManager.instance.unitDataDictionary[mmo.unitPrefabs[0]].range / mmo.speed * 1000f;
		num /= 2f;
		if (mmo.payload.Count <= 0)
		{
			return num;
		}
		List<float> endurance = EngagementManager.instance.unitAirDictionary[mmo.unitPrefabs[0]].endurance;
		if (endurance != null && endurance.Count > 0)
		{
			return num * endurance[mmo.payload[0]];
		}
		return num * 0.667f;
	}

			{
				return 3;

			{

				return 1;

	public bool AirAttemptLand(GameObjectAnimator goa)
	{
		if (goa.mobileMapObject.type != 1)
		{
			Debug.LogError("Sea group called: AirAttemptLand");
			return false;
		}
		GameObjectAnimator gameObjectAnimator = null;
		if (goa.mobileMapObject.createdBy.Count == 1)
		{
			MapLocation mapLocationObject = GetMapLocationObject(goa.mobileMapObject.createdBy[0]);
			if (mapLocationObject != null)
			{
				gameObjectAnimator = mapLocationObject.GetComponent<GameObjectAnimator>();
			}
		}
		else
		{
			MobileMapObject shipMMOThatCreatedAir = GetShipMMOThatCreatedAir(goa.mobileMapObject);
			if (shipMMOThatCreatedAir != null)
			{
				gameObjectAnimator = shipMMOThatCreatedAir.gameObjectAnimator;
			}
		}
		if (!(gameObjectAnimator != null) || GetDistanceInMetresBetweenMapObjects(goa.currentMap2DLocation, gameObjectAnimator.currentMap2DLocation) >= campaignData.interactionDistanceMetres)
		{
			return false;
		}
		if (campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour] && goa.mobileMapObject.escortingThisGOA != null)
		{
			goa.mobileMapObject.currentCampaignAction = 0;
			goa.mobileMapObject.endurance = GetMobileEndurance(goa.mobileMapObject);
			ResetAmmunition(goa.mobileMapObject);
			goa.mobileMapObject.ammunitionRemaining.Clear();
			AutoPlayerManager.instance.MoveGoaToEscorted(goa);
			return true;
		}
		if (goa.mobileMapObject.autoMove)
		{
			if (campaignData.airOpHours[CampaignInterface.instance.dateTime.Hour])
			{
				goa.mobileMapObject.currentCampaignAction = 0;
				goa.mobileMapObject.endurance = GetMobileEndurance(goa.mobileMapObject);
				ResetAmmunition(goa.mobileMapObject);
				if (goa.mobileMapObject.createdBy.Count == 1)
				{
					goa.mobileMapObject.mapWaypoints = new List<Vector2>(goa.mobileMapObject.autoPath);
					if (goa.mobileMapObject.mapWaypoints.Count != 0)
					{
						goa.mobileMapObject.mapWaypoints.RemoveAt(0);
						goa.Stop();
						CampaignInterface.instance.MoveMobileToDestination(goa, goa.mobileMapObject.autoPath[0]);
						return true;
					}
				}
				CampaignInterface.instance.CalculateAutoWaypoints(goa);
				if (goa.mobileMapObject.mapWaypoints.Count != 0)
				{
					goa.Stop();
					CampaignInterface.instance.MoveMobileToDestination(goa, goa.mobileMapObject.mapWaypoints[0]);
				}
			}
			else
			{
				AutoPlayerManager.instance.LandAutoScout(goa.mobileMapObject);
				if (goa.mobileMapObject.createdBy.Count == 1)
				{
					SetAircraftReturnedToLocation(goa.mobileMapObject, goa.mobileMapObject.unitPrefabs.Count);
				}
				else
				{
					SetAircraftReturnedToShip(goa.mobileMapObject, gameObjectAnimator.mobileMapObject);
				}
				CampaignInterface.instance.DestroyMapMobileObject(goa);
			}
			return true;
		}
		if (goa.mobileMapObject.createdBy.Count == 1)
		{
			SetAircraftReturnedToLocation(goa.mobileMapObject, goa.mobileMapObject.unitPrefabs.Count);
		}
		else
		{
			SetAircraftReturnedToShip(goa.mobileMapObject, gameObjectAnimator.mobileMapObject);
		}
		CampaignInterface.instance.DestroyMapMobileObject(goa);
		return true;
	}

	public void RemoveAllCargoFromShips(MobileMapObject mmo)
	{
		MapLocation closestLocation = GetClosestLocation(mmo.gameObjectAnimator);
		if (closestLocation == null)
		{
			return;
		}
		if (mmo.currentFaction == 0)
		{
			ErrorPanel.instance.SetErrorMessage("UnloadedCargo", mmo.mobileName, closestLocation.locationName, string.Empty);
		}
		mmo.unloadTimer = 0f;
		for (int i = 0; i < mmo.cargo.Count; i++)
		{
			List<int> intListFromString = GetIntListFromString(mmo.cargo[i]);
			for (int j = 0; j < intListFromString.Count; j++)
			{
				if (intListFromString[j] != 0)
				{
					CargoInterface.instance.SetCargoChange(mmo, i, j, closestLocation);
				}
			}
		}
		CheckLocationTakeover(closestLocation);
		if (closestLocation.currentFaction == 0)
		{
			if (mmo.currentFaction != 0)
			{
				ErrorPanel.instance.SetErrorMessage("LocationAttacked", closestLocation.locationName, string.Empty, string.Empty);
				CampaignInterface.instance.PauseCampaign();
				if (CampaignInterface.instance.currentMapObject != null && CampaignInterface.instance.currentMapObject.GetComponent<MapLocation>() == closestLocation)
				{
					CampaignInterface.instance.SetLocationPanelData(CampaignInterface.instance.currentMapObject);
				}
			}
		}
		else
		{
			if (closestLocation.currentFaction != 1)
			{
				return;
			}
			bool[] array = new bool[2]
			{
				IsPortUpgradable(closestLocation, Faction.Enemy),
				IsAirfieldUpgradable(closestLocation, Faction.Enemy)
			};
			if (campaignData.maxAirAt.Contains(closestLocation.locationID))
			{
				if (array[1])
				{
					ApplyUpgradeLocation(closestLocation, 1, player: false);
					return;
				}
				if (closestLocation.airfieldLevel != campaignData.maxAirfield[1])
				{
					return;
				}
			}
			if (campaignData.maxPortAt.Contains(closestLocation.locationID))
			{
				if (array[0])
				{
					ApplyUpgradeLocation(closestLocation, 0, player: false);
					return;
				}
				if (closestLocation.portLevel != campaignData.maxPort[1])
				{
					return;
				}
			}
			if (IsAirfieldUpgradable(closestLocation, Faction.Enemy))
			{
				ApplyUpgradeLocation(closestLocation, 1, player: false);
			}
			else if (IsPortUpgradable(closestLocation, Faction.Enemy))
			{
				ApplyUpgradeLocation(closestLocation, 0, player: false);
			}
		}
	}

	private void CheckWinCondition()
	{
		if (campaignData.campaignEnded || !CheckEachSideHasHomePort())
		{
			return;
		}
		if (campaignData.winEnemyShipsSunk > 0)
		{
			int num = 0;
			int num2 = 1;
			for (int i = 0; i < campaignData.factionSunk.Count; i++)
			{
				if (campaignData.factionSunk[i] == num2)
				{
					num++;
				}
			}
			int num3 = 0;
			num2 = 0;
			for (int j = 0; j < campaignData.factionSunk.Count; j++)
			{
				if (campaignData.factionSunk[j] == num2)
				{
					num3++;
				}
			}
			if (num - num3 >= campaignData.winEnemyShipsSunk)
			{
				if (campaignData.winEnemyShipSunkSupplyIncrease <= 1f)
				{
					CampaignEvent.instance.CreateEvent("shipsSunk");
					campaignData.campaignEnded = true;
					return;
				}
				if (!campaignData.campaignDominated)
				{
					ErrorPanel.instance.SetErrorMessage("SeaLanesOpen", campaignData.winEnemyShipSunkSupplyIncrease + " " + LanguageManager.instance.GetDictionaryString(LanguageManager.generalDictionary, "Multiplied"), string.Empty, string.Empty);
					campaignData.campaignDominated = true;
				}
			}
			else if (campaignData.campaignDominated)
			{
				ErrorPanel.instance.SetErrorMessage("SeaLanesClosed", string.Empty, string.Empty, string.Empty);
				campaignData.campaignDominated = false;
			}
		}
		int num4 = campaignData.mustControl.Count;
		for (int k = 0; k < landMapLocations.Count; k++)
		{
			if ((landMapLocations[k].currentFaction == 0 || landMapLocations[k].currentFaction == 2) && campaignData.mustControl.Contains(landMapLocations[k].locationID))
			{
				num4--;
			}
		}
		if (num4 == 0 && CheckMaxAirAndPort())
		{
			CampaignEvent.instance.CreateEvent("victory");
			campaignData.campaignEnded = true;
			return;
		}
		num4 = campaignData.mustControl.Count;
		for (int l = 0; l < landMapLocations.Count; l++)
		{
			if ((landMapLocations[l].currentFaction == 1 || landMapLocations[l].currentFaction == 2) && campaignData.mustControl.Contains(landMapLocations[l].locationID))
			{
				num4--;
			}
		}
		if (num4 == 0 && CheckMaxAirAndPort())
		{
			CampaignEvent.instance.CreateEvent("defeat");
			campaignData.campaignEnded = true;
		}
	}

	private bool CheckMaxAirAndPort()

		{
			if (GetMapLocationObject(campaignData.maxAirAt[i]).airfieldLevel != campaignData.maxAirfield[0])

		{
			if (GetMapLocationObject(campaignData.maxPortAt[j]).portLevel != campaignData.maxPort[0])
			{
				return false;
			}
		}
		return true;
	}

			return;

		int num4 = 0;

		{
			float value = UnityEngine.Random.value;
			int num6 = 0;
			if (value < campaignData.strategicBombLosses[0] + num2 && UnityEngine.Random.value > armorAbsorbProbability * campaignData.strategicBombAttackerArmorModifier[0])
			{
				num6 += 3;
			}
			if (value < campaignData.strategicBombLosses[1] + num2 && UnityEngine.Random.value > armorAbsorbProbability * campaignData.strategicBombAttackerArmorModifier[1])
			{
				num6++;
			}
			if (value < campaignData.strategicBombLosses[2] + num2 && UnityEngine.Random.value > armorAbsorbProbability * campaignData.strategicBombAttackerArmorModifier[2])

				num6++;
			}
			if (num6 > 2)

				mmo.unitPrefabs.RemoveAt(num5);
				list.RemoveAt(num5);
				num4++;
				if (mmo.currentFaction == 0)
				{
					campaignData.playerAircraftDowned++;
				}
				else
				{
					campaignData.otherAircraftDowned++;
				}
			}
			else
			{

		for (int i = 0; i < indexes.Count; i++)
		{
			num += GetBombardDamage(mmo.unitPrefabs[indexes[i]]) * GetBombardFactor(mmo.bombardments[indexes[i]]);
			mmo.bombardments[indexes[i]]++;
		}
		if (num == 0f)
		{
			if (mmo.currentFaction == 0)
			{
				ErrorPanel.instance.SetErrorMessage("NoBombardAmmo", ml.locationName, string.Empty, string.Empty);
			}
			return;
		}
		if (mmo.currentFaction == 0)
		{
			ErrorPanel.instance.SetErrorMessage("Bombarded", mmo.mobileName, ml.locationName, string.Empty);
		}
		else
		{
			ErrorPanel.instance.SetErrorMessage("EnemyBombarded", ml.locationName, string.Empty, string.Empty);
			CampaignInterface.instance.PauseCampaign();
		}
		DamageLocation(ml, num);
		CampaignInterface.instance.CreateStratMapExplosion(new Vector3(ml.locationPosition.x, ml.locationPosition.y, 0f));
		if (Config.instance.showCampaignLocationRadii)
		{
			MapRadii.instance.SetMapRadii();
		}
	}

	private void DamageLocation(MapLocation ml, float damage, float cooldownModifier = 1f)
	{
		damage *= (float)campaignData.maxPort[0] + (float)campaignData.maxAirfield[0] + 1f - (float)ml.portLevel - (float)ml.airfieldLevel;
		damage *= UnityEngine.Random.Range(campaignData.bombardDamageRange[0], campaignData.bombardDamageRange[1]);
		float num = 1f - Mathf.Clamp01(ml.aircraftCooldown / campaignData.bombardMaxAirCooldown);
		damage *= num;
		ml.integrity -= (int)damage;
		if (ml.integrity < -100)
		{
			ml.portLevel = DecreaseUpgrade(ml, ml.portLevel);
			ml.airfieldLevel = DecreaseUpgrade(ml, ml.airfieldLevel);
		}
		else if (ml.integrity < 1)
		{
			if (ml.airfieldLevel >= ml.portLevel)
			{
				ml.airfieldLevel = DecreaseUpgrade(ml, ml.airfieldLevel);
			}
			else
			{
				ml.portLevel = DecreaseUpgrade(ml, ml.portLevel);
			}
		}
		if (ml.airfieldLevel > 0)
		{
			ml.aircraftCooldown += damage * campaignData.bombardAirCooldownFactor * cooldownModifier;
		}
	}

	private int DecreaseUpgrade(MapLocation ml, int facility)
	{
		facility--;
		if (facility < 1)
		{
			facility = 0;
		}
		ml.integrity = 100;
		if (ml.GetComponent<GameObjectAnimator>() == CampaignInterface.instance.currentMapObject)
		{
			CampaignInterface.instance.SetLocationPanelData(ml.GetComponent<GameObjectAnimator>());
		}
		return facility;
	}

	private float GetBombardFactor(int number)
	{
		if (number > 2)
		{
			return 0f;
		}
		return 1f;
	}

	private float GetBombardDamage(string prefabName)
	{
		if (EngagementManager.instance.unitDataDictionary[prefabName].unitSubtype == UnitSubType.Battleship)
		{
			return 10f;
		}
		if (EngagementManager.instance.unitDataDictionary[prefabName].unitSubtype == UnitSubType.Battlecruiser)
		{
			return 8f;
		}
		if (EngagementManager.instance.unitDataDictionary[prefabName].unitSubtype == UnitSubType.Heavy_Cruiser)
		{
			return 5f;
		}
		return 2f;
	}

	public List<int> GetbombardIndexes(MobileMapObject mmo)
	{
		List<int> list = Utilities.DoesMobileContainSubtype(mmo, UnitSubType.Battleship);
		list.AddRange(Utilities.DoesMobileContainSubtype(mmo, UnitSubType.Battlecruiser));
		list.AddRange(Utilities.DoesMobileContainSubtype(mmo, UnitSubType.Heavy_Cruiser));
		for (int num = list.Count - 1; num > -1; num--)
		{
			if (mmo.cellVariables[num] != string.Empty && !Utilities.GetSubsystemAvailable(mmo.unitPrefabs[num], mmo.cellVariables[num], CellType.Primary))
			{
				list.RemoveAt(num);
			}
		}
		return list;
	}
