	private void Drive(CampaignInterface ci)
	{
		if (!CampaignManager.instance.gameObject.activeInHierarchy)
		{
			Off();
			return;
		}
		bool flag = OverGameUi();
		if (!flag && (Input.GetMouseButtonDown(0) || Input.GetMouseButtonDown(1)))
		{
			dragging = true;
			lastMouse = Input.mousePosition;
			downButton = ((!Input.GetMouseButtonDown(0)) ? 1 : 0);
			downPos = Input.mousePosition;
		}
		if (downButton >= 0 && Input.GetMouseButtonUp(downButton))
		{
			bool flag2 = (Input.mousePosition - downPos).magnitude < 6f;
			int button = downButton;
			downButton = -1;
			if (flag2 && Clicks && !flag)
			{
				Click(ci, button);
			}
		}
		if (!Input.GetMouseButton(0) && !Input.GetMouseButton(1))
		{
			dragging = false;
		}
		if (dragging)
		{
			Vector3 mousePosition = Input.mousePosition;
			float num = DragSpeed * dist / (ZoomStart * Radius);
			viewLon -= (mousePosition.x - lastMouse.x) * num;
			viewLat -= (mousePosition.y - lastMouse.y) * num;
			lastMouse = mousePosition;
		}
		float unscaledDeltaTime = Time.unscaledDeltaTime;
		if (Input.GetKey(KeyCode.LeftArrow))
		{
			viewLon -= KeySpeed * unscaledDeltaTime;
		}
		if (Input.GetKey(KeyCode.RightArrow))
		{
			viewLon += KeySpeed * unscaledDeltaTime;
		}
		if (Input.GetKey(KeyCode.UpArrow))
		{
			viewLat += KeySpeed * unscaledDeltaTime;
		}
		if (Input.GetKey(KeyCode.DownArrow))
		{
			viewLat -= KeySpeed * unscaledDeltaTime;
		}
		for (viewLat = Mathf.Clamp(viewLat, -85f, 85f); viewLon < 0f; viewLon += 360f)
		{
		}
		while (viewLon >= 360f)
		{
			viewLon -= 360f;
		}
		float num2 = (flag ? 0f : Outward(ci.map));
		if (num2 < -0.0001f)
		{
			if (dist <= ZoomMin * Radius + 0.01f && AutoDropToMap && Time.unscaledTime - lastSwitch >= SwitchCooldown)
			{
				DropToMap(ci);
				return;
			}
			dist *= 1f - ZoomStep;
		}
		else if (num2 > 0.0001f)
		{
			dist *= 1f + ZoomStep;
		}
		dist = Mathf.Clamp(dist, ZoomMin * Radius, ZoomMax * Radius);
		Place();
		sunTimer -= unscaledDeltaTime;
		if (sunTimer <= 0f)
		{
			sunTimer = 0.5f;
			Shade(ci);
		}
		markerTimer -= unscaledDeltaTime;
		if (markerTimer <= 0f)
		{
			markerTimer = 1f;
			Markers(ci);
		}
		Facing();
		if (Clicks)
		{
			Hover(ci, flag);
			RouteMouse(ci, flag);
			RouteLines(ci);
			Cursor(ci, flag);
		}
		if (Rings)
		{
			RingLines(ci);
		}
		if (Weather)
		{
			WeatherPatches(ci);
		}
	}

	private void Place()
	{
		Vector3 vector = Dir(viewLat, viewLon);
		cam.transform.position = centre + vector * dist;
		cam.transform.LookAt(centre, Vector3.up);
		if (haloObj != null)
		{
			haloObj.transform.position = centre;
			haloObj.transform.rotation = cam.transform.rotation;
		}
		if (starsObj != null)
		{
			starsObj.transform.position = centre;
		}
	}

	private bool OverGameUi()
	{
		try
		{
			EventSystem current = EventSystem.current;
			if (current == null)
			{
				return false;
			}
			PointerEventData pointerEventData = new PointerEventData(current);
			pointerEventData.position = Input.mousePosition;
			List<RaycastResult> list = new List<RaycastResult>();
			current.RaycastAll(pointerEventData, list);
			for (int i = 0; i < list.Count; i++)
			{
				if (!(list[i].gameObject == null) && !(list[i].gameObject == blocker))
				{
					return true;
				}
			}
			return false;
		}
		catch (Exception)
		{
			return false;
		}
	}

	private void DropToMap(CampaignInterface ci)
	{
		CampaignData campaignData = CampaignManager.instance.campaignData;
		Vector2 destination = MapPoint(campaignData, viewLat, viewLon);
		Off();
		lastSwitch = Time.unscaledTime;
		if (ci.map != null && destination.x >= -0.5f && destination.x <= 0.5f && destination.y >= -0.5f && destination.y <= 0.5f)
		{
			float zoomLevel = Mathf.Lerp(ci.map.zoomMinDistance, ci.map.zoomMaxDistance, Mathf.Clamp01(DropZoom));
			ci.map.FlyToLocation(destination, 0.6f, zoomLevel);
			Debug.Log("[GLOBE] dropped to map at " + destination.x.ToString("0.000") + " " + destination.y.ToString("0.000"));
		}
	}

	public static Vector3 Dir(float lat, float lon)
	{
		float f = lat * ((float)Math.PI / 180f);
		float f2 = lon * ((float)Math.PI / 180f);
		return new Vector3(Mathf.Cos(f) * Mathf.Cos(f2), Mathf.Sin(f), Mathf.Cos(f) * Mathf.Sin(f2));
	}

	public static Vector2 MapPoint(CampaignData d, float lat, float lon)
	{
		double num = ModGeo.Lon(d, -0.5f);
		double num2;
		for (num2 = lon; num2 < num; num2 += 360.0)
		{
		}
		while (num2 >= num + 360.0)
		{
			num2 -= 360.0;
		}
		float x = (float)((num2 * 240.0 - (double)d.mapOriginXY[0]) / (double)d.mapPixelWidth - 0.5);
		float y = (float)((((double)lat + 90.0) * 240.0 - (double)d.mapOriginXY[1]) / ((double)d.mapPixelWidth * 0.5) - 0.5);
		return new Vector2(x, y);
	}

	private Vector3 SunDir(CampaignInterface ci)
	{
		float num = UtcOffsetHours;
		if (num > 24f || num < -24f)
		{
			CampaignData campaignData = CampaignManager.instance.campaignData;
			double num2 = ModGeo.Lon(campaignData, 0f);
			if (num2 > 180.0)
			{
				num2 -= 360.0;
			}
			num = Mathf.Round((float)(num2 / 15.0));
		}
		return SunDir(ci.dateTime, num);
	}

	private void Facing()
	{
		Vector3 normalized = (cam.transform.position - centre).normalized;
		float num = Radius / dist;
		if (globeTris != null && (cam.transform.position - lastCamPos).sqrMagnitude > 0.0001f)
		{
			lastCamPos = cam.transform.position;
			frontTris.Clear();
			Vector3 vector = cam.transform.position - centre;
			for (int i = 0; i < globeTris.Length; i += 3)
			{
				int num2 = globeTris[i];
				int num3 = globeTris[i + 1];
				int num4 = globeTris[i + 2];
				Vector3 lhs = globeNormals[num2] + globeNormals[num3] + globeNormals[num4];
				Vector3 vector2 = (globeNormals[num2] + globeNormals[num3] + globeNormals[num4]) * (Radius / 3f);
				if (Vector3.Dot(lhs, vector - vector2) > 0f)
				{
					frontTris.Add(num2);
					frontTris.Add(num3);
					frontTris.Add(num4);
				}
			}
			globeMesh.SetTriangles(frontTris, 0);
		}
		PlaceEntries(normalized, num);
		if (!(ribbonObj != null) || ribbonColors == null)
		{
			return;
		}
		bool flag = false;
		for (int j = 0; j < ribbonDirs.Count; j++)
		{
			float num5 = ((Vector3.Dot(ribbonDirs[j], normalized) > num) ? 1f : 0f);
			int num6 = j * 2;
			if (ribbonColors[num6].a != num5)
			{
				ribbonColors[num6].a = num5;
				ribbonColors[num6 + 1].a = num5;
				flag = true;
			}
		}
		if (flag)
		{
			ribbonObj.GetComponent<MeshFilter>().sharedMesh.colors = ribbonColors;
		}
	}

	private void Markers(CampaignInterface ci)
	{
		for (int i = 0; i < entries.Count; i++)
		{
			if (entries[i].go != null)
			{
				UnityEngine.Object.Destroy(entries[i].go);
			}
		}
		entries.Clear();
		hover = null;
		if (!ShowMarkers || overlay == null)
		{
			return;
		}
		CampaignManager instance = CampaignManager.instance;
		CampaignData campaignData = instance.campaignData;
		if (instance.landMapLocations != null)
		{
			for (int j = 0; j < instance.landMapLocations.Count; j++)
			{
				MapLocation mapLocation = instance.landMapLocations[j];
				if (!(mapLocation == null))
				{
					AddEntry(campaignData, mapLocation.locationPosition, mapLocation.GetComponentInChildren<SpriteRenderer>(), mapLocation.locationName, 1f, mapLocation.gameObjectAnimator);
				}
			}
		}
		if (instance.playerMobile != null)
		{
			for (int k = 0; k < instance.playerMobile.Count; k++)
			{
				GameObjectAnimator gameObjectAnimator = instance.playerMobile[k];
				if (!(gameObjectAnimator == null) && !(gameObjectAnimator.mobileMapObject == null) && gameObjectAnimator.mobileMapObject.unitPrefabs != null && gameObjectAnimator.mobileMapObject.unitPrefabs.Count != 0)
				{
					AddEntry(campaignData, gameObjectAnimator.currentMap2DLocation, gameObjectAnimator.GetComponentInChildren<SpriteRenderer>(), gameObjectAnimator.mobileMapObject.mobileName, 1.1f, gameObjectAnimator);
				}
			}
		}
		if (instance.spottedContacts != null)
		{
			for (int l = 0; l < instance.spottedContacts.Count; l++)
			{
				MobileMapObject mobileMapObject = instance.spottedContacts[l];
				if (!(mobileMapObject == null) && !(mobileMapObject.gameObjectAnimator == null))
				{
					AddEntry(campaignData, mobileMapObject.gameObjectAnimator.currentMap2DLocation, mobileMapObject.gameObjectAnimator.GetComponentInChildren<SpriteRenderer>(), mobileMapObject.mobileName, 1.1f, mobileMapObject.gameObjectAnimator);
				}
			}
		}
		if (Notes && ci.mapMarkerParent != null)
		{
			MapLocation[] componentsInChildren = ci.mapMarkerParent.GetComponentsInChildren<MapLocation>();
			foreach (MapLocation mapLocation2 in componentsInChildren)
			{
				if (!(mapLocation2 == null) && !(mapLocation2.gameObjectAnimator == null))
				{
					AddEntry(campaignData, mapLocation2.locationPosition, mapLocation2.GetComponentInChildren<SpriteRenderer>(), mapLocation2.locationName, 0.9f, mapLocation2.gameObjectAnimator);
				}
			}
		}
		if (Weather && WeatherManager.instance != null && WeatherManager.instance.seaStateMeshes != null)
		{
			List<TextMesh> seaStateMeshes = WeatherManager.instance.seaStateMeshes;
			for (int n = 0; n < seaStateMeshes.Count; n++)
			{
				TextMesh textMesh = seaStateMeshes[n];
				if (!(textMesh == null) && !string.IsNullOrEmpty(textMesh.text) && textMesh.gameObject.activeInHierarchy)
				{
					Vector3 localPosition = textMesh.transform.localPosition;
					AddEntry(campaignData, new Vector2(localPosition.x / 200f, localPosition.y / 100f), null, textMesh.text, 0.01f, null);
					Entry entry = entries[entries.Count - 1];
					entry.icon.enabled = false;
					if (entry.label != null)
					{
						entry.label.color = new Color(0.8f, 0.9f, 1f, 0.9f);
						entry.label.fontSize = LabelPx - 1;
						entry.shadow.fontSize = LabelPx - 1;
					}
				}
			}
		}
		Vector3 normalized = (cam.transform.position - centre).normalized;
		PlaceEntries(normalized, Radius / dist);
	}

	private void Build(CampaignData d)
	{
		built = true;
		shader = Shader.Find("Particles/Alpha Blended");
		if (shader == null)
		{
			shader = Shader.Find("Sprites/Default");
			Debug.Log("[GLOBE] Particles/Alpha Blended not found, using Sprites/Default");
		}
		white = new Texture2D(2, 2, TextureFormat.RGBA32, mipmap: false);
		white.SetPixels(new Color[4]
		{
			Color.white,
			Color.white,
			Color.white,
			Color.white
		});
		white.Apply();
		root = new GameObject("ModGlobeRoot");
		root.transform.position = centre;
		UnityEngine.Object.DontDestroyOnLoad(root);
		GameObject gameObject = new GameObject("ModGlobeCamera");
		gameObject.transform.SetParent(root.transform, worldPositionStays: false);
		cam = gameObject.AddComponent<Camera>();
		cam.clearFlags = CameraClearFlags.Color;
		cam.backgroundColor = Color.black;
		cam.nearClipPlane = 1f;
		cam.farClipPlane = 60000f;
		cam.fieldOfView = 40f;
		Camera main = Camera.main;
		cam.depth = ((main != null) ? main.depth : 0f) + 5f;
		cam.cullingMask = -1;
		cam.enabled = false;
		globeMesh = Sphere(96, 192, Radius, out globeNormals);
		globeTris = globeMesh.triangles;
		globeMesh.MarkDynamic();
		globeColors = new Color[globeNormals.Length];
		for (int i = 0; i < globeColors.Length; i++)
		{
			globeColors[i] = Color.white;
		}
		globeMesh.colors = globeColors;
		globeObj = MeshObj("Globe", globeMesh, EarthTexture(), Color.white, 3002);
		starsObj = MeshObj("Stars", Sphere(24, 48, 30000f, out var _), StarTexture(), Color.white, 3000);
		haloObj = MeshObj("Halo", Quad(Radius * 2.6f), HaloTexture(), new Color(0.45f, 0.65f, 1f, 1f), 3001);
		if (d != null)
		{
			BuildRibbon(d);
			BuildBlocker();
		}
		root.SetActive(value: false);
		Debug.Log("[GLOBE] built: " + globeNormals.Length + " vertices, shader " + ((shader != null) ? shader.name : "none"));
	}

	private Material Unlit(Color c, int queue)
	{
		Material material = new Material(shader);
		material.mainTexture = white;
		if (material.HasProperty("_TintColor"))
		{
			material.SetColor("_TintColor", new Color(0.5f, 0.5f, 0.5f, 0.5f));
		}
		if (material.HasProperty("_Color"))
		{
			material.color = Color.white;
		}
		material.renderQueue = queue;
		return Tinted(material, c);
	}

	private Material Tinted(Material m, Color c)
	{
		if (m.HasProperty("_TintColor"))
		{
			m.SetColor("_TintColor", new Color(c.r * 0.5f, c.g * 0.5f, c.b * 0.5f, c.a * 0.5f));
		}
		else if (m.HasProperty("_Color"))
		{
			m.color = c;
		}
		return m;
	}

	private GameObject MeshObj(string name, Mesh mesh, Texture2D tex, Color tint, int queue)
	{
		GameObject gameObject = new GameObject(name);
		gameObject.transform.SetParent(root.transform, worldPositionStays: false);
		gameObject.AddComponent<MeshFilter>().sharedMesh = mesh;
		MeshRenderer meshRenderer = gameObject.AddComponent<MeshRenderer>();
		Material material = new Material(shader);
		material.mainTexture = tex;
		material.renderQueue = queue;
		Tinted(material, tint);
		meshRenderer.sharedMaterial = material;
		meshRenderer.shadowCastingMode = ShadowCastingMode.Off;
		meshRenderer.receiveShadows = false;
		return gameObject;
	}

	private Texture2D EarthTexture()
	{
		Texture2D texture2D = null;
		try
		{
			texture2D = AssetLoader.instance.ImportTexture2D("ModAssets/globe/earth.jpg");
		}
		catch (Exception ex)
		{
			Debug.Log("[GLOBE] earth.jpg: " + ex.Message);
		}
		if (texture2D != null && texture2D.width > 64)
		{

	private bool MouseMapPoint(CampaignData d, out Vector2 p, out Vector3 dir)
	{
		p = Vector2.zero;
		dir = Vector3.zero;
		if (cam == null)
		{
			return false;
		}
		Ray ray = cam.ScreenPointToRay(Input.mousePosition);
		Vector3 vector = ray.origin - centre;
		float num = Vector3.Dot(vector, ray.direction);
		float num2 = Vector3.Dot(vector, vector) - Radius * Radius;
		float num3 = num * num - num2;
		if (num3 < 0f)
		{
			return false;
		}
		float num4 = 0f - num - Mathf.Sqrt(num3);
		if (num4 < 0f)
		{
			return false;
		}
		dir = (ray.origin + ray.direction * num4 - centre).normalized;
		float lat = Mathf.Asin(Mathf.Clamp(dir.y, -1f, 1f)) * 57.29578f;
		float num5 = Mathf.Atan2(dir.z, dir.x) * 57.29578f;
		if (num5 < 0f)
		{
			num5 += 360f;
		}
		p = MapPoint(d, lat, num5);
		return true;
	}
