	public static void Build()
	{
		try
		{
			MenuSystem instance = MenuSystem.instance;
			if (instance == null || instance.mainInterface == null)
			{
				return;
			}
			if (root != null)
			{
				UnityEngine.Object.Destroy(root);
			}
			root = null;
			panel = null;
			Canvas componentInParent = instance.mainInterface.GetComponentInParent<Canvas>();
			if (componentInParent == null)
			{
				Debug.LogError("[VERSION] no canvas found");
				return;
			}
			componentInParent = componentInParent.rootCanvas;
			Font font = null;
			Text componentInChildren = instance.mainInterface.GetComponentInChildren<Text>(includeInactive: true);
			if (componentInChildren != null)
			{
				font = componentInChildren.font;
			}
			if (font == null)
			{
				font = Resources.GetBuiltinResource<Font>("Arial.ttf");
			}
			Font font2 = font;
			int fontSize = 16;
			if (TooltipManager.instance != null && TooltipManager.instance.toolTipText != null && TooltipManager.instance.toolTipText.font != null)
			{
				font2 = TooltipManager.instance.toolTipText.font;
				fontSize = TooltipManager.instance.toolTipText.fontSize;
			}
			Vector3[] array = new Vector3[4];
			((RectTransform)componentInParent.transform).GetWorldCorners(array);
			float x = componentInParent.transform.lossyScale.x;
			GameObject gameObject = new GameObject("ModVersion", typeof(RectTransform));
			gameObject.transform.SetParent(instance.mainInterface.transform, worldPositionStays: false);
			gameObject.transform.SetAsLastSibling();
			root = gameObject;
			RectTransform rectTransform = NewRect("ChangeLogButton", gameObject.transform, new Vector2(1f, 1f));
			rectTransform.sizeDelta = new Vector2(190f, 44f);
			rectTransform.position = new Vector3(array[2].x - 30f * x, array[2].y - 20f * x, array[2].z);
			MakeButton(rectTransform, font, 22, "Change Log".ToUpper(), Open);
			RectTransform rectTransform2 = NewRect("Version", gameObject.transform, new Vector2(0f, 0f));
			rectTransform2.sizeDelta = new Vector2(500f, 60f);
			rectTransform2.position = new Vector3(array[0].x + 30f * x, array[0].y + 20f * x, array[0].z);
			NewText(rectTransform2, font, 20, TextAnchor.LowerLeft, bodyColour).text = "Plague's Better WOTS mod\n<color=#E0B030>Version 0.32.19 globe test 19</color>";
			BuildPanel(gameObject.transform, array, font2, fontSize);
			gameObject.transform.SetAsLastSibling();
		}
		catch (Exception ex)
		{
			Debug.LogError("[VERSION] build failed: " + ex.ToString());
		}
	}

	private static RectTransform NewRect(string name, Transform parent, Vector2 pivot)
	{
		GameObject gameObject = new GameObject(name, typeof(RectTransform));
		gameObject.transform.SetParent(parent, worldPositionStays: false);
		RectTransform component = gameObject.GetComponent<RectTransform>();
		component.anchorMin = pivot;
		component.anchorMax = pivot;
		component.pivot = pivot;
		return component;
	}

	private static Text NewText(RectTransform rt, Font font, int size, TextAnchor anchor, Color colour)
	{
		Text text = rt.gameObject.AddComponent<Text>();
		text.font = font;
		text.fontSize = size;
		text.alignment = anchor;
		text.color = colour;
		text.supportRichText = true;
		text.horizontalOverflow = HorizontalWrapMode.Wrap;
		text.verticalOverflow = VerticalWrapMode.Overflow;
		text.raycastTarget = false;
		return text;
	}

	private static void Open()
	{
		if (!(panel == null))
		{
			panel.transform.SetAsLastSibling();
			panel.SetActive(value: true);
			if (contentRect != null)
			{
				contentRect.anchoredPosition = Vector2.zero;
			}
		}
	}

	private static void Close()
	{
		if (panel != null)
		{
			panel.SetActive(value: false);
		}
	}

	private static void MakeButton(RectTransform rt, Font font, int size, string label, UnityAction action)
	{
		Image image2 = rt.gameObject.AddComponent<Image>();
		image2.color = gold;
		Button button = rt.gameObject.AddComponent<Button>();
		button.targetGraphic = image2;
		button.onClick.AddListener(action);
		RectTransform rectTransform = NewRect("Back", rt, new Vector2(0.5f, 0.5f));
		Stretch(rectTransform, 2f, 2f, 2f, 2f);
		Image image3 = rectTransform.gameObject.AddComponent<Image>();
		image3.color = buttonBack;
		image3.raycastTarget = false;
		RectTransform rt2 = NewRect("Text", rt, new Vector2(0.5f, 0.5f));
		Stretch(rt2, 0f, 0f, 0f, 0f);
		NewText(rt2, font, size, TextAnchor.MiddleCenter, Color.white).text = label;
	}

	private static void Stretch(RectTransform rt, float left, float bottom, float right, float top)
	{
		rt.anchorMin = Vector2.zero;
		rt.anchorMax = Vector2.one;
		rt.offsetMin = new Vector2(left, bottom);
		rt.offsetMax = new Vector2(0f - right, 0f - top);
	}
