	public void SetCampaignDifficulty()
	{
		switch (LanguageManager.instance.campaignDifficultyDropdown.value)
		{
		case 0:
			OptionsManager.instance.campaignDifficultyMultipliers = campDifficultyPreset0;
			break;
		case 1:
			OptionsManager.instance.campaignDifficultyMultipliers = campDifficultyPreset1;
			break;
		case 2:
			OptionsManager.instance.campaignDifficultyMultipliers = campDifficultyPreset2;
			break;
		case 3:
			OptionsManager.instance.campaignDifficultyMultipliers = campDifficultyPreset3;
			break;
		case 4:
			OptionsManager.instance.campaignDifficultyMultipliers = campDifficultyPreset4;
			break;
		}
	}
