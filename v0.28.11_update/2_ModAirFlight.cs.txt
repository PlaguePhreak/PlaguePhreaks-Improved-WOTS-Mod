using UnityEngine;

// v0.28: per-aircraft flight characteristics and the pure physics formulas of the energy model. Table for ALL 135 aircraft (the 0.27 table had 21).
// Changes from 0.27:
//   * a hand-set row for every aircraft key (stock names zero, hellcat, corsair, val, kate ... included), see Build()
//   * altitude performance: engine power falls above the plane's critical altitude (critAlt, altLoss), the stall speed rises with thin air (AltStallShare)
//   * compressibility: parasite drag rises steeply near the dive speed limit, so the limit is no longer a hard cap (0.26.8). Beyond the structural limit the
//     wings come off (ModAirDamage.Overspeed)
//   * one LIVE row per aircraft (Of) that ModAirDamage.Refresh updates every tick: damage, altitude, bombs and fuel change the thrust, drag, stall and turn
//     scale fields of that copy while the table row stays the pristine one
//   * per-aircraft flight FILE: unit/air/<key>/<key>_flight.txt (in the Override folder or the game folder) overwrites any field of the row (JSON, see Docs)
//   * a row with maxSpeed -1 keeps the game's value (it was a bug in 0.27)
// To change one plane, edit its line in Build(), Compile, Save Module. Or drop a flight file next to the aircraft's data file (no compile needed).
//
// ---------------------------------------------------------------------------------------------------------------------------------------
// UNITS. Speeds are game units per second (x 10 = metres per second, x 36 = km/h: 14.2 = 142 m/s = 511 km/h). Turn rates are degrees per second.
// Climb power is metres per second. Convergence is metres. Wing loading is kg per square metre. Altitudes in metres.
//
// THE ENERGY EQUATION (this is what every number below feeds):
//     E = h + v^2 / (2 g)                specific energy in metres: height plus the height the speed would climb to
//     dE/dt = Ps = v (T - D) / (m g)     it only grows by thrust beyond drag, and falls with every turn, climb or skid that costs drag
// A dive turns height into speed at no cost, a zoom climb turns speed into height at no cost. Only thrust adds energy and only drag takes it.
//
// ENGINE: constant power P. Thrust acceleration T = throttle * P / v (never less than 35% of top speed). Parasite drag D0 = P v^2 / vmax^3, so
//     thrust equals drag exactly at top speed. P is set from climbPower: at the best climb speed (60% of top) the spare power climbs at climbPower.
//     Above critAlt the power falls by altLoss per 1000 m (0.10 = 10% of the power per km), never below MinAltPower.
// TURN DRAG:   Dturn = (turnRate / instantTurnRate)^2 * wingLoading * TurnDragK        (metres per second squared, taken off the speed)
// TURN LIMIT:  lift limits the turn below corner speed, the airframe limit (load factor nMax) above it, so the turn rate is
//     n = min(nMax, (v / stall)^2),   w = g sqrt(n^2 - 1) / v,   scaled so that w = instantTurnRate at corner speed.
// ---------------------------------------------------------------------------------------------------------------------------------------
public static class ModAirFlight
{
	[Serializable]
	public class Row
	{
		public string name;
		public float maxSpeed;           // level top speed. > 0 replaces the game's own value for this type, -1 keeps the game's value
		public float cornerSpeed;        // speed of the best instantaneous turn (and the highest speed that can use the full airframe load factor)
		public float stallSpeed;         // 1 g stall speed, flaps up
		public float sustainedTurnRate;  // degrees per second it can hold at corner speed without losing speed
		public float instantTurnRate;    // degrees per second at corner speed, paid for with speed
		public float climbPower;         // best sustained climb rate, metres per second
		public float wingLoading;        // kg per square metre. Heavier planes bleed more speed in a turn
		public float rollRate;           // degrees per second at low speed
		public float flapBonus;          // share by which combat flaps lower the stall speed (0.12 = 12% lower)
		public float flapSpeedLimit;     // flaps cannot be used (they are retracted) above this speed
		public float flapDragMultiplier; // parasite drag multiplier with flaps out
		public float convergeAt;         // metres where the wing guns cross
		public float noseConvergeAt;     // metres for guns mounted on the centre line (cowl guns)
		public float diveSpeed;          // dive speed limit (the structural limit, Vne) as a multiple of maxSpeed
		public float rollFade;           // share of the roll rate left at the dive speed limit (heavy ailerons at speed: Zero 0.25)
		public bool tuned;               // true = hand-set row from this table, false = made from UnitData
		// ---- v0.28
		public float critAlt;            // metres: highest altitude with full engine power (supercharger critical altitude)
		public float altLoss;            // share of the power lost per 1000 m above critAlt
		public float fuelMinutes;        // minutes of fuel at full power (combat fuel). Cruise burns about a third
		// ---- v0.28 live scales (the table row has 1; ModAirDamage.Refresh sets them on the per-aircraft copy every tick)
		public float thrustScale = 1f;   // engine damage, altitude, bombs
		public float dragScale = 1f;     // damaged skin, bombs, open canopy
		public float stallScale = 1f;    // thin air, wing damage, bombs
		public float turnScale = 1f;     // wing damage lowers the instant turn rate
		public float rollScale = 1f;     // aileron damage
		public float controlScale = 1f;  // elevator damage: pitch authority

		public Row Copy()
		{
			return (Row)MemberwiseClone();
		}
	}

	public const float G = 9.81f;
	// Turn drag constant. Calibrated so that the sustainedTurnRate values below cost about the spare thrust at corner speed.
	public const float TurnDragK = 0.022f;
	// Closed throttle: the windmilling propeller adds this share of the parasite drag.
	public const float IdleDrag = 0.5f;
	public const float MinThrustShare = 0.35f;
	// Best climb speed as a share of top speed, and the climb angle a zoom may use while speed lasts.
	public const float BestClimbShare = 0.6f;
	public const float ZoomClimbAngle = 55f;
	// Governor: throttle changes from closed to open over this share of top speed around the wanted speed.
	public const float ThrottleBand = 0.08f;
	public const float MaxLoadFactor = 9f;
	// A gun this close to the centre line (game units, local x) is a nose gun.
	public const float NoseGunX = 0.15f;
	// The AI opens fire out to this multiple of the gun convergence distance.
	public const float RangePastConverge = 1.2f;
	// Doctrine cruise altitudes in game units (metres x 0.1). Used when a fighter has not been seen cruising yet.
	public const float CruiseUS = 450f;
	public const float CruiseJapan = 420f;
	// ---- v0.28 altitude and compressibility
	// The engine never loses more than this share of its power to altitude.
	public const float MinAltPower = 0.35f;
	// How much of the real thin-air effect on the stall speed is used (0 = none, 1 = full). At 5000 m the full effect raises the stall speed by 29%.
	public const float AltStallShare = 0.5f;
	// Parasite drag grows near the structural dive speed: drag x (1 + CompressK * x^2), x = 0 at CompressStart * vne and 1 at vne.
	public const float CompressK = 4f;
	public const float CompressStart = 0.9f;

	// ------------------------------------------------------------------------------------------------------------------------------------
	// THE TABLE. Key = prefab name in lower case (the same key ModAirStats uses and the [AIRDMG] lines print).
	// Four helpers fill the long columns from a few numbers, then the line can be edited freely:
	//   Fighter(key, name, kmh, stallKmh, sust, inst, climb, wl, roll, critAlt, conv, nose, japanese)
	//       corner speed = 2.6 x stall, flaps -12% (Japanese Zero family -12%), dive limit 1.28 x top (US) or 1.15 x (Japan), roll fade 0.5 (US) / 0.25 (Japan)
	//   Bomber(key, name, kmh, stallKmh, sust, inst, climb, wl, roll, critAlt)      corner speed = 2.05 x stall, dive limit 1.12, no forward gun convergence
	//   Heavy(key, name, kmh, stallKmh, sust, inst, climb, wl, roll, critAlt)       corner speed = 1.8 x stall, dive limit 1.08, long fuel
	//   AddRow(...)  every column by hand (the 0.27 rows below use it)
	// sust / inst = sustained / instantaneous turn rate in degrees per second at corner speed, climb in m/s, wl in kg/m2, roll in degrees/s.
	// ------------------------------------------------------------------------------------------------------------------------------------

	// name, max, corner, stall, sustained, instant, climb, wingLoading, roll, flapBonus, flapLimit, flapDrag, converge, noseConverge, dive, rollFade
	private static void AddRow(string key, string name, float maxSpeed, float corner, float stall, float sustained, float instant, float climb, float wl, float roll, float flapBonus, float flapLimit, float flapDrag, float converge, float nose, float dive, float fade)
	{
		Row r = new Row();
		r.name = name;
		r.maxSpeed = maxSpeed;
		r.cornerSpeed = corner;
		r.stallSpeed = stall;
		r.sustainedTurnRate = sustained;
		r.instantTurnRate = instant;
		r.climbPower = climb;
		r.wingLoading = wl;
		r.rollRate = roll;
		r.flapBonus = flapBonus;
		r.flapSpeedLimit = flapLimit;
		r.flapDragMultiplier = flapDrag;
		r.convergeAt = converge;
		r.noseConvergeAt = nose;
		r.diveSpeed = dive;
		r.rollFade = fade;
		r.critAlt = 5000f;
		r.altLoss = 0.10f;
		r.fuelMinutes = 100f;   // v0.28.11: minutes at full power; each row sets its own below (Alt / Fuel)
		r.tuned = true;
		table[key] = r;
	}

	// altitude and fuel columns of a row that exists already
	private static void Alt(string key, float critAlt, float altLoss, float fuelMinutes)
	{
		Row r = table[key];
		r.critAlt = critAlt;
		r.altLoss = altLoss;
		r.fuelMinutes = fuelMinutes;
	}

	// v0.28.11: fuel column alone. Minutes of flight at full power from the internal tanks (no drop tank); cruise burns about a third of that.
	private static void Fuel(string key, float fuelMinutes)
	{
		table[key].fuelMinutes = fuelMinutes;
	}

	private static void Fighter(string key, string name, float kmh, float stallKmh, float sust, float inst, float climb, float wl, float roll, float critAlt, float conv, float nose, bool japanese)
	{
		float stall = stallKmh / 36f;
		float corner = stall * 2.6f;
		AddRow(key, name, kmh / 36f, corner, stall, sust, inst, climb, wl, roll, 0.12f, corner * 1.08f, 1.5f, conv, nose, japanese ? 1.15f : 1.28f, japanese ? 0.25f : 0.5f);
		Alt(key, critAlt, 0.10f, 26f);
	}

	private static void Bomber(string key, string name, float kmh, float stallKmh, float sust, float inst, float climb, float wl, float roll, float critAlt)
	{
		float stall = stallKmh / 36f;
		float corner = stall * 2.05f;
		AddRow(key, name, kmh / 36f, corner, stall, sust, inst, climb, wl, roll, 0.10f, corner * 1.05f, 1.5f, -1f, -1f, 1.12f, 0.5f);
		Alt(key, critAlt, 0.10f, 40f);
	}

	private static void Heavy(string key, string name, float kmh, float stallKmh, float sust, float inst, float climb, float wl, float roll, float critAlt)
	{
		float stall = stallKmh / 36f;
		float corner = stall * 1.8f;
		AddRow(key, name, kmh / 36f, corner, stall, sust, inst, climb, wl, roll, 0.08f, corner * 1.05f, 1.4f, -1f, -1f, 1.08f, 0.6f);
		Alt(key, critAlt, 0.08f, 90f);
	}

	// A copy of an existing row under another key (stock names and near-identical marks)
	private static void Like(string key, string likeKey, string name)
	{
		Row r = table[likeKey].Copy();
		r.name = name;
		table[key] = r;
	}

	// Row of the table (Build() helper that edits a row just added)
	private static Row Table(string key)
	{
		return table[key];
	}

	// Lookup
	// ------------------------------------------------------------------------------------------------------------------------------------

	// cacheMiss false is used while a prefab is being built (its name may not be final yet): a miss is then not remembered.
	private static Row Find(Unit unit, bool cacheMiss = true)
	{
		if (table == null)
		{
			Build();
		}
		if (unit == null)
		{
			return null;
		}
		int id = ((UnityEngine.Object)unit).GetInstanceID();
		Row row;
		if (tunedCache.TryGetValue(id, out row))
		{
			return row;
		}
		string key = DamageableAir.ModAirKey(unit);
		if (!table.TryGetValue(key, out row))
		{
			row = null;
		}
		// v0.28: a flight file in the aircraft's folder overwrites the table row (or the row made from the data file)
		Row fromFile = FileRow(key, row, unit.unitData);
		if (fromFile != null)
		{
			row = fromFile;
		}
		if (row != null || cacheMiss)
		{
			tunedCache[id] = row;
		}
		return row;
	}

	// unit/air/<key>/<key>_flight.txt, looked up in the mod (Override) folder first, then the game's own. JSON: any field of Row, e.g.
	// {"maxSpeed":14.5,"sustainedTurnRate":19,"critAlt":4200}. Fields that are left out keep their table value. Read once per aircraft kind.
	private static Row FileRow(string key, Row baseRow, UnitData data)
	{
		Row cached;
		if (fileRows.TryGetValue(key, out cached))
		{
			return cached;
		}
		Row result = null;
		try
		{
			string text = ReadFlightFile(key);
			if (text != null && (baseRow != null || data != null))
			{
				result = (baseRow != null) ? baseRow.Copy() : Default(data);
				JsonUtility.FromJsonOverwrite(text, result);
				result.tuned = true;
				if (fileLogged.Add(key))
				{
					Debug.Log((object)("[AIRFLIGHT] " + key + ": flight file read, row now " + JsonUtility.ToJson(result)));
				}
			}
		}
		catch (Exception ex)
		{
			Debug.LogError((object)("[AIRFLIGHT] flight file for " + key + " could not be read, using the table: " + ex.Message));
			result = null;
		}
		if (data != null || baseRow != null)
		{
			fileRows[key] = result;
		}
		return result;
	}

	private static string ReadFlightFile(string key)
	{
		if (AssetLoader.instance == null || AssetLoader.instance.fullPathLocations == null)
		{
			return null;
		}
		string[] places = AssetLoader.instance.fullPathLocations;
		for (int i = 0; i < places.Length; i++)
		{
			string path = places[i] + "unit/air/" + key + "/" + key + "_flight.txt";
			if (File.Exists(path))
			{
				return File.ReadAllText(path);
			}
		}
		return null;
	}

	// The pristine row for an aircraft: the hand-set one, or one made from its UnitData. A top speed of -1 means "the game's value".
	public static Row BaseOf(UnitAir ua)
	{
		int id = ((UnityEngine.Object)ua).GetInstanceID();
		Row row;
		if (baseCache.TryGetValue(id, out row))
		{
			return row;
		}
		Unit u = (ua.unitAI != null) ? ua.unitAI.parentUnit : null;
		row = Find(u);
		if (row == null)
		{
			row = Default(ua.unitData);
		}
		else if (row.maxSpeed <= 0f)
		{
			row = row.Copy();
			row.maxSpeed = ua.unitData.maxSpeed;
		}
		baseCache[id] = row;
		return row;
	}

	// The LIVE row for an aircraft: a private copy of its row whose scale fields ModAirDamage.Refresh keeps up to date (damage, altitude, bombs).
	public static Row Of(UnitAir ua)
	{
		int id = ((UnityEngine.Object)ua).GetInstanceID();
		Row row;
		if (airCache.TryGetValue(id, out row))
		{
			return row;
		}
		row = BaseOf(ua).Copy();
		airCache[id] = row;
		return row;
	}

	public static Row Of(Unit u)
	{
		if (u == null || u.unitAir == null)
		{
			return null;
		}
		return Of(u.unitAir);
	}

	// Row for a plane that has none: the old rules (fighters stall at 26% of top speed, others 38%, corner at 60%) expressed as a row.
	public static Row Default(UnitData d)
	{
		Row r = new Row();
		bool fighter = d.unitSubtype == UnitSubType.Fighter;
		bool dive = d.unitSubtype == UnitSubType.Dive_Bomber;
		bool heavy = d.unitSubtype == UnitSubType.Medium_Bomber || d.unitSubtype == UnitSubType.Heavy_Bomber;
		r.name = d.unitName;
		r.maxSpeed = d.maxSpeed;
		r.cornerSpeed = d.maxSpeed * 0.6f;
		r.stallSpeed = d.maxSpeed * (fighter ? 0.26f : 0.38f);
		r.sustainedTurnRate = d.maxTurnRate * 0.7f;
		r.instantTurnRate = d.maxTurnRate * 1.1f;
		r.climbPower = fighter ? 12f : (dive ? 8f : 6f);
		r.wingLoading = fighter ? 150f : (dive ? 180f : 220f);
		r.rollRate = fighter ? 90f : 45f;
		r.flapBonus = 0.1f;
		r.flapSpeedLimit = d.maxSpeed * 0.65f;
		r.flapDragMultiplier = 1.5f;
		r.convergeAt = -1f;
		r.noseConvergeAt = -1f;
		r.diveSpeed = fighter ? 1.25f : (dive ? 1.05f : 1.15f);
		r.rollFade = 0.5f;
		r.critAlt = fighter ? 5500f : 4500f;
		r.altLoss = 0.10f;
		r.fuelMinutes = fighter ? 100f : (heavy ? 240f : 180f);   // v0.28.11: realistic endurance at full power
		r.tuned = false;
		return r;
	}

	// ---------------- pure formulas ----------------

	// Stall speed in game units. A skid (rudder used to force a turn) raises it, handled by the caller.
	public static float StallSpeed(Row r, bool flaps)
	{
		return r.stallSpeed * r.stallScale * (flaps ? (1f - r.flapBonus) : 1f);
	}

	// Highest turn rate in degrees per second the plane can pull at this speed (game units). See the header for the formula.
	public static float TurnLimit(Row r, float speed, bool flaps)
	{
		float c = r.cornerSpeed * 10f;
		float vs0 = r.stallSpeed * r.stallScale * 10f;
		float vs = vs0 * (flaps ? (1f - r.flapBonus) : 1f);
		float nMax = Mathf.Clamp((c / vs0) * (c / vs0), 2f, MaxLoadFactor);
		float reference = G / c * Mathf.Sqrt(nMax * nMax - 1f);
		float v = speed * 10f;
		if (v <= vs)
		{
			return 0f;
		}
		float n = Mathf.Min(nMax, (v / vs) * (v / vs));
		if (n <= 1.0001f)
		{
			return 0f;
		}
		float omega = G / v * Mathf.Sqrt(n * n - 1f);
		return r.instantTurnRate * r.turnScale * omega / reference;
	}

	// Engine power in square metres per second cubed, from the climb power. Not scaled: drag is worked out from this too.
	public static float Power(Row r)
	{
		return G * r.climbPower / (1f - BestClimbShare * BestClimbShare * BestClimbShare);
	}

	// Full-throttle thrust acceleration in metres per second squared (thrustScale: engine damage, altitude, bombs).
	public static float ThrustAvail(Row r, float speed)
	{
		float v = Mathf.Max(speed * 10f, MinThrustShare * r.maxSpeed * 10f);
		return Power(r) * r.thrustScale / v;
	}

	// Share of the engine power left at a height (game units): 1 up to the critical altitude, then altLoss per 1000 m.
	public static float AltitudePower(Row r, float height)
	{
		float over = height * 10f - r.critAlt;
		if (over <= 0f)
		{
			return 1f;
		}
		return Mathf.Max(MinAltPower, 1f - r.altLoss * over / 1000f);
	}

	// Stall speed multiplier from thin air at a height (game units). Density from the standard atmosphere, only AltStallShare of the effect is used.
	public static float AltitudeStall(float height)
	{
		float h = Mathf.Clamp(height * 10f, 0f, 20000f);
		float density = Mathf.Pow(1f - 2.2558e-5f * h, 4.2559f);
		float full = 1f / Mathf.Sqrt(Mathf.Max(0.2f, density));
		return 1f + AltStallShare * (full - 1f);
	}

	// Parasite drag deceleration in metres per second squared. Rises steeply near the structural dive speed (compressibility), see CompressK.
	public static float Drag(Row r, float speed, bool flaps)
	{
		float v = speed * 10f;
		float vm = r.maxSpeed * 10f;
		float d = Power(r) * v * v / (vm * vm * vm) * (flaps ? r.flapDragMultiplier : 1f) * r.dragScale;
		float vne = r.diveSpeed * r.maxSpeed * 10f;
		float x = (v - CompressStart * vne) / Mathf.Max(1f, vne * (1f - CompressStart));
		if (x > 0f)
		{
			d *= 1f + CompressK * x * x;
		}
		return d;
	}

	// Drag from turning, metres per second squared: (turn rate / max turn rate)^2 * wing loading * constant.
	public static float TurnDrag(Row r, float turnDeg, bool flaps)
	{
		float ratio = Mathf.Clamp(turnDeg / Mathf.Max(r.instantTurnRate * r.turnScale, 1f), 0f, 1.5f);
		return ratio * ratio * r.wingLoading * TurnDragK;
	}

	// Throttle (0 to 1) a pilot uses to hold wantedSpeed: the throttle that balances drag, turn drag and gravity, plus a correction for the speed error.
	// sinPitch is positive nose down (gravity then helps).
	public static float Throttle(Row r, float speed, float wantedSpeed, float sinPitch, float turnDeg, bool flaps)
	{
		float avail = ThrustAvail(r, speed);
		float need = Drag(r, speed, flaps) + TurnDrag(r, turnDeg, flaps) - G * sinPitch;
		float trim = need / avail;
		float t = trim + (wantedSpeed - speed) / (ThrottleBand * r.maxSpeed);
		return Mathf.Clamp01(t);
	}

	// Speed change in game units per second squared from thrust and drag only (gravity is added by the caller from the nose angle).
	public static float Accel(Row r, float speed, float throttle, bool flaps, float turnDeg, float skid, float skidDragShare)
	{
		float thrust = throttle * ThrustAvail(r, speed);
		float drag = Drag(r, speed, flaps) * (1f + IdleDrag * (1f - throttle));
		float turn = TurnDrag(r, turnDeg, flaps) * (1f + skidDragShare * skid);
		return (thrust - drag - turn) / 10f;
	}

	// Turn rate (degrees per second) that spare thrust at this speed can pay for. The AI uses the lower of this and the table's sustained rate.
	public static float NeutralTurn(Row r, float speed, bool flaps)
	{
		float excess = ThrustAvail(r, speed) - Drag(r, speed, flaps);
		if (excess <= 0f)
		{
			return 0f;
		}
		float ratio = Mathf.Sqrt(excess / (r.wingLoading * TurnDragK));
		return Mathf.Min(r.instantTurnRate * ratio, TurnLimit(r, speed, flaps));
	}

	public static float SustainedTurn(Row r, float speed, bool flaps)
	{
		return Mathf.Min(r.sustainedTurnRate * r.turnScale * (flaps ? 1.1f : 1f), NeutralTurn(r, speed, flaps));
	}

	// How far the nose may be pulled up (degrees). At normal speed the power-limited angle (excess thrust over gravity); with speed to spare a zoom
	// climb may be steeper, tapering back to the power limit as the speed is used up, and to zero as the plane nears its stall.
	public static float ClimbLimit(Row r, float speed, bool flaps)
	{
		float v = speed * 10f;
		float vm = r.maxSpeed * 10f;
		float vs = StallSpeed(r, flaps) * 10f;
		if (v < vs * 1.1f)
		{
			return 0f;
		}
		float excess = ThrustAvail(r, speed) - Drag(r, speed, flaps);
		float sustained = Mathf.Asin(Mathf.Clamp(excess / G, 0f, 1f)) * Mathf.Rad2Deg;
		float vClimb = Mathf.Max(1.3f * vs, BestClimbShare * vm);
		float k = Mathf.Clamp01((v - vClimb) / (0.25f * vm));
		float limit = Mathf.Lerp(sustained, ZoomClimbAngle, k);
		limit *= Mathf.Clamp01((v - 1.1f * vs) / (0.2f * vs));
		return limit;
	}

	// 0.5 to 1: how much of its full pitch and roll authority the plane has at this speed (it is the turn-limit curve, normalised to corner speed).
	// Elevator damage (controlScale) lowers it.
	public static float Authority(Row r, float speed, bool flaps)
	{
		return Mathf.Clamp(TurnLimit(r, speed, flaps) / Mathf.Max(r.instantTurnRate, 1f), 0.5f, 1f) * r.controlScale;
	}

	// Roll rate in degrees per second at this speed: full at low speed, fading to rollFade of it at the dive speed limit (stiff ailerons).
	public static float RollRate(Row r, float speed)
	{
		float vne = r.diveSpeed * r.maxSpeed;
		float start = 0.55f * r.maxSpeed;
		float t = Mathf.Clamp01((speed - start) / Mathf.Max(0.1f, vne - start));
		return r.rollRate * r.rollScale * Mathf.Lerp(1f, r.rollFade, t);
	}

	// The structural speed limit in game units (Vne).
	public static float Vne(Row r)
	{
		return r.diveSpeed * r.maxSpeed;
	}

	// Specific energy in metres for a height and speed in game units.
	public static float Energy(float height, float speed)
	{
		float v = speed * 10f;
		return height * 10f + v * v / (2f * G);
	}

	// ---------------- gun convergence (PrefabBuilder) ----------------
	// Replaces weaponParticle.convergeAt where the particle systems are aimed: wing guns converge at convergeAt, centre-line guns at noseConvergeAt.
	// localX is the gun's local x position. Planes without a row keep the data file's value.
	public static float Converge(Unit u, float stock, float localX)
	{
		Row r = Find(u, false);
		if (r == null)
		{
			return stock;
		}
		float c = (Mathf.Abs(localX) < NoseGunX) ? r.noseConvergeAt : r.convergeAt;
		return (c > 0f) ? c : stock;
	}

	// Distance in game units out to which the AI opens fire: a little past where the guns cross, so a burst is pulled where it does damage.
	public static float FireRange(Unit u, float fallback)
	{
		Row r = Find(u);
		if (r == null || r.convergeAt <= 0f)
		{
			return fallback;
		}
		return r.convergeAt * 0.1f * RangePastConverge;
	}

	// Doctrine cruise altitude in game units.
	public static float Cruise(Unit u)
	{
		return (u != null && u.unitData != null && u.unitData.nationIndex == 3) ? CruiseJapan : CruiseUS;
	}

	// For the readout and the auto-resolve: does this key have a hand-set row?
	public static bool HasRow(string key)
	{
		if (table == null)
		{
			Build();
		}
		return table.ContainsKey(key);
	}

	// The table row for a prefab key (null if none). Used by ModAirResolve to rate aircraft that are not in a battle.
	public static Row ForKey(string key)
	{
		if (table == null)
		{
			Build();
		}
		Row r;
		return table.TryGetValue(key, out r) ? r : null;
	}

	// ---------------- v0.28.5: chase power ----------------
	// A fighter that chases a plane it cannot gain on (same top speed, target flying away) used to trail it for ever: wantedSpeed was above the top speed but
	// the thrust model holds the plane to its top speed. Now a chaser may run its engine at emergency power for a while: ChaseBoost x thrust (about +10% top speed),
	// ChaseBudget seconds, recovering at ChaseRecover seconds per second while not used, and burning fuel faster. The AI asks for it every tick it wants it
	// (ChaseRequest, a request lasts 0.6 s, like the flaps), ModFlight.End asks for the factor (ChaseFactor).
	public const float ChaseBoost = 1.35f;
	public const float ChaseBudget = 240f;
	public const float ChaseRecover = 0.15f;
	public const float ChaseFuel = 1.8f;       // fuel burn: the throttle given to ModAirDamage.Tick is multiplied by this while boosting
	public static int chaseStarted;
	private static int chaseLogged;
	private static Dictionary<int, float> chaseUntil = new Dictionary<int, float>();
	private static Dictionary<int, float> chaseSpent = new Dictionary<int, float>();
	private static HashSet<int> chaseOn = new HashSet<int>();
	private static HashSet<int> chaseLocked = new HashSet<int>();

	public static void ChaseRequest(UnitAir ua)
	{
		if (ua != null)
		{
			chaseUntil[((UnityEngine.Object)ua).GetInstanceID()] = Time.time + 0.6f;
		}
	}

	public static bool Chasing(Unit u)
	{
		return u != null && u.unitAir != null && chaseOn.Contains(((UnityEngine.Object)u.unitAir).GetInstanceID());
	}

	// seconds of real time since the last tick (the game's duration is multiplied by the time compression twice)
	public static float ChaseFactor(UnitAir ua, float seconds)
	{
		int id = ((UnityEngine.Object)ua).GetInstanceID();
		float until;
		bool want = chaseUntil.TryGetValue(id, out until) && Time.time <= until;
		float spent;
		chaseSpent.TryGetValue(id, out spent);
		if (chaseLocked.Contains(id) && spent < ChaseBudget * 0.5f)
		{
			chaseLocked.Remove(id);
		}
		if (want && !chaseLocked.Contains(id) && spent < ChaseBudget)
		{
			spent += seconds;
			chaseSpent[id] = spent;
			if (spent >= ChaseBudget)
			{
				chaseLocked.Add(id);
			}
			if (chaseOn.Add(id))
			{
				chaseStarted++;
				if (chaseLogged < 40)
				{
					chaseLogged++;
					Debug.Log((object)("[CHASE] " + ua.unitData.unitName + " id " + id + " (" + ((ua.faction == Faction.Player) ? "ours" : "enemy") + ") chasing a plane it cannot gain on: emergency power on (" + ((ChaseBudget - spent)).ToString("F0") + " s left, +" + ((Mathf.Pow(ChaseBoost, 1f / 3f) - 1f) * 100f).ToString("F0") + "% top speed)"));
				}
			}
			return ChaseBoost;
		}
		if (chaseOn.Remove(id) && chaseLogged < 40 && spent >= ChaseBudget)
		{
			chaseLogged++;
			Debug.Log((object)("[CHASE] " + ua.unitData.unitName + " id " + id + " (" + ((ua.faction == Faction.Player) ? "ours" : "enemy") + ") emergency power used up, back to normal power until it has rested"));
		}
		if (spent > 0f)
		{
			chaseSpent[id] = Mathf.Max(0f, spent - seconds * ChaseRecover);
		}
		return 1f;
	}
}
