using System;
using System.Collections.Generic;
using UnityEngine;

// v0.26.1 flight core. The game flies planes by sliding them along their nose at a speed that never depends on pitch. This adds, on top of
// the game's own movement and without replacing it:
//   Energy:      a dive gains speed, a climb loses it (gravity along the nose). In a dive the game's pull back to the wanted speed is eased.
//   Dive limit:  a plane cannot dive faster than a limit per type (fighters 125% of top speed, dive bombers 105% because of dive brakes,
//                other bombers 115%).
//   Stall:       below a minimum speed (35% of top speed for fighters, 45% for others) a plane cannot hold its climb. It lets the nose drop
//                until it has flying speed again.
//   Turn limit:  the turn rate is capped by speed. Above corner speed (heavy controls) and below it (nearly stalled) turns are slower.
// v0.26.2 adds pitch control: while a fighter lines up a shot, the nose is pulled to the angle it needs at a real pull rate
// (the game only allowed 10 degrees per second, so a fighter could never get its nose onto a target in time).
// v0.26.3 adds banking: a plane rolls into a turn at a real roll rate (fighters 90 degrees per second, others 45) and the turn only builds as
// fast as the bank allows, so turns start and finish smoothly instead of the nose swinging the moment the AI asks.
// v0.26.4 adds pilot rank (Green, Trained, Veteran, Elite; crews off = everyone Elite) and acrobatics the game never had: a roll-over attack
// (roll inverted and dive straight down on a plane below), a split-S escape and a defensive barrel roll. During those the class turns the plane
// itself (UnitAir.ApplyUnitMovement hands over the rotation), then gives the plane back to the game upright.
// Always on: this is the game's flying now, not an option. Every number is below and can be changed, then Compile and Save Module.
public static class ModFlight
{
	// Not a player switch. Only set if the core throws an error, so one bug cannot spam the log or break a battle (the plane then flies the original way).
	private static bool failed = false;

	// Gravity along the nose in game units per second squared (9.81 m/s2, 10 m per game unit). EnergyScale makes it stronger or weaker.
	public const float Gravity = 0.981f;
	public const float EnergyScale = 2f;

	// Share of the game's pull back toward the wanted speed that is given back while diving (0 to 1).
	public const float DiveRefund = 0.8f;

	// Turn rate factor limits (share of maxTurnRate).
	public const float CornerShare = 0.6f;
	public const float MinTurnFactor = 0.5f;
	public const float MinFastTurnFactor = 0.55f;

	private static Dictionary<int, float> startSpeed = new Dictionary<int, float>();
	private static Dictionary<int, float> lastStallLog = new Dictionary<int, float>();
	private static Dictionary<int, float> aimPitch = new Dictionary<int, float>();
	private static Dictionary<int, float> aimUntil = new Dictionary<int, float>();
	// v0.26.3: the turn the pilot is easing toward, so the AI's sudden left/right wishes become smooth stick movements.
	private static Dictionary<int, float> smoothTurn = new Dictionary<int, float>();
	public const float StickTime = 0.35f;
	private static HashSet<int> stalled = new HashSet<int>();

	// Pull rates in degrees per second.
	public const float FighterPitchRate = 40f;
	public const float OtherPitchRate = 20f;
	public const float FighterEasyRate = 25f;
	public const float OtherEasyRate = 12f;
	public const float MaxAimDown = 60f;
	public const float MaxAimUp = 50f;
	// v0.26.3 banking. Roll rate in degrees per second. The game's bank is 5 degrees per degree-per-second of turn rate, kept here.
	public const float FighterRollRate = 90f;
	public const float OtherRollRate = 45f;
	public const float BankPerTurn = 5f;
	public const float MaxBank = 80f;
	// v0.26.4 acrobatics (degrees per second, seconds, units of 10 m)
	// Slow and heavy like a real fighter at speed: a half roll takes about 2.5 seconds and a full split-S about 6 seconds.
	public const float RollOverRate = 70f;
	public const float PullRate = 30f;
	public const float BarrelTime = 3.5f;
	public const float PullOutHeight = 45f;
	public const float AcroMinHeight = 100f;
	public const float BulletTravelSpeed = 80f;

	private class AcroState
	{
		public int kind;      // 1 roll-over attack, 2 split-S, 3 barrel roll
		public int phase;     // 0 rolling inverted, 1 pulling, 2 pulling out
		public float t;
		public float x;       // nose angle, degrees, positive = down (past 90 the plane is going back the way it came)
		public float yaw;
		public float hold;    // seconds spent lined up on the target
		public float z;       // roll, degrees (barrel roll: how far round)
		public Unit target;
	}

	private static Dictionary<int, AcroState> acro = new Dictionary<int, AcroState>();
	private static Dictionary<int, int> ranks = new Dictionary<int, int>();
	// What the game's pitch and roll should read after an acrobatic tick (UnitAir copies these into currentPitch / currentRoll).
	public static float AcroPitch = 0f;
	public static float AcroRoll = 0f;

	private static bool Applies(UnitAir ua)
	{
		if (failed || ua == null || ua.unitData == null || ua.isDestroyed || ua.shootDown || ua.isParked)
		{
			return false;
		}
		if (ua.unitAI == null)
		{
			return false;
		}
		Maneuvre m = ua.unitAI.currentManeuvre;
		return m == Maneuvre.None || m == Maneuvre.Break_Left || m == Maneuvre.Break_Right;
	}

	// Call as the first line of UnitAir.UpdateMovementData.
	public static void Begin(UnitAir ua)
	{
		if (failed)
		{
			return;
		}
		startSpeed[((UnityEngine.Object)ua).GetInstanceID()] = ua.currentSpeed;
	}

	// Call as the last line of UnitAir.UpdateMovementData.
	public static void End(UnitAir ua, float duration)
	{
		try
		{
			if (duration <= 0f || !Applies(ua))
			{
				return;
			}
			int id = ((UnityEngine.Object)ua).GetInstanceID();
			float before;
			if (!startSpeed.TryGetValue(id, out before))
			{
				return;
			}
			UnitData d = ua.unitData;
			float maxSpeed = d.maxSpeed;
			if (maxSpeed <= 0f)
			{
				return;
			}
			bool fighter = d.unitSubtype == UnitSubType.Fighter;

			// Nose angle in degrees, positive = nose down.
			float pitch = ((Component)ua).transform.eulerAngles.x;
			if (pitch > 180f)
			{
				pitch -= 360f;
			}
			float sinPitch = Mathf.Sin(pitch * ((float)Math.PI / 180f));

			float speed = ua.currentSpeed;

			// Energy. In a dive give back part of what the game took off to pull the plane back to its wanted speed.
			if (sinPitch > 0.1f && speed < before)
			{
				speed += (before - speed) * DiveRefund * Mathf.Clamp01(sinPitch * 2f);
			}
			speed += Gravity * EnergyScale * sinPitch * duration;

			// Dive limit.
			float limit = 1.15f;
			if (fighter)
			{
				limit = 1.25f;
			}
			else if (d.unitSubtype == UnitSubType.Dive_Bomber)
			{
				limit = 1.05f;
			}
			float diveSpeed = maxSpeed * limit;
			if (speed > diveSpeed)
			{
				speed = diveSpeed;
			}

			// Stall: keep a little above stall speed once recovered, so the nose does not flicker.
			float stall = maxSpeed * (fighter ? 0.35f : 0.45f);
			if (speed >= stall * 1.3f)
			{
				stalled.Remove(id);
			}
			if (speed < stall * 1.1f && ((Component)ua).transform.position.y > 15f)
			{
				stalled.Add(id);
				float y = ((Component)ua).transform.position.y;
				if (ua.wantedHeight > y - 10f)
				{
					ua.wantedHeight = y - 10f;
				}
				float last;
				if (!lastStallLog.TryGetValue(id, out last) || Time.time - last > 5f)
				{
					lastStallLog[id] = Time.time;
					Debug.Log((object)("[FLIGHT] " + d.unitName + " id " + id + " stalling at speed " + speed.ToString("F1") + " (stall " + stall.ToString("F1") + "): nose dropping"));
				}
			}
			if (speed < 0f)
			{
				speed = 0f;
			}
			ua.currentSpeed = speed;
			ua.percentMaxSpeed = speed / maxSpeed * ua.currentSpeedPenalty;

			// Turn limit by speed.
			float corner = maxSpeed * CornerShare;
			float factor = 1f;
			if (speed > corner)
			{
				factor = Mathf.Clamp(corner / speed, MinFastTurnFactor, 1f);
			}
			else if (speed < corner)
			{
				float t = Mathf.Clamp01((speed - stall) / Mathf.Max(0.01f, corner - stall));
				factor = Mathf.Lerp(MinTurnFactor, 1f, t);
			}
			float cap = d.maxTurnRate * factor;
			if (ua.currentTurnRate > cap)
			{
				ua.currentTurnRate = cap;
			}
			else if (ua.currentTurnRate < -cap)
			{
				ua.currentTurnRate = -cap;
			}
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] flight core failed and stopped: " + ex.Message));
		}
	}

	// Called every tick by ModDogfight.GunRun: pitchDown is the nose angle that points at the aim point, in degrees, positive = nose down.
	public static void Aim(UnitAir ua, float pitchDown)
	{
		if (failed || ua == null)
		{
			return;
		}
		int id = ((UnityEngine.Object)ua).GetInstanceID();
		aimPitch[id] = pitchDown;
		aimUntil[id] = Time.time + 0.5f;
	}

	// Text for the flight log: the pitch the pilot is aiming for, or nothing.
	public static string AimText(UnitAir ua)
	{
		int id = ((UnityEngine.Object)ua).GetInstanceID();
		float until;
		float p;
		string acroText = "";
		AcroState a;
		if (acro.TryGetValue(id, out a))
		{
			acroText = " acro " + ((a.kind == 1) ? "roll-over-attack" : ((a.kind == 2) ? "split-S" : "barrel-roll"));
		}
		string pilot = "";
		int r;
		if (ranks.TryGetValue(id, out r))
		{
			pilot = " pilot " + RankName(r);
		}
		if (aimUntil.TryGetValue(id, out until) && Time.time <= until && aimPitch.TryGetValue(id, out p))
		{
			return " aim " + p.ToString("F1") + acroText + pilot;
		}
		return acroText + pilot;
	}

	// Replaces the pitch line of UnitAir.ApplyFlightControls. current = the plane's pitch now, heightGap = the game's own wish (positive = nose down).
	public static float Pitch(UnitAir ua, float current, float heightGap, float duration)
	{
		try
		{
			if (failed || ua.unitData == null)
			{
				return Mathf.SmoothDampAngle(current, heightGap, ref ua.pitchVelocity, 1f, 10f, duration);
			}
			int id = ((UnityEngine.Object)ua).GetInstanceID();
			bool fighter = ua.unitData.unitSubtype == UnitSubType.Fighter;
			float until;
			float wanted;
			float y = ((Component)ua).transform.position.y;
			if (aimUntil.TryGetValue(id, out until) && Time.time <= until && aimPitch.TryGetValue(id, out wanted) && Applies(ua))
			{
				// Do not fly into the sea: the lower the plane, the less it may dive.
				float maxDown = MaxAimDown;
				if (y < 50f)
				{
					maxDown = Mathf.Clamp((y - 10f) * 1.5f, 0f, MaxAimDown);
				}
				float target = Mathf.Clamp(wanted, -MaxAimUp, maxDown);
				if (y < 12f)
				{
					target = Mathf.Min(target, -5f);
				}
				if (stalled.Contains(id))
				{
					target = Mathf.Max(target, 12f);
				}
				ua.pitchVelocity = 0f;
				return Mathf.MoveTowards(current, target, (fighter ? FighterPitchRate : OtherPitchRate) * duration);
			}
			return Mathf.SmoothDampAngle(current, heightGap, ref ua.pitchVelocity, 1f, fighter ? FighterEasyRate : OtherEasyRate, duration);
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] pitch control failed and stopped: " + ex.Message));
			return Mathf.SmoothDampAngle(current, heightGap, ref ua.pitchVelocity, 1f, 10f, duration);
		}
	}

	// v0.26.3. Replaces the roll line of UnitAir.ApplyFlightControls. Rolls toward the bank the wanted turn needs, at a real roll rate.
	public static float Roll(UnitAir ua, float currentRoll, float currentTurnRate, float wantedTurnRate, float duration)
	{
		try
		{
			if (failed || ua.unitData == null)
			{
				return Mathf.MoveTowards(currentRoll, currentTurnRate * -BankPerTurn, 40f * duration);
			}
			bool fighter = ua.unitData.unitSubtype == UnitSubType.Fighter;
			float eased = wantedTurnRate;
			if (!smoothTurn.TryGetValue(((UnityEngine.Object)ua).GetInstanceID(), out eased))
			{
				eased = wantedTurnRate;
			}
			float target = Mathf.Clamp(eased * -BankPerTurn, -MaxBank, MaxBank);
			return Mathf.MoveTowards(currentRoll, target, (fighter ? FighterRollRate : OtherRollRate) * duration);
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] roll control failed and stopped: " + ex.Message));
			return Mathf.MoveTowards(currentRoll, currentTurnRate * -BankPerTurn, 40f * duration);
		}
	}

	// v0.26.3. Replaces the turn rate line of UnitAir.ApplyFlightControls. The turn rate still moves toward the wish at the plane's own rate,
	// but cannot be faster than the current bank supports (plus 1 degree per second so a level plane can start rolling into a turn).
	public static float TurnRate(UnitAir ua, float current, float wanted, float currentRoll, float duration)
	{
		float next = Mathf.MoveTowards(current, wanted, duration * ua.unitData.turnRate * Time.timeScale);
		try
		{
			if (failed)
			{
				return next;
			}
			int id = ((UnityEngine.Object)ua).GetInstanceID();
			float eased;
			if (!smoothTurn.TryGetValue(id, out eased))
			{
				eased = current;
			}
			eased = Mathf.Lerp(eased, wanted, 1f - Mathf.Exp(-duration / StickTime));
			smoothTurn[id] = eased;
			next = Mathf.MoveTowards(current, eased, duration * ua.unitData.turnRate * Time.timeScale);
			float supported = -currentRoll / BankPerTurn;
			float sign = Mathf.Sign(next);
			float allowed = 1f;
			if (sign * supported > 0f)
			{
				allowed = Mathf.Abs(supported) + 1f;
			}
			if (Mathf.Abs(next) > allowed && Mathf.Abs(next) > Mathf.Abs(current))
			{
				next = sign * allowed;
			}
			return next;
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] bank control failed and stopped: " + ex.Message));
			return next;
		}
	}

	// ---------------- v0.26.4 pilot rank ----------------
	// 0 Green, 1 Trained, 2 Veteran, 3 Elite. Crews off (Mod Options) = every pilot is Elite so every manoeuvre can be tested and planes are never
	// locked into simple attacks. Until air crew experience exists the rank comes from nation and year, fixed for each plane by its id.
	public static int Rank(Unit u)
	{
		try
		{
			if (u == null || u.unitData == null)
			{
				return 3;
			}
			if (!ModOptions.Crew)
			{
				return 3;
			}
			int id = ((UnityEngine.Object)u).GetInstanceID();
			int r;
			if (ranks.TryGetValue(id, out r))
			{
				return r;
			}
			int year = 1942;
			if (CampaignInterface.instance != null && CampaignInterface.instance.dateTime.Year > 1900)
			{
				year = CampaignInterface.instance.dateTime.Year;
			}
			// chance in percent of Green, Trained, Veteran (Elite is the rest)
			int g;
			int t;
			int v;
			if (u.unitData.nationIndex == 3)
			{
				if (year <= 1942) { g = 5; t = 25; v = 45; }
				else if (year == 1943) { g = 15; t = 40; v = 35; }
				else { g = 40; t = 40; v = 17; }
			}
			else
			{
				if (year <= 1942) { g = 15; t = 45; v = 30; }
				else if (year == 1943) { g = 10; t = 35; v = 40; }
				else { g = 5; t = 25; v = 45; }
			}
			int roll = new System.Random(id).Next(0, 100);
			r = 3;
			if (roll < g) { r = 0; }
			else if (roll < g + t) { r = 1; }
			else if (roll < g + t + v) { r = 2; }
			ranks[id] = r;
			Debug.Log((object)("[FLIGHT] " + u.unitData.unitName + " id " + id + " pilot is " + RankName(r) + " (nation " + u.unitData.nationIndex + ", " + year + ")"));
			return r;
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] pilot rank failed and stopped: " + ex.Message));
			return 3;
		}
	}

	public static string RankName(int rank)
	{
		switch (rank)
		{
		case 0: return "Green";
		case 1: return "Trained";
		case 2: return "Veteran";
		default: return "Elite";
		}
	}

	// How well the pilot leads his shots: 1 = perfect.
	public static float LeadScale(Unit u)
	{
		int r = Rank(u);
		if (r == 0)
		{
			return 0.6f;
		}
		if (r == 1)
		{
			return 0.85f;
		}
		return 1f;
	}

	// ---------------- v0.26.4 acrobatics ----------------
	public static bool AcroActive(UnitAir ua)
	{
		return ua != null && acro.ContainsKey(((UnityEngine.Object)ua).GetInstanceID());
	}

	public static int AcroKind(UnitAir ua)
	{
		AcroState a;
		if (ua != null && acro.TryGetValue(((UnityEngine.Object)ua).GetInstanceID(), out a))
		{
			return a.kind;
		}
		return 0;
	}

	private static AcroState NewAcro(UnitAir ua, int kind)
	{
		AcroState a = new AcroState();
		a.kind = kind;
		Vector3 e = ((Component)ua).transform.eulerAngles;
		a.x = Wrap(e.x);
		a.yaw = e.y;
		a.z = Wrap(e.z);
		acro[((UnityEngine.Object)ua).GetInstanceID()] = a;
		return a;
	}

	private static float Wrap(float angle)
	{
		while (angle > 180f)
		{
			angle -= 360f;
		}
		while (angle < -180f)
		{
			angle += 360f;
		}
		return angle;
	}

	// Roll over and dive down on a plane below. False if the plane cannot do it now (already in a manoeuvre, too low).
	public static bool StartRollAttack(UnitAir ua, Unit target)
	{
		try
		{
			if (!Applies(ua) || target == null || AcroActive(ua) || ((Component)ua).transform.position.y < AcroMinHeight)
			{
				return false;
			}
			AcroState a = NewAcro(ua, 1);
			a.target = target;
						Debug.Log((object)("[FLIGHT] " + ua.unitData.unitName + " id " + ((UnityEngine.Object)ua).GetInstanceID() + " roll-over attack: rolling inverted at " + (((Component)ua).transform.position.y * 10f).ToString("F0") + " m"));
			return true;
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] roll-over attack failed and stopped: " + ex.Message));
			return false;
		}
	}

	// Roll inverted and pull through into a dive away (split-S). Needs height.
	public static bool StartSplitS(UnitAir ua)
	{
		try
		{
			if (!Applies(ua) || AcroActive(ua) || ((Component)ua).transform.position.y < 110f)
			{
				return false;
			}
			NewAcro(ua, 2);
			Debug.Log((object)("[FLIGHT] " + ua.unitData.unitName + " id " + ((UnityEngine.Object)ua).GetInstanceID() + " split-S at " + (((Component)ua).transform.position.y * 10f).ToString("F0") + " m"));
			return true;
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] split-S failed and stopped: " + ex.Message));
			return false;
		}
	}

	// A quick full roll while turning. Does nothing if an acrobatic move is already running.
	public static void StartBarrelRoll(UnitAir ua)
	{
		try
		{
			if (!Applies(ua) || AcroActive(ua))
			{
				return;
			}
			AcroState a = NewAcro(ua, 3);
			a.z = 0f;
			if (ua.currentTurnRate < 0f)
			{
				a.z = 0.001f;
			}
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] barrel roll failed and stopped: " + ex.Message));
		}
	}

	// Called from UnitAir.ApplyUnitMovement in place of the game's rotation. pitch and roll are the game's own currentPitch and currentRoll.
	// Returns true when it turned the plane this tick (the game's rotation is then skipped); the caller copies AcroPitch and AcroRoll back.
	public static bool Acro(UnitAir ua, float dur, float pitch, float roll)
	{
		try
		{
			if (failed || ua == null)
			{
				return false;
			}
			int id = ((UnityEngine.Object)ua).GetInstanceID();
			AcroState a;
			if (!acro.TryGetValue(id, out a))
			{
				return false;
			}
			if (!Applies(ua))
			{
				acro.Remove(id);
				return false;
			}
			Transform tr = ((Component)ua).transform;
			float y = tr.position.y;
			a.t += dur;
			AcroPitch = pitch;
			AcroRoll = roll;

			if (a.kind == 3)
			{
				// Barrel roll: the normal flight with an extra turn round the nose.
				float sign = (ua.currentTurnRate < 0f) ? -1f : 1f;
				a.z += 360f / BarrelTime * dur;
				if (a.z >= 360f)
				{
					acro.Remove(id);
					return false;
				}
				Vector3 e = tr.eulerAngles;
				e.x = pitch;
				e.y += ua.currentTurnRate * dur;
				e.z = roll - sign * a.z;
				tr.eulerAngles = e;
				return true;
			}

			if (y < 35f && a.phase < 2)
			{
				a.phase = 2;
			}
			if (a.phase == 0)
			{
				a.z = Mathf.MoveTowards(a.z, 180f, RollOverRate * dur);
				if (a.z >= 179.9f)
				{
					a.phase = 1;
				}
			}
			if (a.phase == 1)
			{
				float goal = 180f;
				if (a.kind == 1)
				{
					// Half roll, then pull through the bottom of a loop. Past 90 degrees the plane is upright again, diving on the
					// target on the way back (heading is the reverse of where it started). Follow the target down to the end.
					Unit t = a.target;
					if (t == null || t.isDestroyed || y < PullOutHeight)
					{
						a.phase = 2;
					}
					else
					{
						Vector3 toNow = t.transform.position - tr.position;
						float dist = toNow.magnitude;
						Vector3 to = t.transform.position + t.transform.forward * t.currentActualSpeed * (dist / BulletTravelSpeed) - tr.position;
						float flat = new Vector2(to.x, to.z).magnitude;
						float down = Mathf.Atan2(-to.y, Mathf.Max(flat, 1f)) * 57.29578f;
						goal = 180f - Mathf.Clamp(down, 10f, 70f);
						if (a.x > 90f)
						{
							a.yaw = Mathf.MoveTowardsAngle(a.yaw, Mathf.Atan2(to.x, to.z) * 57.29578f + 180f, 30f * dur);
						}
						// finished: lined up for 4 seconds, passed the target, or too close
						if (a.x >= goal - 2f)
						{
							a.hold += dur;
							if (a.hold > 4f || dist < 12f)
							{
								a.phase = 2;
							}
						}
						else if (a.t > 14f)
						{
							a.phase = 2;
						}
					}
				}
				if (a.phase == 1)
				{
					a.x = Mathf.MoveTowards(a.x, goal, PullRate * dur);
				}
			}
			if (a.phase == 2)
			{
				a.x = Mathf.MoveTowards(a.x, 180f, PullRate * dur);
				if (a.x >= 179.5f)
				{
					// Finished upright and level, flying back the way it came.
					tr.rotation = Quaternion.Euler(0f, a.yaw + 180f, 0f);
					AcroPitch = 0f;
					AcroRoll = 0f;
					ua.currentTurnRate = 0f;
					ua.wantedTurnRate = 0f;
					ua.wantedHeight = y;
					ua.pitchVelocity = 0f;
					acro.Remove(id);
					return true;
				}
			}
			tr.rotation = Quaternion.Euler(a.x, a.yaw, a.z);
			Vector3 now = tr.eulerAngles;
			AcroPitch = Wrap(now.x);
			AcroRoll = Wrap(now.z);
			return true;
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] acrobatics failed and stopped: " + ex.Message));
			return false;
		}
	}
}
