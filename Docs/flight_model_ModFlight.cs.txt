using System;
using System.Collections.Generic;
using UnityEngine;

// v0.26.1 flight core. The game flies planes by sliding them along their nose at a speed that never depends on pitch. This adds, on top of
// the game's own movement and without replacing it:
//   Energy:      a dive gains speed, a climb loses it (gravity along the nose). In a dive the game's pull back to the wanted speed is eased.
//   Dive limit:  a plane cannot dive faster than a limit per type (fighters 125% of top speed, dive bombers 105% because of dive brakes,
//                other bombers 115%).
//   Stall:       below a minimum speed (35% of top speed for fighters, 45% for others) a plane cannot hold its climb. It lets the nose drop
//                until it has flying speed again.
//   Turn limit:  the turn rate is capped by speed. Above corner speed (heavy controls) and below it (nearly stalled) turns are slower.
// v0.26.2 adds pitch control: while a fighter lines up a shot, the nose is pulled to the angle it needs at a real pull rate
// (the game only allowed 10 degrees per second, so a fighter could never get its nose onto a target in time).
// v0.26.3 adds banking: a plane rolls into a turn at a real roll rate (fighters 90 degrees per second, others 45) and the turn only builds as
// fast as the bank allows, so turns start and finish smoothly instead of the nose swinging the moment the AI asks.
// Always on: this is the game's flying now, not an option. Every number is below and can be changed, then Compile and Save Module.
public static class ModFlight
{
	// Not a player switch. Only set if the core throws an error, so one bug cannot spam the log or break a battle (the plane then flies the original way).
	private static bool failed = false;

	// Gravity along the nose in game units per second squared (9.81 m/s2, 10 m per game unit). EnergyScale makes it stronger or weaker.
	public const float Gravity = 0.981f;
	public const float EnergyScale = 2f;

	// Share of the game's pull back toward the wanted speed that is given back while diving (0 to 1).
	public const float DiveRefund = 0.8f;

	// Turn rate factor limits (share of maxTurnRate).
	public const float CornerShare = 0.6f;
	public const float MinTurnFactor = 0.5f;
	public const float MinFastTurnFactor = 0.55f;

	private static Dictionary<int, float> startSpeed = new Dictionary<int, float>();
	private static Dictionary<int, float> lastStallLog = new Dictionary<int, float>();
	private static Dictionary<int, float> aimPitch = new Dictionary<int, float>();
	private static Dictionary<int, float> aimUntil = new Dictionary<int, float>();
	// v0.26.3: the turn the pilot is easing toward, so the AI's sudden left/right wishes become smooth stick movements.
	private static Dictionary<int, float> smoothTurn = new Dictionary<int, float>();
	public const float StickTime = 0.35f;
	private static HashSet<int> stalled = new HashSet<int>();

	// Pull rates in degrees per second.
	public const float FighterPitchRate = 40f;
	public const float OtherPitchRate = 20f;
	public const float FighterEasyRate = 25f;
	public const float OtherEasyRate = 12f;
	public const float MaxAimDown = 60f;
	public const float MaxAimUp = 50f;
	// v0.26.3 banking. Roll rate in degrees per second. The game's bank is 5 degrees per degree-per-second of turn rate, kept here.
	public const float FighterRollRate = 90f;
	public const float OtherRollRate = 45f;
	public const float BankPerTurn = 5f;
	public const float MaxBank = 80f;

	private static bool Applies(UnitAir ua)
	{
		if (failed || ua == null || ua.unitData == null || ua.isDestroyed || ua.shootDown || ua.isParked)
		{
			return false;
		}
		if (ua.unitAI == null)
		{
			return false;
		}
		Maneuvre m = ua.unitAI.currentManeuvre;
		return m == Maneuvre.None || m == Maneuvre.Break_Left || m == Maneuvre.Break_Right;
	}

	// Call as the first line of UnitAir.UpdateMovementData.
	public static void Begin(UnitAir ua)
	{
		if (failed)
		{
			return;
		}
		startSpeed[((UnityEngine.Object)ua).GetInstanceID()] = ua.currentSpeed;
	}

	// Call as the last line of UnitAir.UpdateMovementData.
	public static void End(UnitAir ua, float duration)
	{
		try
		{
			if (duration <= 0f || !Applies(ua))
			{
				return;
			}
			int id = ((UnityEngine.Object)ua).GetInstanceID();
			float before;
			if (!startSpeed.TryGetValue(id, out before))
			{
				return;
			}
			UnitData d = ua.unitData;
			float maxSpeed = d.maxSpeed;
			if (maxSpeed <= 0f)
			{
				return;
			}
			bool fighter = d.unitSubtype == UnitSubType.Fighter;

			// Nose angle in degrees, positive = nose down.
			float pitch = ((Component)ua).transform.eulerAngles.x;
			if (pitch > 180f)
			{
				pitch -= 360f;
			}
			float sinPitch = Mathf.Sin(pitch * ((float)Math.PI / 180f));

			float speed = ua.currentSpeed;

			// Energy. In a dive give back part of what the game took off to pull the plane back to its wanted speed.
			if (sinPitch > 0.1f && speed < before)
			{
				speed += (before - speed) * DiveRefund * Mathf.Clamp01(sinPitch * 2f);
			}
			speed += Gravity * EnergyScale * sinPitch * duration;

			// Dive limit.
			float limit = 1.15f;
			if (fighter)
			{
				limit = 1.25f;
			}
			else if (d.unitSubtype == UnitSubType.Dive_Bomber)
			{
				limit = 1.05f;
			}
			float diveSpeed = maxSpeed * limit;
			if (speed > diveSpeed)
			{
				speed = diveSpeed;
			}

			// Stall: keep a little above stall speed once recovered, so the nose does not flicker.
			float stall = maxSpeed * (fighter ? 0.35f : 0.45f);
			if (speed >= stall * 1.3f)
			{
				stalled.Remove(id);
			}
			if (speed < stall * 1.1f && ((Component)ua).transform.position.y > 15f)
			{
				stalled.Add(id);
				float y = ((Component)ua).transform.position.y;
				if (ua.wantedHeight > y - 10f)
				{
					ua.wantedHeight = y - 10f;
				}
				float last;
				if (!lastStallLog.TryGetValue(id, out last) || Time.time - last > 5f)
				{
					lastStallLog[id] = Time.time;
					Debug.Log((object)("[FLIGHT] " + d.unitName + " id " + id + " stalling at speed " + speed.ToString("F1") + " (stall " + stall.ToString("F1") + "): nose dropping"));
				}
			}
			if (speed < 0f)
			{
				speed = 0f;
			}
			ua.currentSpeed = speed;
			ua.percentMaxSpeed = speed / maxSpeed * ua.currentSpeedPenalty;

			// Turn limit by speed.
			float corner = maxSpeed * CornerShare;
			float factor = 1f;
			if (speed > corner)
			{
				factor = Mathf.Clamp(corner / speed, MinFastTurnFactor, 1f);
			}
			else if (speed < corner)
			{
				float t = Mathf.Clamp01((speed - stall) / Mathf.Max(0.01f, corner - stall));
				factor = Mathf.Lerp(MinTurnFactor, 1f, t);
			}
			float cap = d.maxTurnRate * factor;
			if (ua.currentTurnRate > cap)
			{
				ua.currentTurnRate = cap;
			}
			else if (ua.currentTurnRate < -cap)
			{
				ua.currentTurnRate = -cap;
			}
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] flight core failed and stopped: " + ex.Message));
		}
	}

	// Called every tick by ModDogfight.GunRun: pitchDown is the nose angle that points at the aim point, in degrees, positive = nose down.
	public static void Aim(UnitAir ua, float pitchDown)
	{
		if (failed || ua == null)
		{
			return;
		}
		int id = ((UnityEngine.Object)ua).GetInstanceID();
		aimPitch[id] = pitchDown;
		aimUntil[id] = Time.time + 0.5f;
	}

	// Text for the flight log: the pitch the pilot is aiming for, or nothing.
	public static string AimText(UnitAir ua)
	{
		int id = ((UnityEngine.Object)ua).GetInstanceID();
		float until;
		float p;
		if (aimUntil.TryGetValue(id, out until) && Time.time <= until && aimPitch.TryGetValue(id, out p))
		{
			return " aim " + p.ToString("F1");
		}
		return "";
	}

	// Replaces the pitch line of UnitAir.ApplyFlightControls. current = the plane's pitch now, heightGap = the game's own wish (positive = nose down).
	public static float Pitch(UnitAir ua, float current, float heightGap, float duration)
	{
		try
		{
			if (failed || ua.unitData == null)
			{
				return Mathf.SmoothDampAngle(current, heightGap, ref ua.pitchVelocity, 1f, 10f, duration);
			}
			int id = ((UnityEngine.Object)ua).GetInstanceID();
			bool fighter = ua.unitData.unitSubtype == UnitSubType.Fighter;
			float until;
			float wanted;
			float y = ((Component)ua).transform.position.y;
			if (aimUntil.TryGetValue(id, out until) && Time.time <= until && aimPitch.TryGetValue(id, out wanted) && Applies(ua))
			{
				// Do not fly into the sea: the lower the plane, the less it may dive.
				float maxDown = MaxAimDown;
				if (y < 50f)
				{
					maxDown = Mathf.Clamp((y - 10f) * 1.5f, 0f, MaxAimDown);
				}
				float target = Mathf.Clamp(wanted, -MaxAimUp, maxDown);
				if (y < 12f)
				{
					target = Mathf.Min(target, -5f);
				}
				if (stalled.Contains(id))
				{
					target = Mathf.Max(target, 12f);
				}
				ua.pitchVelocity = 0f;
				return Mathf.MoveTowards(current, target, (fighter ? FighterPitchRate : OtherPitchRate) * duration);
			}
			return Mathf.SmoothDampAngle(current, heightGap, ref ua.pitchVelocity, 1f, fighter ? FighterEasyRate : OtherEasyRate, duration);
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] pitch control failed and stopped: " + ex.Message));
			return Mathf.SmoothDampAngle(current, heightGap, ref ua.pitchVelocity, 1f, 10f, duration);
		}
	}

	// v0.26.3. Replaces the roll line of UnitAir.ApplyFlightControls. Rolls toward the bank the wanted turn needs, at a real roll rate.
	public static float Roll(UnitAir ua, float currentRoll, float currentTurnRate, float wantedTurnRate, float duration)
	{
		try
		{
			if (failed || ua.unitData == null)
			{
				return Mathf.MoveTowards(currentRoll, currentTurnRate * -BankPerTurn, 40f * duration);
			}
			bool fighter = ua.unitData.unitSubtype == UnitSubType.Fighter;
			float eased = wantedTurnRate;
			if (!smoothTurn.TryGetValue(((UnityEngine.Object)ua).GetInstanceID(), out eased))
			{
				eased = wantedTurnRate;
			}
			float target = Mathf.Clamp(eased * -BankPerTurn, -MaxBank, MaxBank);
			return Mathf.MoveTowards(currentRoll, target, (fighter ? FighterRollRate : OtherRollRate) * duration);
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] roll control failed and stopped: " + ex.Message));
			return Mathf.MoveTowards(currentRoll, currentTurnRate * -BankPerTurn, 40f * duration);
		}
	}

	// v0.26.3. Replaces the turn rate line of UnitAir.ApplyFlightControls. The turn rate still moves toward the wish at the plane's own rate,
	// but cannot be faster than the current bank supports (plus 1 degree per second so a level plane can start rolling into a turn).
	public static float TurnRate(UnitAir ua, float current, float wanted, float currentRoll, float duration)
	{
		float next = Mathf.MoveTowards(current, wanted, duration * ua.unitData.turnRate * Time.timeScale);
		try
		{
			if (failed)
			{
				return next;
			}
			int id = ((UnityEngine.Object)ua).GetInstanceID();
			float eased;
			if (!smoothTurn.TryGetValue(id, out eased))
			{
				eased = current;
			}
			eased = Mathf.Lerp(eased, wanted, 1f - Mathf.Exp(-duration / StickTime));
			smoothTurn[id] = eased;
			next = Mathf.MoveTowards(current, eased, duration * ua.unitData.turnRate * Time.timeScale);
			float supported = -currentRoll / BankPerTurn;
			float sign = Mathf.Sign(next);
			float allowed = 1f;
			if (sign * supported > 0f)
			{
				allowed = Mathf.Abs(supported) + 1f;
			}
			if (Mathf.Abs(next) > allowed && Mathf.Abs(next) > Mathf.Abs(current))
			{
				next = sign * allowed;
			}
			return next;
		}
		catch (Exception ex)
		{
			failed = true;
			Debug.LogError((object)("[FLIGHT] bank control failed and stopped: " + ex.Message));
			return next;
		}
	}
}
