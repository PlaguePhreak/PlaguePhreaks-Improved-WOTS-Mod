	// True when a notification of this category (1 Contacts ... 7 Other) goes into the log.
	public static bool LogCategory(int cat)
	{
		if (!Get(INotify))
		{
			return false;
		}
		if (cat < 1 || cat > 7)
		{
			return true;
		}
		return Get(10 + cat);
	}

	public static bool Get(int i)
	{
		if (!loaded)
		{
			Load();
		}
		if (i < LockedCount)
		{
			string s = CampaignLock();
			if (s != null)
			{
				return s[i] == '1';
			}
		}
		return values[i];
	}

	// The locked settings of the running campaign, saved with it (CampaignData.modLocked).
	// null when no campaign is running. An old save without them takes the current settings.
	private static string CampaignLock()
	{
		try
		{
			CampaignManager cm = CampaignManager.instance;
			if (cm == null || cm.campaignData == null || string.IsNullOrEmpty(cm.campaignData.campaignID))
			{
				return null;
			}
			CampaignData cd = cm.campaignData;
			if (string.IsNullOrEmpty(cd.modLocked) || cd.modLocked.Length < LockedCount)
			{
				char[] c = new char[LockedCount];
				for (int i = 0; i < LockedCount; i++)
				{
					c[i] = values[i] ? '1' : '0';
				}
				cd.modLocked = new string(c);
			}
			return cd.modLocked;
		}
		catch
		{
			return null;
		}
	}

	private static string FilePath()
	{
		return Path.Combine(Application.persistentDataPath, "modoptions.txt");
	}

	public static void Load()
	{
		loaded = true;
		try
		{
			string path = FilePath();
			if (!File.Exists(path))
			{
				return;
			}
			string[] lines = File.ReadAllLines(path);
			for (int l = 0; l < lines.Length; l++)
			{
				int eq = lines[l].IndexOf('=');
				if (eq < 1)
				{
					continue;
				}
				string key = lines[l].Substring(0, eq).Trim();
				string val = lines[l].Substring(eq + 1).Trim();
				for (int i = 0; i < Count; i++)
				{
					if (keys[i] == key)
					{
						values[i] = val != "0";
					}
				}
			}
		}
		catch (Exception ex)
		{
			Debug.LogError("[MODOPT] load failed: " + ex.Message);
		}
	}

	public static void Save()
	{
		try
		{
			List<string> lines = new List<string>();
			for (int i = 0; i < Count; i++)
			{
				lines.Add(keys[i] + "=" + (values[i] ? "1" : "0"));
			}
			File.WriteAllLines(FilePath(), lines.ToArray());
		}
		catch (Exception ex)
		{
			Debug.LogError("[MODOPT] save failed: " + ex.Message);
		}
	}

	public static void ApplyNights()
	{
		try
		{
			if (Config.instance == null)
			{
				return;
			}
			bool on = Get(IDark);
			Config.instance.visibilityByHour = (float[])(on ? darkTable : stockTable).Clone();
			Config.instance.nightEncounterMaxRange = on ? 10f : 15f;
		}
		catch (Exception ex)
		{
			Debug.LogError("[MODOPT] nights failed: " + ex.Message);
		}
	}

	private static Button loadBtn;

	// Called at the end of PauseMenu.OpenPauseMenu.
	public static void BuildPauseButtons(PauseMenu pm)
	{
		try
		{
			if (pm == null || pm.saveButton == null)
			{
				return;
			}
			if (loadBtn == null || settingsBtn == null || modBtn == null)
			{
				loadBtn = Clone(pm.saveButton, "Load Game", new UnityAction(AskLoad));
				settingsBtn = Clone(pm.saveButton, "Settings", new UnityAction(OpenGameSettings));
				modBtn = Clone(pm.saveButton, "Mod Options", new UnityAction(OpenPanelFromPause));
				PlaceAfter(pm.saveButton, new Button[3] { loadBtn, settingsBtn, modBtn });
			}
			// No loading from inside the tutorial or the museum (the same places Save is hidden).
			loadBtn.gameObject.SetActive(pm.saveButton.gameObject.activeSelf);
		}
		catch (Exception ex)
		{
			Debug.LogError("[MODOPT] pause buttons failed: " + ex.ToString());
		}
	}

	private static Button Clone(Button template, string label, UnityAction action)
	{
		GameObject go = (GameObject)UnityEngine.Object.Instantiate(template.gameObject, template.transform.parent);
		go.name = "Mod " + label;
		go.SetActive(true);
		Button b = go.GetComponent<Button>();
		b.onClick = new Button.ButtonClickedEvent();
		b.onClick.AddListener(action);
		// The button can hold more than one Text (a shadow or duplicate label), so set them all.
		Text[] texts = go.GetComponentsInChildren<Text>(true);
		for (int i = 0; i < texts.Length; i++)
		{
			texts[i].text = label.ToUpper();
		}
		if (texts.Length == 0)
		{
			Debug.LogError("[MODOPT] no Text found on the button to copy, the label could not be set");
		}
		return b;
	}

	// Puts the new buttons right under "anchor" and, when the buttons are not in a layout group,
	// spaces all the buttons of that list evenly.
	private static void PlaceAfter(Button anchor, Button[] added)
	{
		Transform parent = anchor.transform.parent;
		int at = anchor.transform.GetSiblingIndex();
		for (int i = 0; i < added.Length; i++)
		{
			added[i].transform.SetSiblingIndex(at + 1 + i);
		}
		if (parent.GetComponent<LayoutGroup>() != null)
		{
			return;
		}
		List<RectTransform> rows = new List<RectTransform>();
		for (int c = 0; c < parent.childCount; c++)
		{
			Transform ch = parent.GetChild(c);
			if (ch.GetComponent<Button>() != null)
			{
				rows.Add(ch.GetComponent<RectTransform>());
			}
		}
		if (rows.Count < 3)
		{
			return;
		}
		Vector2 first = rows[0].anchoredPosition;
		float step = rows[1].anchoredPosition.y - first.y;
		if (Mathf.Abs(step) < 1f)
		{
			step = -rows[0].rect.height * 1.1f;
		}
		for (int r = 0; r < rows.Count; r++)
		{
			rows[r].anchoredPosition = new Vector2(first.x, first.y + step * (float)r);
		}
	}

	// ---------------------------------------------------------------- load game from the pause menu

	private static void AskLoad()
	{
		// Reuses the pause menu's confirmation panel (action 2 is handled in PauseMenu.OnConfirmation).
		PauseMenu.instance.OpenConfirmation(2);
	}

	// Called from PauseMenu.OnConfirmation (action 2): back to the main menu, which then opens the load list.
	public static void LoadFromPause()
	{
		PauseMenu pm = PauseMenu.instance;
		bool campaign = pm.isStratMap || (EngagementData.instance != null && EngagementData.instance.isCampaignEngagement);
		pendingLoad = campaign ? 1 : 2;
		Time.timeScale = 1f;
		Map.instance.ClearMap();
		Utilities.ReloadMain();
	}

	// ---------------------------------------------------------------- game settings from the pause menu

	private static void OpenGameSettings()
	{
		try
		{
			MenuSystem ms = MenuSystem.instance;
			if (ms == null || ms.optionsInterface == null)
			{
				return;
			}
			fromPause = true;
			PauseMenu.instance.gameObject.SetActive(false);
			ms.menusystemBackground.gameObject.SetActive(false);
			ms.gameObject.SetActive(true);
			ms.mainInterface.SetActive(false);
			ms.optionsInterface.SetActive(true);
			string path = string.Empty;
			Transform t = ms.optionsInterface.transform;
			while (t != null)
			{
				if (!t.gameObject.activeSelf)
				{
					t.gameObject.SetActive(true);
				}
				path = t.name + "/" + path;
				t = t.parent;
			}
			Canvas c = ms.optionsInterface.GetComponentInParent<Canvas>();
			if (c != null && c.rootCanvas.sortingOrder < 100)
			{
				c.rootCanvas.sortingOrder = 100;
			}
			Debug.Log("[MODOPT] options path " + path + " visible " + ms.optionsInterface.activeInHierarchy);
			if (!ms.optionsInterface.activeInHierarchy)
			{
				fromPause = false;
				PauseMenu.instance.gameObject.SetActive(true);
			}
		}
		catch (Exception ex)
		{
			fromPause = false;
			PauseMenu.instance.gameObject.SetActive(true);
			Debug.LogError("[MODOPT] settings failed: " + ex.ToString());
		}
	}

	// First line of MenuSystem.BackToMain. True = handled here (the options screen was opened from the pause menu).
	public static bool OnBack()
	{
		if (!fromPause)
		{
			return false;
		}
		fromPause = false;
		try
		{
			MenuSystem ms = MenuSystem.instance;
			ms.optionsInterface.SetActive(false);
			ms.mainInterface.SetActive(true);
			ms.gameObject.SetActive(false);
			PauseMenu.instance.gameObject.SetActive(true);
		}
		catch (Exception ex)
		{
			Debug.LogError("[MODOPT] back failed: " + ex.ToString());
		}
		return true;
	}

	// ---------------------------------------------------------------- main menu

	// Last line of MenuSystem.Start.
	public static void AfterMenuStart()
	{
		try
		{
			Load();
			ApplyNights();
			BuildMainMenuButton();
			int p = pendingLoad;
			pendingLoad = 0;
			if (p == 1)
			{
				MenuSystem.instance.NewCampaign(true);
			}
			else if (p == 2)
			{
				MenuSystem.instance.OpenMission(false);
				MenuSystem.instance.LoadSelection();
			}
		}
		catch (Exception ex)
		{
			Debug.LogError("[MODOPT] menu start failed: " + ex.ToString());
		}
	}

	private static void BuildMainMenuButton()
	{
		MenuSystem ms = MenuSystem.instance;
		if (ms == null || ms.mainInterface == null)
		{
			return;
		}
		Button[] all = ms.mainInterface.GetComponentsInChildren<Button>(true);
		if (all.Length == 0)
		{
			return;
		}
		Button anchor = all[all.Length - 1];
		for (int i = 0; i < all.Length; i++)
		{
			Text t = all[i].GetComponentInChildren<Text>(true);
			if (t != null && t.text.ToUpper().Contains("OPTION"))
			{
				anchor = all[i];
				break;
			}
		}
		Button mod = Clone(anchor, "Mod Options", new UnityAction(OpenPanelFromMain));
		PlaceAfter(anchor, new Button[1] { mod });
	}

	private static void OpenPanelFromMain()
	{
		Button[] all = MenuSystem.instance.mainInterface.GetComponentsInChildren<Button>(true);
		for (int i = 0; i < all.Length; i++)
		{
			if (all[i].gameObject.name == "Mod Mod Options")
			{
				OpenPanel(all[i]);
				return;
			}
		}
	}

	private static void OpenPanel(Button from)
	{
		try
		{
			if (panel == null)
			{
				Canvas c = from.GetComponentInParent<Canvas>();
				if (c == null)
				{
					Debug.LogError("[MODOPT] no canvas found for the panel");
					return;
				}
				BuildPanel(c.rootCanvas.transform);
			}
			Load();
			UpdateRows();
			panel.transform.SetAsLastSibling();
			panel.SetActive(true);
		}
		catch (Exception ex)
		{
			Debug.LogError("[MODOPT] open panel failed: " + ex.ToString());
		}
	}

	// Called from the first line of PauseMenu.Resume.
	public static void Close()
	{
		if (panel != null)
		{
			panel.SetActive(false);
		}
	}

	private static void Flip(int i)
	{
		if (i < LockedCount && CampaignLock() != null)
		{
			return;
		}
		values[i] = !values[i];
		Save();
		if (i == IDark)
		{
			ApplyNights();
		}
		UpdateRows();
	}

	private static void UpdateRows()
	{
		bool inCampaign = CampaignLock() != null;
		for (int i = 0; i < Count; i++)
		{
			if (stateTexts[i] == null)
			{
				continue;
			}
			bool on = Get(i);
			bool isLocked = inCampaign && i < LockedCount;
			stateTexts[i].text = on ? "ON" : "OFF";
			stateTexts[i].color = isLocked ? new Color(0.6f, 0.6f, 0.6f, 1f) : Color.white;
			stateBacks[i].color = isLocked ? new Color(0.25f, 0.25f, 0.25f, 1f) : (on ? lit : dark);
			labelTexts[i].text = names[i] + (i < LockedCount ? " *" : string.Empty) + (isLocked ? "  (locked)" : string.Empty);
			labelTexts[i].color = isLocked ? new Color(0.6f, 0.6f, 0.6f, 1f) : Color.white;
		}
		if (noteText != null)
		{
			noteText.text = inCampaign
				? "Settings marked (locked) cannot be changed during a campaign, because changing them can break a save. Change them from the main menu before you start a new campaign."
				: "Settings marked * are locked once a campaign has started, because changing them can break a save. Everything else can be changed at any time.";
		}
	}

			row = AddSetting(row, i2);
		}
		row = AddHeader(row, "Notification log: what goes into the log (the pop-ups still appear)");
		for (int i3 = 11; i3 < 18; i3++)
		{
			row = AddSetting(row, i3);
		}
		contentRect.sizeDelta = new Vector2(0f, (float)row * RowH);
		panel.SetActive(false);
	}

	private static int AddHeader(int row, string text)
	{
		RectTransform rt = NewRect("Header", contentRect);
		rt.anchorMin = new Vector2(0f, 1f);
		rt.anchorMax = new Vector2(1f, 1f);
		rt.pivot = new Vector2(0.5f, 1f);
		rt.anchoredPosition = new Vector2(0f, -(float)row * RowH);
		rt.sizeDelta = new Vector2(0f, RowH);
		Text t = NewText(rt, fontSize, TextAnchor.LowerLeft, gold);
		t.text = text;
		return row + 1;
	}

	private static int AddSetting(int row, int index)
	{
		RectTransform rt = NewRect("Row " + keys[index], contentRect);
		rt.anchorMin = new Vector2(0f, 1f);
		rt.anchorMax = new Vector2(1f, 1f);
		rt.pivot = new Vector2(0.5f, 1f);
		rt.anchoredPosition = new Vector2(0f, -(float)row * RowH);
		rt.sizeDelta = new Vector2(0f, RowH);
		Image rowBack = rt.gameObject.AddComponent<Image>();
		rowBack.color = (row % 2 == 0) ? new Color(1f, 1f, 1f, 0.05f) : new Color(1f, 1f, 1f, 0f);
		Button rowBtn = rt.gameObject.AddComponent<Button>();
		rowBtn.targetGraphic = rowBack;
		int captured = index;
		rowBtn.onClick.AddListener(new UnityAction(delegate { Flip(captured); }));
		RectTransform lbl = NewRect("Label", rt);
		Stretch(lbl, 12f, 2f, 90f, 2f);
		labelTexts[index] = NewText(lbl, fontSize, TextAnchor.MiddleLeft, Color.white);
		labelTexts[index].horizontalOverflow = HorizontalWrapMode.Overflow;
		RectTransform st = NewRect("State", rt);
		st.anchorMin = new Vector2(1f, 0.5f);
		st.anchorMax = new Vector2(1f, 0.5f);
		st.pivot = new Vector2(1f, 0.5f);
		st.anchoredPosition = new Vector2(-6f, 0f);
		st.sizeDelta = new Vector2(64f, 26f);
		stateBacks[index] = st.gameObject.AddComponent<Image>();
		stateBacks[index].color = lit;
		stateBacks[index].raycastTarget = false;
		RectTransform stText = NewRect("Text", st);
		Stretch(stText, 0f, 0f, 0f, 0f);
		stateTexts[index] = NewText(stText, fontSize, TextAnchor.MiddleCenter, Color.white);
		return row + 1;
	}

	private static RectTransform NewRect(string name, Transform parent)
	{
		GameObject go = new GameObject(name, typeof(RectTransform));
		go.transform.SetParent(parent, false);
		return go.GetComponent<RectTransform>();
	}

	private static void Stretch(RectTransform rt, float left, float bottom, float right, float top)
	{
		rt.anchorMin = Vector2.zero;
		rt.anchorMax = Vector2.one;
		rt.offsetMin = new Vector2(left, bottom);
		rt.offsetMax = new Vector2(-right, -top);
	}

	private static Text NewText(RectTransform rt, int size, TextAnchor anchor, Color colour)
	{
		Text t = rt.gameObject.AddComponent<Text>();
		t.font = font;
		t.fontSize = size;
		t.alignment = anchor;
		t.color = colour;
		t.supportRichText = false;
		t.horizontalOverflow = HorizontalWrapMode.Wrap;
		t.verticalOverflow = VerticalWrapMode.Truncate;
		t.raycastTarget = false;
		return t;
	}
