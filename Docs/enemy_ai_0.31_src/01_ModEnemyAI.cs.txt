	public static int CampaignStep()
	{
		Load();
		try
		{
			return Mathf.Clamp(LanguageManager.instance.campaignDifficultyDropdown.value, 0, 4);
		}
		catch
		{
			return 2;
		}
	}

	public static int CombatStep()
	{
		Load();
		try
		{
			return Mathf.Clamp(LanguageManager.instance.combatDifficultyDropdown.value, 0, 4);
		}
		catch
		{
			return 2;
		}
	}

	public static void Log(string text)

		Debug.Log((object)("[EAI] " + text));

	public static string FilePath()
	{
		return Path.Combine(Path.Combine(ModAssetPath.Override(), "ModAssets"), "enemy_ai.txt");
	}

	public static void Load()
	{
		if (loaded)
		{
			return;
		}
		loaded = true;
		try
		{
			string path = FilePath();
			if (!File.Exists(path))
			{
				Log("no " + path + ", using the built-in table");
				return;
			}
			Dictionary<string, float[]> dictionary = Knobs();
			string[] array = File.ReadAllLines(path);
			int num = 0;
			for (int i = 0; i < array.Length; i++)
			{
				string text = array[i].Trim();
				int num2 = text.IndexOf('#');
				if (num2 >= 0)
				{
					text = text.Substring(0, num2).Trim();
				}
				int num3 = text.IndexOf('=');
				if (num3 < 1)
				{
					continue;
				}
				string key = text.Substring(0, num3).Trim();
				if (!dictionary.TryGetValue(key, out var value))
				{
					Log("enemy_ai.txt: unknown name '" + key + "'");
					continue;
				}
				string[] array2 = text.Substring(num3 + 1).Split(new char[1] { ',' });
				for (int j = 0; j < array2.Length && j < 5; j++)
				{
					if (float.TryParse(array2[j].Trim(), System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out var result))
					{
						value[j] = result;
					}
				}
				num++;
			}
			Log("enemy_ai.txt read, " + num + " values changed");
		}
		catch (Exception ex)
		{
			Debug.LogError((object)("[EAI] enemy_ai.txt failed: " + ex.Message));
		}
	}

	// ---- search plane types (fly searches only, never sent on strikes) ----
	public static readonly string[] SearchTypes = new string[7] { "h6k", "emily", "j1n1", "pby", "pbm", "sunderland", "pby5" };

	public static bool IsSearchType(string prefab)
	{
		if (string.IsNullOrEmpty(prefab))
		{
			return false;
		}
		for (int i = 0; i < SearchTypes.Length; i++)
		{
			if (SearchTypes[i] == prefab)
			{
				return true;
			}
		}
		try
		{
			if (EngagementManager.instance.unitDataDictionary.TryGetValue(prefab, out var value) && value.unitSubtype == UnitSubType.Scout)
			{
				return true;
			}
		}
		catch
		{
		}
		return false;
	}

	public static bool newCampaignAir;

	public static void AfterInit(CampaignData data, string filePath)
	{
		try
		{
			if (!string.IsNullOrEmpty(filePath) || data == null || !On)
			{
				return;
			}
			string[] array = null;
			int[] array2 = null;
			int[] array3 = null;
			string nation = null;
			if (data.nations1 != null && data.nations1.Contains("japan"))
			{
				array = JapanTypes;
				array2 = JapanNumbers;
				array3 = JapanReplenish;
				nation = "japan";
			}
			else if (data.nations1 != null && data.nations1.Contains("usa"))
			{
				array = UsaTypes;
				array2 = UsaNumbers;
				array3 = UsaReplenish;
				nation = "usa";
			}
			if (array == null)
			{
				Log("new campaign: enemy nations not Japan or USA, airfield aircraft left as they are");
				return;
			}
			List<int> list = new List<int>();
			for (int i = 0; i < array.Length; i++)
			{
				if (Ready(array[i], nation))
				{
					list.Add(i);
				}
				else
				{
					Log("new campaign: aircraft '" + array[i] + "' is not installed, left out");
				}
			}
			if (list.Count < 2 || list[0] != 0)
			{
				Log("new campaign: too few new aircraft installed, airfield aircraft left as they are");
				return;
			}
			List<string> list2 = new List<string>();
			List<int> list3 = new List<int>();
			int[] array4 = new int[Mathf.Max(8, list.Count)];
			for (int j = 0; j < list.Count; j++)
			{
				list2.Add(array[list[j]]);
				array4[j] = array3[list[j]];
			}
			for (int k = 0; k < 5; k++)
			{
				for (int l = 0; l < list.Count; l++)
				{
					list3.Add(array2[k * array.Length + list[l]]);
				}
			}
			data.aircraft1 = list2;
			data.aircraft1Numbers = list3;
			data.replenishDailyAircraft1 = array4;
			newCampaignAir = true;
			Log("new campaign: enemy airfields now fly " + string.Join(", ", list2.ToArray()));
		}
		catch (Exception ex)
		{
			Debug.LogError((object)("[EAI] AfterInit failed: " + ex));
		}
	}

	// the campaign's airUnits.txt may not list the new types: add an entry copied from a stock one so replenish costs work
	private static bool Ready(string type, string nation)
	{
		EngagementManager instance = EngagementManager.instance;
		if (!instance.unitDataDictionary.ContainsKey(type))
		{
			return false;
		}
		if (instance.unitAirDictionary.ContainsKey(type))
		{
			return true;
		}
		UnitListData unitListData = new UnitListData();
		unitListData.nation = nation;
		unitListData.unitID = type;
		unitListData.isAvailable = false;
		unitListData.available = new List<int> { 1, 1, 1930 };
		unitListData.payloadDisplayNames = new List<string>();
		unitListData.payloadIndexes = new List<string>();
		unitListData.endurance = new List<float>();
		unitListData.notAtLocation = new List<string>();
		unitListData.orderAvailable = new List<int>();
		unitListData.sunkYield = new int[5];
		unitListData.resourceCost = new int[4] { 0, 12, 3, 4 };
		instance.unitAirDictionary[type] = unitListData;
		Log("added '" + type + "' to the campaign aircraft list (not in its airUnits.txt)");
		return true;
	}

	// first campaign tick after a new campaign: bases with their own enemy aircraft list switch to the new one
	public static void FixLocations()
	{
		if (!newCampaignAir)
		{
			return;
		}
		newCampaignAir = false;
		try
		{
			CampaignManager instance = CampaignManager.instance;
			int num = 0;
			for (int i = 0; i < instance.landMapLocations.Count; i++)
			{
				MapLocation mapLocation = instance.landMapLocations[i];
				if (mapLocation.aircraft1 != null && mapLocation.aircraft1.Count > 0)
				{
					mapLocation.aircraft1 = new List<string>();
					mapLocation.aircraft1Numbers = new List<int>();
					mapLocation.replenishDailyAircraft1 = new int[0];
					num++;
				}
				if (mapLocation.currentFaction == 1)
				{
					int count = instance.GetAirTypesAtLocation(mapLocation).Count;
					mapLocation.inTheAir = new List<int>();
					mapLocation.destroyedAir = new List<int>();
					for (int j = 0; j < count; j++)
					{
						mapLocation.inTheAir.Add(0);
						mapLocation.destroyedAir.Add(0);
					}
				}
			}
			Log("new campaign: enemy bases set to the new aircraft list (" + num + " had their own list)");
		}
		catch (Exception ex)
		{
			Debug.LogError((object)("[EAI] FixLocations failed: " + ex));
		}
	}

	// ---- floatplane scouts at every enemy airfield (0.31.6) ----
	// planes per airfield level 1-5 for the added floatplane type
	private static readonly int[] FloatNumbers = new int[5] { 2, 3, 4, 6, 6 };

	// well known floatplane and scout prefabs, tried first; any other Scout-type plane of the right nation is used after them
	private static readonly string[] FloatPrefer = new string[10] { "e13a", "f1m", "e7k", "e8n", "e16a", "os2u", "soc", "so3c", "kingfisher", "seagull" };

	public static bool IsScoutPlane(string prefab)
	{
		if (IsSearchType(prefab))
		{
			return true;
		}
		UnitData value;
		return !string.IsNullOrEmpty(prefab) && EngagementManager.instance.unitDataDictionary.TryGetValue(prefab, out value) && value.unitSubtype == UnitSubType.Scout;
	}

	private static string FindFloatplane(int nation)
	{
		List<string> list = new List<string>();
		foreach (KeyValuePair<string, UnitData> item in EngagementManager.instance.unitDataDictionary)
		{
			if (item.Value != null && item.Value.unitSubtype == UnitSubType.Scout && item.Value.nationIndex == nation && !IsSearchType(item.Key))
			{
				list.Add(item.Key);
			}
		}
		if (list.Count == 0)
		{
			return null;
		}
		list.Sort(StringComparer.Ordinal);
		Log("floatplane scouts in the game for this side: " + string.Join(", ", list.ToArray()));
		for (int i = 0; i < FloatPrefer.Length; i++)
		{
			for (int j = 0; j < list.Count; j++)
			{
				if (list[j].ToLowerInvariant().StartsWith(FloatPrefer[i]))
				{
					return list[j];
				}
			}
		}
		return list[0];
	}

	// called once when the map starts: if the enemy's airfield list has no floatplane scout, add one (new and saved campaigns)
	public static void EnsureFloatplanes()
	{
		try
		{
			CampaignManager instance = CampaignManager.instance;
			CampaignData campaignData = instance.campaignData;
			if (campaignData.aircraft1 == null || campaignData.aircraft1.Count == 0 || campaignData.aircraft1Numbers == null || campaignData.nations1 == null)
			{
				return;
			}
			for (int i = 0; i < campaignData.aircraft1.Count; i++)
			{
				UnitData value;
				if (EngagementManager.instance.unitDataDictionary.TryGetValue(campaignData.aircraft1[i], out value) && value.unitSubtype == UnitSubType.Scout)
				{
					return;
				}
			}
			string text = (campaignData.nations1.Contains("japan") ? "japan" : (campaignData.nations1.Contains("usa") ? "usa" : null));
			if (text == null)
			{
				return;
			}
			string text2 = FindFloatplane((int)(Country)Enum.Parse(typeof(Country), text));
			if (text2 == null)
			{
				Log("no floatplane scout for " + text + " in the game; airfields scout with their flying boats only");
				return;
			}
			if (!Ready(text2, text))
			{
				return;
			}
			int count = campaignData.aircraft1.Count;
			int num = campaignData.aircraft1Numbers.Count / count;
			List<int> list = new List<int>();
			for (int j = 0; j < num; j++)
			{
				for (int k = 0; k < count; k++)
				{
					list.Add(campaignData.aircraft1Numbers[j * count + k]);
				}
				list.Add(FloatNumbers[Mathf.Clamp(j, 0, FloatNumbers.Length - 1)]);
			}
			campaignData.aircraft1 = new List<string>(campaignData.aircraft1);
			campaignData.aircraft1.Add(text2);
			campaignData.aircraft1Numbers = list;
			int[] array = new int[Mathf.Max(8, count + 1)];
			if (campaignData.replenishDailyAircraft1 != null)
			{
				Array.Copy(campaignData.replenishDailyAircraft1, array, Mathf.Min(campaignData.replenishDailyAircraft1.Length, array.Length));
			}
			array[count] = 1;
			campaignData.replenishDailyAircraft1 = array;
			int num2 = 0;
			for (int l = 0; l < instance.landMapLocations.Count; l++)
			{
				MapLocation mapLocation = instance.landMapLocations[l];
				if (mapLocation.currentFaction != 1 || (mapLocation.aircraft1 != null && mapLocation.aircraft1.Count > 0))
				{
					continue;
				}
				if (mapLocation.inTheAir != null && mapLocation.inTheAir.Count == count)
				{
					mapLocation.inTheAir.Add(0);
				}
				if (mapLocation.destroyedAir != null && mapLocation.destroyedAir.Count == count)
				{
					mapLocation.destroyedAir.Add(0);
				}
				if (mapLocation.airfieldLevel > 0)
				{
					num2++;
				}
			}
			Log("enemy airfields get " + text2 + " floatplane scouts (" + string.Join("/", Array.ConvertAll(FloatNumbers, (int n) => n.ToString())) + " by airfield level), " + num2 + " airfields");
		}
		catch (Exception ex)
		{
			Debug.LogError((object)("[EAI] floatplanes failed: " + ex));
		}
	}
